   296→        protected function _generateSeoURL($string, $wordLimit = 0)
   297→        {
   298→                $separator = '-';
   299→
   300→                if ($wordLimit != 0) {
   301→                        $wordArr = explode(' ', $string);
   302→                        $string = implode(' ', array_slice($wordArr, 0, $wordLimit));
   303→                }
   304→
   305→                $quoteSeparator = preg_quote($separator, '#');
   306→
   307→                $trans = array(
   308→                        '&.+?;' => '',
   309→                        '[^\w\d _-]' => '',
   310→                        '\s+' => $separator,
   311→                        '(' . $quoteSeparator . ')+' => $separator
   312→                );
   313→
   314→                $string = strip_tags($string);
   315→                foreach ($trans as $key => $val) {
   316→                        $string = preg_replace('#' . $key . '#i' . (UTF8_ENABLED ? 'u' : ''), $val, $string);
   317→                }
   318→
   319→                $string = strtolower($string);
   320→
   321→                return trim(trim($string, $separator));
   322→        }

   330→        public function next_available_booking_days($days, $holidays)
   331→        {
   332→                $date = date('Y-m-d'); //today date
   333→                $nextDays = array();
   334→                for ($i = 1; $i <= $days; $i++) {
   335→                        $date = date('Y-m-d', strtotime('+1 day', strtotime($date)));
   336→                        $nextDays[] = date('Y-m-d', strtotime($date));
   337→                }
   338→
   339→                $newHolydays = array();
   340→                foreach ($holidays as $holiday) {
   341→                        array_push($newHolydays, $holiday['date']);
   342→                }
   343→
   344→                $final_array = (array)array_diff($nextDays, $newHolydays);
   345→
   346→                return $final_array;
   347→        }
