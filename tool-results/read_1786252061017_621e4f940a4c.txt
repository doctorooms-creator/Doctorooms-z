   103→function getRelativeTime(isoString: string): string {
   104→  try {
   105→    return formatDistanceToNow(new Date(isoString), { addSuffix: true })
   106→  } catch {
   107→    return ''
   108→  }
   109→}
   110→
   111→function getOpdColor(percentage: number): string {
   112→  if (percentage < 80) return 'bg-emerald-500'
   113→  if (percentage <= 95) return 'bg-amber-500'
   114→  return 'bg-red-500'
   115→}

   585→                    </div>

   588→              </div>
   589→            </div>
   590→          )}
   591→        </CardContent>
   592→      </Card>

   594→      {/* Confirmation dialog */}
   595→      <AlertDialog
   596→        open={confirmDialog.open}

   599→        <AlertDialogContent>

   609→            <AlertDialogAction

   612→            >

   614→            </AlertDialogAction>
   615→          </AlertDialogFooter>
   616→        </AlertDialogContent>
   617→      </AlertDialog>
   618→    </>
   619→  )
