   104→function getRelativeTime(isoString: string): string {
   105→  try {
   106→    return formatDistanceToNow(new Date(isoString), { addSuffix: true })
   107→  } catch {
   108→    return ''
   109→  }
   110→}
   111→
   112→function getOpdColor(percentage: number): string {
   113→  if (percentage < 80) return 'bg-emerald-500'
   114→  if (percentage <= 95) return 'bg-amber-500'
   115→  return 'bg-red-500'
   116→}

   653→            </AlertDialogAction>
   654→          </AlertDialogFooter>

   656→      </AlertDialog>
   657→    </>
   658→  )
