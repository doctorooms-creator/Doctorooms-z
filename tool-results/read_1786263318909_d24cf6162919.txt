   108→function formatFileSize(bytes: number) {
   109→  if (!bytes || bytes === 0) return null
   110→  const k = 1024
   111→  const sizes = ['B', 'KB', 'MB', 'GB']
   112→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   113→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`
   114→}
