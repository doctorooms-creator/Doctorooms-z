   181→
   182→function isAbnormal(param: string, value: number): boolean {
   183→  const T = VITAL_THRESHOLDS as Record<string, Record<string, number>>
   184→  const t = T[param]
   185→  if (!t) return false
   186→  if (param === 'spo2') return value < t.warning
   187→  if (param === 'bpSystolic') return value < (t.criticalLow ?? 999) || value > (t.warningHigh ?? 999)
   188→  if (param === 'bpDiastolic') return value > (t.warningHigh ?? 999)
   189→  if (param === 'pulse') return value < (t.criticalLow ?? 999) || value > (t.warningHigh ?? 999)
   190→  if (param === 'temperature') return value > (t.warningHigh ?? 999)
   191→  if (param === 'respiratoryRate') return value < (t.criticalLow ?? 999) || value > (t.criticalHigh ?? 999)
   192→  return false
   193→}

   195→function getMedStatusBadge(status: string) {
   196→  switch (status) {
   197→    case 'Given':
   198→      return <Badge className="bg-emerald-100 text-emerald-700 hover:bg-emerald-100 dark:bg-emerald-950/50 dark:text-emerald-400">Given</Badge>
   199→    case 'Pending':
   200→      return <Badge className="bg-amber-100 text-amber-700 hover:bg-amber-100 dark:bg-amber-950/50 dark:text-amber-400">Pending</Badge>
   201→    case 'Overdue':
   202→      return <Badge className="bg-red-100 text-red-700 hover:bg-red-100 dark:bg-red-950/50 dark:text-red-400">Overdue</Badge>
   203→    case 'Missed':
   204→      return <Badge className="bg-red-100 text-red-700 hover:bg-red-100 dark:bg-red-950/50 dark:text-red-400">Missed</Badge>
   205→    case 'Refused':
   206→      return <Badge className="bg-orange-100 text-orange-700 hover:bg-orange-100 dark:bg-orange-950/50 dark:text-orange-400">Refused</Badge>
   207→    case 'Skipped':
   208→      return <Badge className="bg-slate-100 text-slate-700 hover:bg-slate-100 dark:bg-slate-800 dark:text-slate-400">Skipped</Badge>
   209→    case 'NotAvailable':
   210→      return <Badge className="bg-red-100 text-red-700 hover:bg-red-100 dark:bg-red-950/50 dark:text-red-400">Not Available</Badge>
   211→    default:
   212→      return <Badge variant="secondary">{status}</Badge>
   213→  }
   214→}

   216→function getRouteBadge(route: string) {
   217→  const colors: Record<string, string> = {
   218→    Oral: 'border-teal-300 text-teal-700 dark:border-teal-700 dark:text-teal-400',
   219→    IV: 'border-red-300 text-red-700 dark:border-red-700 dark:text-red-400',
   220→    IM: 'border-amber-300 text-amber-700 dark:border-amber-700 dark:text-amber-400',
   221→    SC: 'border-sky-300 text-sky-700 dark:border-sky-700 dark:text-sky-400',
   222→    Topical: 'border-emerald-300 text-emerald-700 dark:border-emerald-700 dark:text-emerald-400',
   223→    Nebulization: 'border-purple-300 text-purple-700 dark:border-purple-700 dark:text-purple-400',
   224→  }
   225→  return (
   226→    <Badge variant="outline" className={cn('text-[10px]', colors[route] || '')}>
   227→      {route}
   228→    </Badge>
   229→  )
   230→}
   231→
   232→function InfoRow({ label, value }: { label: string; value?: string | null | boolean }) {
   233→  if (value === undefined || value === null || value === '' || value === '0') return null
   234→  return (
   235→    <div className="flex items-start gap-2 text-sm">
   236→      <span className="w-32 shrink-0 text-muted-foreground">{label}</span>
   237→      <span className="font-medium">{String(value)}</span>
   238→    </div>
   239→  )
   240→}

  1020→function parseResultSummary(resultData: string): string {
  1021→  try {
  1022→    const data = JSON.parse(resultData)
  1023→    if (typeof data === 'object' && data !== null) {
  1024→      return Object.entries(data)
  1025→        .map(([key, value]) => `${key}: ${value}`)
  1026→        .join(', ')
  1027→    }
  1028→    return String(data)
  1029→  } catch {
  1030→    return resultData || 'No data'
  1031→  }
