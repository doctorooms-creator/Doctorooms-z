     1→'use client'
     2→
     3→import { useState, useEffect } from 'react'
     4→import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
     5→import { motion, AnimatePresence } from 'framer-motion'
     6→import { Plus, Pencil, Trash2, Search, Loader2, FlaskConical, X, GripVertical, ListOrdered, IndianRupee } from 'lucide-react'
     7→import { toast } from 'sonner'
     8→
     9→import { Card, CardHeader, CardTitle, CardContent } from '@/components/ui/card'
    10→import {
    11→  Dialog,
    12→  DialogContent,
    13→  DialogHeader,
    14→  DialogTitle,
    15→  DialogFooter,
    16→  DialogDescription,
    17→} from '@/components/ui/dialog'
    18→import {
    19→  AlertDialog,
    20→  AlertDialogAction,
    21→  AlertDialogCancel,
    22→  AlertDialogContent,
    23→  AlertDialogDescription,
    24→  AlertDialogFooter,
    25→  AlertDialogHeader,
    26→  AlertDialogTitle,
    27→} from '@/components/ui/alert-dialog'
    28→import {
    29→  Table,
    30→  TableBody,
    31→  TableCell,
    32→  TableHead,
    33→  TableHeader,
    34→  TableRow,
    35→} from '@/components/ui/table'
    36→import { Button } from '@/components/ui/button'
    37→import { Input } from '@/components/ui/input'
    38→import { Label } from '@/components/ui/label'
    39→import { Badge } from '@/components/ui/badge'
    40→import { Skeleton } from '@/components/ui/skeleton'
    41→import {
    42→  Select,
    43→  SelectContent,
    44→  SelectItem,
    45→  SelectTrigger,
    46→  SelectValue,
    47→} from '@/components/ui/select'
    48→
    49→// ============ Types ============
    50→
    51→interface LabTestParameter {
    52→  paramName: string
    53→  shortCode: string
    54→  unit: string
    55→  normalMaleMin: number
    56→  normalMaleMax: number
    57→  normalFemaleMin: number
    58→  normalFemaleMax: number
    59→  normalChildMin: number
    60→  normalChildMax: number
    61→  sortOrder: number
    62→}
    63→
    64→interface LabTestMaster {
    65→  id: string
    66→  name: string
    67→  shortCode: string
    68→  category: string
    69→  description: string
    70→  specimenType: string
    71→  reportDays: number
    72→  rate: number
    73→  status: string
    74→  sortOrder: number
    75→  _count: { parameters: number }
    76→}
    77→
    78→const CATEGORIES = ['Haematology', 'Biochemistry', 'Microbiology', 'Serology', 'Urine', 'Stool', 'Histopathology', 'Cytology', 'Molecular', 'Immunology', 'Hormones', 'Thyroid', 'Other']
    79→const SPECIMEN_TYPES = ['Blood', 'Urine', 'Stool', 'Sputum', 'CSF', 'Serum', 'Plasma', 'Swab', 'Tissue', 'Other']
    80→
    81→const emptyParameter = (): LabTestParameter => ({
    82→  paramName: '',
    83→  shortCode: '',
    84→  unit: '',
    85→  normalMaleMin: 0,
    86→  normalMaleMax: 0,
    87→  normalFemaleMin: 0,
    88→  normalFemaleMax: 0,
    89→  normalChildMin: 0,
    90→  normalChildMax: 0,
    91→  sortOrder: 0,
    92→})
    93→
    94→// ============ Main Component ============
    95→
    96→export default function LabTestMasterClient() {
    97→  const queryClient = useQueryClient()
    98→  const [search, setSearch] = useState('')
    99→  const [categoryFilter, setCategoryFilter] = useState('')
   100→  const [dialogOpen, setDialogOpen] = useState(false)
   101→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   102→  const [editingId, setEditingId] = useState<string | null>(null)
   103→  const [deletingId, setDeletingId] = useState<string | null>(null)
   104→
   105→  // Form state
   106→  const [formName, setFormName] = useState('')
   107→  const [formShortCode, setFormShortCode] = useState('')
   108→  const [formCategory, setFormCategory] = useState('')
   109→  const [formDescription, setFormDescription] = useState('')
   110→  const [formSpecimenType, setFormSpecimenType] = useState('')
   111→  const [formReportDays, setFormReportDays] = useState(0)
   112→  const [formRate, setFormRate] = useState(0)
   113→  const [parameters, setParameters] = useState<LabTestParameter[]>([emptyParameter()])
   114→
   115→  // Fetch test masters
   116→  const { data, isLoading } = useQuery<{ testMasters: LabTestMaster[] }>({
   117→    queryKey: ['lab-test-masters', categoryFilter, search],
   118→    queryFn: () => {
   119→      const params = new URLSearchParams()
   120→      if (categoryFilter) params.set('category', categoryFilter)
   121→      params.set('status', 'Active')
   122→      if (search) params.set('search', search)
   123→      return fetch(`/api/lab-test-masters?${params}`).then((r) => r.json())
   124→    },
   125→  })
   126→
   127→  const testMasters = data?.testMasters || []
   128→
   129→  // Create / Update mutation
   130→  const saveMutation = useMutation({
   131→    mutationFn: async (body: Record<string, unknown>) => {
   132→      if (editingId) {
   133→        const res = await fetch(`/api/lab-test-masters/${editingId}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   134→        if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Update failed') }
   135→        return res.json()
   136→      } else {
   137→        const res = await fetch('/api/lab-test-masters', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   138→        if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Create failed') }
   139→        return res.json()
   140→      }
   141→    },
   142→    onSuccess: () => {
   143→      queryClient.invalidateQueries({ queryKey: ['lab-test-masters'] })
   144→      toast.success(editingId ? 'Test updated successfully' : 'Test created successfully')
   145→      closeDialog()
   146→    },
   147→    onError: (err: Error) => {
   148→      toast.error(err.message)
   149→    },
   150→  })
   151→
   152→  // Delete mutation
   153→  const deleteMutation = useMutation({
   154→    mutationFn: async (id: string) => {
   155→      const res = await fetch(`/api/lab-test-masters/${id}`, { method: 'DELETE' })
   156→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Delete failed') }
   157→      return res.json()
   158→    },
   159→    onSuccess: () => {
   160→      queryClient.invalidateQueries({ queryKey: ['lab-test-masters'] })
   161→      toast.success('Test deleted successfully')
   162→      setDeleteDialogOpen(false)
   163→      setDeletingId(null)
   164→    },
   165→    onError: (err: Error) => {
   166→      toast.error(err.message)
   167→    },
   168→  })
   169→
   170→  // Fetch detail when editing
   171→  const { data: detailData, isLoading: detailLoading } = useQuery<{ testMaster: LabTestMaster & { parameters: LabTestParameter[] } }>({
   172→    queryKey: ['lab-test-master-detail', editingId],
   173→    queryFn: () => fetch(`/api/lab-test-masters/${editingId}`).then((r) => r.json()),
   174→    enabled: !!editingId,
   175→  })
   176→
   177→  // Populate form on edit
   178→  useEffect(() => {
   179→    if (detailData?.testMaster && editingId) {
   180→      const t = detailData.testMaster
   181→      setFormName(t.name)
   182→      setFormShortCode(t.shortCode)
   183→      setFormCategory(t.category)
   184→      setFormDescription(t.description)
   185→      setFormSpecimenType(t.specimenType)
   186→      setFormReportDays(t.reportDays)
   187→      setFormRate(t.rate)
   188→      setParameters(t.parameters.length > 0 ? t.parameters : [emptyParameter()])
   189→    }
   190→  }, [detailData, editingId])
   191→
   192→  function openCreate() {
   193→    setEditingId(null)
   194→    setFormName('')
   195→    setFormShortCode('')
   196→    setFormCategory('')
   197→    setFormDescription('')
   198→    setFormSpecimenType('')
   199→    setFormReportDays(0)
   200→    setFormRate(0)
   201→    setParameters([emptyParameter()])
   202→    setDialogOpen(true)
   203→  }
   204→
   205→  function openEdit(id: string) {
   206→    setEditingId(id)
   207→    setDialogOpen(true)
   208→  }
   209→
   210→  function closeDialog() {
   211→    setDialogOpen(false)
   212→    setEditingId(null)
   213→  }
   214→
   215→  function handleSave() {
   216→    if (!formName.trim()) { toast.error('Test name is required'); return }
   217→
   218→    const validParams = parameters.filter((p) => p.paramName.trim() !== '')
   219→    saveMutation.mutate({
   220→      name: formName,
   221→      shortCode: formShortCode,
   222→      category: formCategory,
   223→      description: formDescription,
   224→      specimenType: formSpecimenType,
   225→      reportDays: formReportDays,
   226→      rate: formRate,
   227→      parameters: validParams.map((p, idx) => ({ ...p, sortOrder: idx })),
   228→    })
   229→  }
   230→
   231→  function addParameter() {
   232→    setParameters([...parameters, { ...emptyParameter(), sortOrder: parameters.length }])
   233→  }
   234→
   235→  function removeParameter(index: number) {
   236→    if (parameters.length <= 1) return
   237→    setParameters(parameters.filter((_, i) => i !== index))
   238→  }
   239→
   240→  function updateParameter(index: number, field: keyof LabTestParameter, value: string | number) {
   241→    const updated = [...parameters]
   242→    updated[index] = { ...updated[index], [field]: value }
   243→    setParameters(updated)
   244→  }
   245→
   246→  function getCategoryBadgeColor(cat: string) {
   247→    const colors: Record<string, string> = {
   248→      Haematology: 'bg-red-100 text-red-700 dark:bg-red-950/50 dark:text-red-400',
   249→      Biochemistry: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/50 dark:text-emerald-400',
   250→      Microbiology: 'bg-amber-100 text-amber-700 dark:bg-amber-950/50 dark:text-amber-400',
   251→      Serology: 'bg-violet-100 text-violet-700 dark:bg-violet-950/50 dark:text-violet-400',
   252→      Urine: 'bg-yellow-100 text-yellow-700 dark:bg-yellow-950/50 dark:text-yellow-400',
   253→      Thyroid: 'bg-teal-100 text-teal-700 dark:bg-teal-950/50 dark:text-teal-400',
   254→      Hormones: 'bg-pink-100 text-pink-700 dark:bg-pink-950/50 dark:text-pink-400',
   255→    }
   256→    return colors[cat] || 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   257→  }
   258→
   259→  return (
   260→    <div className="space-y-6">
   261→      {/* Header */}
   262→      <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   263→        <div className="space-y-1">
   264→          <h1 className="text-2xl font-bold tracking-tight">Lab Test Master</h1>
   265→          <p className="text-sm text-muted-foreground">Manage lab tests and their parameters</p>
   266→        </div>
   267→        <Button onClick={openCreate} className="bg-teal-600 hover:bg-teal-700">
   268→          <Plus className="mr-2 h-4 w-4" /> Add Test
   269→        </Button>
   270→      </motion.div>
   271→
   272→      {/* Filters */}
   273→      <Card>
   274→        <CardContent className="p-4">
   275→          <div className="flex flex-col gap-3 sm:flex-row">
   276→            <div className="relative flex-1">
   277→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   278→              <Input
   279→                placeholder="Search tests..."
   280→                value={search}
   281→                onChange={(e) => setSearch(e.target.value)}
   282→                className="pl-9"
   283→              />
   284→            </div>
   285→            <Select value={categoryFilter} onValueChange={(v) => setCategoryFilter(v === '_all' ? '' : v)}>
   286→              <SelectTrigger className="w-full sm:w-48">
   287→                <SelectValue placeholder="All Categories" />
   288→              </SelectTrigger>
   289→              <SelectContent>
   290→                <SelectItem value="_all">All Categories</SelectItem>
   291→                {CATEGORIES.map((c) => (
   292→                  <SelectItem key={c} value={c}>{c}</SelectItem>
   293→                ))}
   294→              </SelectContent>
   295→            </Select>
   296→          </div>
   297→        </CardContent>
   298→      </Card>
   299→
   300→      {/* Table */}
   301→      <Card>
   302→        <CardContent className="p-0">
   303→          {isLoading ? (
   304→            <div className="space-y-3 p-4">
   305→              {Array.from({ length: 5 }).map((_, i) => (
   306→                <Skeleton key={i} className="h-12 w-full" />
   307→              ))}
   308→            </div>
   309→          ) : testMasters.length === 0 ? (
   310→            <div className="flex flex-col items-center justify-center py-12">
   311→              <FlaskConical className="mb-3 h-12 w-12 text-muted-foreground/40" />
   312→              <p className="text-sm text-muted-foreground">No lab tests found</p>
   313→              <p className="text-xs text-muted-foreground">Click &quot;Add Test&quot; to create one</p>
   314→            </div>
   315→          ) : (
   316→            <div className="max-h-[480px] overflow-y-auto">
   317→              <Table>
   318→                <TableHeader>
   319→                  <TableRow>
   320→                    <TableHead className="w-12">#</TableHead>
   321→                    <TableHead>Name</TableHead>
   322→                    <TableHead className="hidden sm:table-cell">Code</TableHead>
   323→                    <TableHead className="hidden md:table-cell">Category</TableHead>
   324→                    <TableHead className="hidden lg:table-cell">Specimen</TableHead>
   325→                    <TableHead className="hidden md:table-cell text-right">Rate</TableHead>
   326→                    <TableHead className="text-center">Params</TableHead>
   327→                    <TableHead className="text-right">Actions</TableHead>
   328→                  </TableRow>
   329→                </TableHeader>
   330→                <TableBody>
   331→                  {testMasters.map((test, idx) => (
   332→                    <TableRow key={test.id}>
   333→                      <TableCell className="text-muted-foreground text-xs">{idx + 1}</TableCell>
   334→                      <TableCell>
   335→                        <div>
   336→                          <p className="font-medium">{test.name}</p>
   337→                          <p className="text-xs text-muted-foreground sm:hidden">{test.shortCode}</p>
   338→                        </div>
   339→                      </TableCell>
   340→                      <TableCell className="hidden sm:table-cell">
   341→                        <code className="rounded bg-muted px-1.5 py-0.5 text-xs">{test.shortCode}</code>
   342→                      </TableCell>
   343→                      <TableCell className="hidden md:table-cell">
   344→                        {test.category && (
   345→                          <Badge variant="secondary" className={getCategoryBadgeColor(test.category)}>
   346→                            {test.category}
   347→                          </Badge>
   348→                        )}
   349→                      </TableCell>
   350→                      <TableCell className="hidden lg:table-cell text-sm">{test.specimenType || '—'}</TableCell>
   351→                      <TableCell className="hidden md:table-cell text-right font-medium">
   352→                        {test.rate > 0 ? `₹${test.rate}` : '—'}
   353→                      </TableCell>
   354→                      <TableCell className="text-center">
   355→                        <Badge variant="outline" className="text-xs">
   356→                          <ListOrdered className="mr-1 h-3 w-3" />
   357→                          {test._count.parameters}
   358→                        </Badge>
   359→                      </TableCell>
   360→                      <TableCell className="text-right">
   361→                        <div className="flex items-center justify-end gap-1">
   362→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openEdit(test.id)}>
   363→                            <Pencil className="h-4 w-4" />
   364→                          </Button>
   365→                          <Button
   366→                            variant="ghost"
   367→                            size="icon"
   368→                            className="h-8 w-8 text-red-500 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-950/50"
   369→                            onClick={() => { setDeletingId(test.id); setDeleteDialogOpen(true) }}
   370→                          >
   371→                            <Trash2 className="h-4 w-4" />
   372→                          </Button>
   373→                        </div>
   374→                      </TableCell>
   375→                    </TableRow>
   376→                  ))}
   377→                </TableBody>
   378→              </Table>
   379→            </div>
   380→          )}
   381→        </CardContent>
   382→      </Card>
   383→
   384→      {/* Add/Edit Dialog */}
   385→      <Dialog open={dialogOpen} onOpenChange={(open) => { if (!open) closeDialog() }}>
   386→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   387→          <DialogHeader>
   388→            <DialogTitle>{editingId ? 'Edit Lab Test' : 'Add Lab Test'}</DialogTitle>
   389→            <DialogDescription>
   390→              {editingId ? 'Update test details and parameters' : 'Create a new lab test with its parameters'}
   391→            </DialogDescription>
   392→          </DialogHeader>
   393→
   394→          {detailLoading && editingId ? (
   395→            <div className="space-y-3 py-4">
   396→              {Array.from({ length: 6 }).map((_, i) => (
   397→                <Skeleton key={i} className="h-10 w-full" />
   398→              ))}
   399→            </div>
   400→          ) : (
   401→            <div className="space-y-4 py-2">
   402→              {/* Basic Info Grid */}
   403→              <div className="grid gap-4 sm:grid-cols-2">
   404→                <div className="space-y-2">
   405→                  <Label htmlFor="name">Test Name *</Label>
   406→                  <Input id="name" value={formName} onChange={(e) => setFormName(e.target.value)} placeholder="e.g. Complete Blood Count" />
   407→                </div>
   408→                <div className="space-y-2">
   409→                  <Label htmlFor="shortCode">Short Code</Label>
   410→                  <Input id="shortCode" value={formShortCode} onChange={(e) => setFormShortCode(e.target.value)} placeholder="e.g. CBC" className="uppercase" />
   411→                </div>
   412→                <div className="space-y-2">
   413→                  <Label>Category</Label>
   414→                  <Select value={formCategory} onValueChange={setFormCategory}>
   415→                    <SelectTrigger><SelectValue placeholder="Select category" /></SelectTrigger>
   416→                    <SelectContent>
   417→                      {CATEGORIES.map((c) => (
   418→                        <SelectItem key={c} value={c}>{c}</SelectItem>
   419→                      ))}
   420→                    </SelectContent>
   421→                  </Select>
   422→                </div>
   423→                <div className="space-y-2">
   424→                  <Label>Specimen Type</Label>
   425→                  <Select value={formSpecimenType} onValueChange={setFormSpecimenType}>
   426→                    <SelectTrigger><SelectValue placeholder="Select specimen" /></SelectTrigger>
   427→                    <SelectContent>
   428→                      {SPECIMEN_TYPES.map((s) => (
   429→                        <SelectItem key={s} value={s}>{s}</SelectItem>
   430→                      ))}
   431→                    </SelectContent>
   432→                  </Select>
   433→                </div>
   434→                <div className="space-y-2">
   435→                  <Label htmlFor="reportDays">Report Days</Label>
   436→                  <Input id="reportDays" type="number" min={0} value={formReportDays} onChange={(e) => setFormReportDays(Number(e.target.value))} />
   437→                </div>
   438→                <div className="space-y-2">
   439→                  <Label htmlFor="rate">Rate (₹)</Label>
   440→                  <Input id="rate" type="number" min={0} step={0.01} value={formRate} onChange={(e) => setFormRate(Number(e.target.value))} />
   441→                </div>
   442→              </div>
   443→              <div className="space-y-2">
   444→                <Label htmlFor="description">Description</Label>
   445→                <Input id="description" value={formDescription} onChange={(e) => setFormDescription(e.target.value)} placeholder="Brief description of the test" />
   446→              </div>
   447→
   448→              {/* Parameters Section */}
   449→              <div className="space-y-3">
   450→                <div className="flex items-center justify-between">
   451→                  <Label className="text-base font-semibold">Parameters</Label>
   452→                  <Button type="button" variant="outline" size="sm" onClick={addParameter}>
   453→                    <Plus className="mr-1 h-3 w-3" /> Add Parameter
   454→                  </Button>
   455→                </div>
   456→
   457→                <div className="max-h-[300px] space-y-3 overflow-y-auto rounded-lg border p-3">
   458→                  <AnimatePresence>
   459→                    {parameters.map((param, idx) => (
   460→                      <motion.div
   461→                        key={idx}
   462→                        initial={{ opacity: 0, height: 0 }}
   463→                        animate={{ opacity: 1, height: 'auto' }}
   464→                        exit={{ opacity: 0, height: 0 }}
   465→                        className="rounded-lg border bg-muted/30 p-3"
   466→                      >
   467→                        <div className="mb-2 flex items-center justify-between">
   468→                          <div className="flex items-center gap-2">
   469→                            <GripVertical className="h-4 w-4 text-muted-foreground" />
   470→                            <span className="text-sm font-medium">Parameter {idx + 1}</span>
   471→                          </div>
   472→                          {parameters.length > 1 && (
   473→                            <Button type="button" variant="ghost" size="icon" className="h-7 w-7" onClick={() => removeParameter(idx)}>
   474→                              <X className="h-4 w-4 text-red-500" />
   475→                            </Button>
   476→                          )}
   477→                        </div>
   478→                        <div className="grid gap-3 sm:grid-cols-3">
   479→                          <div className="space-y-1">
   480→                            <Label className="text-xs">Name *</Label>
   481→                            <Input
   482→                              value={param.paramName}
   483→                              onChange={(e) => updateParameter(idx, 'paramName', e.target.value)}
   484→                              placeholder="e.g. Haemoglobin"
   485→                              className="h-8 text-sm"
   486→                            />
   487→                          </div>
   488→                          <div className="space-y-1">
   489→                            <Label className="text-xs">Code</Label>
   490→                            <Input
   491→                              value={param.shortCode}
   492→                              onChange={(e) => updateParameter(idx, 'shortCode', e.target.value)}
   493→                              placeholder="e.g. Hb"
   494→                              className="h-8 text-sm uppercase"
   495→                            />
   496→                          </div>
   497→                          <div className="space-y-1">
   498→                            <Label className="text-xs">Unit</Label>
   499→                            <Input
   500→                              value={param.unit}
   501→                              onChange={(e) => updateParameter(idx, 'unit', e.target.value)}
   502→                              placeholder="e.g. g/dL"
   503→                              className="h-8 text-sm"
   504→                            />
   505→                          </div>
   506→                        </div>
   507→                        <div className="mt-2 grid gap-3 sm:grid-cols-3">
   508→                          <div className="rounded border p-2">
   509→                            <p className="mb-1 text-xs font-medium text-teal-600 dark:text-teal-400">Male Range</p>
   510→                            <div className="flex gap-2">
   511→                              <Input
   512→                                type="number" step="any"
   513→                                value={param.normalMaleMin || ''}
   514→                                onChange={(e) => updateParameter(idx, 'normalMaleMin', Number(e.target.value))}
   515→                                placeholder="Min" className="h-7 text-xs"
   516→                              />
   517→                              <Input
