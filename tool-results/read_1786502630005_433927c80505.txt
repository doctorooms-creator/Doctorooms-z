    87→    87→
    88→    88→export default function ComplaintsPage() {
    89→    89→  const queryClient = useQueryClient()
    90→    90→  const [search, setSearch] = useState('')
    91→    91→  const [statusFilter, setStatusFilter] = useState<'Active' | 'Inactive' | 'All'>('Active')
    92→    92→  const [categoryFilter, setCategoryFilter] = useState('')
    93→    93→  const [dialogOpen, setDialogOpen] = useState(false)
    94→    94→  const [editingItem, setEditingItem] = useState<Complaint | null>(null)
    95→    95→  const [form, setForm] = useState<FormData>(emptyForm)
    96→    96→  const [deleteTarget, setDeleteTarget] = useState<Complaint | null>(null)
    97→    97→
    98→    98→  // Fetch categories for dropdown (active only, no pagination needed)
    99→    99→  const { data: catData } = useQuery<{ categories: CategoryOption[] }>({
   100→   100→    queryKey: ['doctor-categories-dropdown'],
   101→   101→    queryFn: () =>
   102→   102→      fetch('/api/dashboard/doctor/prescription-settings/categories?status=Active').then((r) => r.json()),
   103→   103→    staleTime: 30_000,
   104→   104→  })
   105→   105→  const categoryOptions = catData?.categories || []
   106→   106→
   107→   107→  // Build query string for complaints
   108→   108→  const queryString = new URLSearchParams()
   109→   109→  if (search.trim()) queryString.set('search', search.trim())
   110→   110→  if (statusFilter !== 'All') queryString.set('status', statusFilter)
   111→   111→  if (categoryFilter) queryString.set('categoryId', categoryFilter)
   112→   112→
   113→   113→  const { data, isLoading } = useQuery<{ complaints: Complaint[] }>({
   114→   114→    queryKey: ['doctor-complaints', search, statusFilter, categoryFilter],
   115→   115→    queryFn: () =>
   116→   116→      fetch(`/api/dashboard/doctor/prescription-settings/complaints?${queryString.toString()}`).then((r) => r.json()),
   117→   117→  })
   118→   118→
   119→   119→  const complaints = data?.complaints || []
   120→   120→
   121→   121→  // Create mutation
   122→   122→  const createMutation = useMutation({
   123→   123→    mutationFn: (body: FormData) =>
   124→   124→      fetch('/api/dashboard/doctor/prescription-settings/complaints', {
   125→   125→        method: 'POST',
   126→   126→        headers: { 'Content-Type': 'application/json' },
   127→   127→        body: JSON.stringify(body),
   128→   128→      }).then((r) => {
   129→   129→        if (!r.ok) return r.json().then((e) => { throw new Error(e.error || 'Failed to create') })
   130→   130→        return r.json()
   131→   131→      }),
   132→   132→    onMutate: async (newItem) => {
   133→   133→      await queryClient.cancelQueries({ queryKey: ['doctor-complaints'] })
   134→   134→      const prev = queryClient.getQueryData<{ complaints: Complaint[] }>(['doctor-complaints', search, statusFilter, categoryFilter])
   135→   135→      const cat = categoryOptions.find((c) => c.id === newItem.categoryId) || null
   136→   136→      queryClient.setQueryData<{ complaints: Complaint[] }>(['doctor-complaints', search, statusFilter, categoryFilter], (old) => ({
   137→   137→        complaints: [
   138→   138→          {
   139→   139→            id: 'optimistic-' + Date.now(),
   140→   140→            coCode: newItem.coCode,
   141→   141→            coDetail: newItem.coDetail,
   142→   142→            coDetailEn: newItem.coDetailEn,
   143→   143→            categoryId: newItem.categoryId || null,
   144→   144→            status: 'Active',
   145→   145→            createdAt: new Date().toISOString(),
   146→   146→            updatedAt: new Date().toISOString(),
   147→   147→            category: cat,
   148→   148→          },
   149→   149→          ...(old?.complaints || []),
   150→   150→        ],
   151→   151→      }))
   152→   152→      return { prev }
   153→   153→    },
   154→   154→    onSuccess: () => {
   155→   155→      queryClient.invalidateQueries({ queryKey: ['doctor-complaints'] })
   156→   156→      toast.success('Complaint added successfully')
   157→   157→      closeDialog()
   158→   158→    },
   159→   159→    onError: (err, _vars, context) => {
   160→   160→      if (context?.prev) {
   161→   161→        queryClient.setQueryData(['doctor-complaints', search, statusFilter, categoryFilter], context.prev)
   162→   162→      }
   163→   163→      toast.error(err.message || 'Failed to add complaint')
   164→   164→    },
   165→   165→  })
   166→   166→
   167→   167→  // Update mutation
   168→   168→  const updateMutation = useMutation({
   169→   169→    mutationFn: ({ id, body }: { id: string; body: FormData }) =>
   170→   170→      fetch(`/api/dashboard/doctor/prescription-settings/complaints/${id}`, {
   171→   171→        method: 'PUT',
   172→   172→        headers: { 'Content-Type': 'application/json' },
   173→   173→        body: JSON.stringify(body),
   174→   174→      }).then((r) => {
   175→   175→        if (!r.ok) return r.json().then((e) => { throw new Error(e.error || 'Failed to update') })
   176→   176→        return r.json()
   177→   177→      }),
   178→   178→    onMutate: async ({ id, body }) => {
   179→   179→      await queryClient.cancelQueries({ queryKey: ['doctor-complaints'] })
   180→   180→      const prev = queryClient.getQueryData<{ complaints: Complaint[] }>(['doctor-complaints', search, statusFilter, categoryFilter])
   181→   181→      queryClient.setQueryData<{ complaints: Complaint[] }>(['doctor-complaints', search, statusFilter, categoryFilter], (old) => {
   182→   182→        const cat = categoryOptions.find((c) => c.id === body.categoryId) || null
   183→   183→        return {
   184→   184→          complaints: (old?.complaints || []).map((c) =>
   185→   185→            c.id === id
   186→   186→              ? {
   187→   187→                  ...c,
   188→   188→                  coCode: body.coCode ?? c.coCode,
   189→   189→                  coDetail: body.coDetail ?? c.coDetail,
   190→   190→                  coDetailEn: body.coDetailEn ?? c.coDetailEn,
   191→   191→                  categoryId: body.categoryId || null,
   192→   192→                  status: body.status ?? c.status,
   193→   193→                  category: cat,
   194→   194→                  updatedAt: new Date().toISOString(),
   195→   195→                }
   196→   196→              : c
   197→   197→          ),
   198→   198→        }
   199→   199→      })
   200→   200→      return { prev }
   201→   201→    },
   202→   202→    onSuccess: () => {
   203→   203→      queryClient.invalidateQueries({ queryKey: ['doctor-complaints'] })
   204→   204→      toast.success('Complaint updated successfully')
   205→   205→      closeDialog()
   206→   206→    },
   207→   207→    onError: (err, _vars, context) => {
   208→   208→      if (context?.prev) {
   209→   209→        queryClient.setQueryData(['doctor-complaints', search, statusFilter, categoryFilter], context.prev)
   210→   210→      }
   211→   211→      toast.error(err.message || 'Failed to update complaint')
   212→   212→    },
   213→   213→  })
   214→   214→
   215→   215→  // Delete mutation (soft delete)
   216→   216→  const deleteMutation = useMutation({
   217→   217→    mutationFn: (id: string) =>
   218→   218→      fetch(`/api/dashboard/doctor/prescription-settings/complaints/${id}`, {
   219→   219→        method: 'DELETE',
   220→   220→      }).then((r) => {
   221→   221→        if (!r.ok) return r.json().then((e) => { throw new Error(e.error || 'Failed to delete') })
   222→   222→        return r.json()
   223→   223→      }),
   224→   224→    onMutate: async (id) => {
   225→   225→      await queryClient.cancelQueries({ queryKey: ['doctor-complaints'] })
   226→   226→      const prev = queryClient.getQueryData<{ complaints: Complaint[] }>(['doctor-complaints', search, statusFilter, categoryFilter])
   227→   227→      queryClient.setQueryData<{ complaints: Complaint[] }>(['doctor-complaints', search, statusFilter, categoryFilter], (old) => ({
   228→   228→        complaints: (old?.complaints || []).filter((c) => c.id !== id),
   229→   229→      }))
   230→   230→      return { prev }
   231→   231→    },
   232→   232→    onSuccess: () => {
   233→   233→      queryClient.invalidateQueries({ queryKey: ['doctor-complaints'] })
   234→   234→      toast.success('Complaint deactivated')
   235→   235→      setDeleteTarget(null)
   236→   236→    },
   237→   237→    onError: (err, _vars, context) => {
   238→   238→      if (context?.prev) {
   239→   239→        queryClient.setQueryData(['doctor-complaints', search, statusFilter, categoryFilter], context.prev)
   240→   240→      }
   241→   241→      toast.error(err.message || 'Failed to delete complaint')
   242→   242→    },
   243→   243→  })
   244→   244→
   245→   245→  const openCreateDialog = () => {
   246→   246→    setEditingItem(null)
   247→   247→    setForm(emptyForm)
   248→   248→    setDialogOpen(true)
   249→   249→  }
   250→   250→
   251→   251→  const openEditDialog = (comp: Complaint) => {
   252→   252→    setEditingItem(comp)
   253→   253→    setForm({
   254→   254→      coCode: comp.coCode || '',
   255→   255→      coDetail: comp.coDetail,
   256→   256→      coDetailEn: comp.coDetailEn || '',
   257→   257→      categoryId: comp.categoryId || '',
   258→   258→      status: comp.status,
   259→   259→    })
   260→   260→    setDialogOpen(true)
   261→   261→  }
   262→   262→
   263→   263→  const closeDialog = () => {
   264→   264→    setDialogOpen(false)
   265→   265→    setEditingItem(null)
   266→   266→    setForm(emptyForm)
   267→   267→  }
   268→   268→
   269→   269→  const handleSubmit = () => {
   270→   270→    if (!form.coDetail.trim()) {
   271→   271→      toast.error('Complaint detail is required')
   272→   272→      return
   273→   273→    }
   274→   274→    if (editingItem) {
   275→   275→      updateMutation.mutate({ id: editingItem.id, body: form })
   276→   276→    } else {
   277→   277→      createMutation.mutate(form)
   278→   278→    }
   279→   279→  }
   280→   280→
   281→   281→  const handleDelete = () => {
   282→   282→    if (deleteTarget) {
   283→   283→      deleteMutation.mutate(deleteTarget.id)
   284→   284→    }
   285→   285→  }
   286→   286→
   287→   287→  const getCategoryName = (comp: Complaint) => {
   288→   288→    if (comp.category) return comp.category.name
   289→   289→    return null
   290→   290→  }
   291→   291→
   292→   292→  return (
   293→   293→    <div className="space-y-6">
   294→   294→      {/* Header */}
   295→   295→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   296→   296→        <div>
   297→   297→          <h2 className="text-lg font-semibold flex items-center gap-2">
   298→   298→            <Thermometer className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   299→   299→            Complaints (C/O)
   300→   300→          </h2>
   301→   301→          <p className="text-sm text-muted-foreground mt-1">
   302→   302→            Manage chief complaints for your prescriptions
   303→   303→          </p>
   304→   304→        </div>
   305→   305→        <Button
   306→   306→          className="bg-teal-600 hover:bg-teal-700"
   307→   307→          onClick={openCreateDialog}
   308→   308→        >
   309→   309→          <Plus className="mr-2 h-4 w-4" />
   310→   310→          Add Complaint
   311→   311→        </Button>
   312→   312→      </div>
   313→   313→
   314→   314→      {/* Search & Filter Bar */}
   315→   315→      <div className="flex flex-col sm:flex-row gap-3">
   316→   316→        <div className="relative flex-1">
   317→   317→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   318→   318→          <Input
   319→   319→            placeholder="Search complaints..."
   320→   320→            value={search}
   321→   321→            onChange={(e) => setSearch(e.target.value)}
   322→   322→            className="pl-9"
   323→   323→          />
   324→   324→        </div>
   325→   325→        <Select value={categoryFilter} onValueChange={(v) => setCategoryFilter(v === '__all__' ? '' : v)}>
   326→   326→          <SelectTrigger className="w-full sm:w-[180px]">
   327→   327→            <SelectValue placeholder="All Categories" />
   328→   328→          </SelectTrigger>
   329→   329→          <SelectContent>
   330→   330→            <SelectItem value="__all__">All Categories</SelectItem>
   331→   331→            {categoryOptions.map((cat) => (
   332→   332→              <SelectItem key={cat.id} value={cat.id}>
   333→   333→                {cat.name}{cat.nameEn ? ` (${cat.nameEn})` : ''}
   334→   334→              </SelectItem>
   335→   335→            ))}
   336→   336→          </SelectContent>
   337→   337→        </Select>
   338→   338→        <div className="flex gap-1.5 rounded-lg border border-border bg-card p-1">
   339→   339→          {(['Active', 'Inactive', 'All'] as const).map((s) => (
   340→   340→            <button
   341→   341→              key={s}
   342→   342→              onClick={() => setStatusFilter(s)}
   343→   343→              className={cn(
   344→   344→                'rounded-md px-3 py-1.5 text-xs font-medium transition-colors',
   345→   345→                statusFilter === s
   346→   346→                  ? 'bg-teal-600 text-white shadow-sm'
   347→   347→                  : 'text-muted-foreground hover:text-foreground hover:bg-muted'
   348→   348→              )}
   349→   349→            >
   350→   350→              {s}
   351→   351→            </button>
   352→   352→          ))}
   353→   353→        </div>
   354→   354→      </div>
   355→   355→
   356→   356→      {/* Table */}
   357→   357→      {isLoading ? (
   358→   358→        <div className="rounded-xl border border-border overflow-hidden">
   359→   359→          <div className="space-y-0">
   360→   360→            {Array.from({ length: 5 }).map((_, i) => (
   361→   361→              <div key={i} className="flex items-center gap-4 px-4 py-3 border-b border-border last:border-b-0">
   362→   362→                <div className="h-4 w-12 animate-pulse rounded bg-muted" />
   363→   363→                <div className="h-4 w-40 animate-pulse rounded bg-muted" />
   364→   364→                <div className="h-4 w-32 animate-pulse rounded bg-muted" />
   365→   365→                <div className="h-4 w-20 animate-pulse rounded bg-muted" />
   366→   366→              </div>
   367→   367→            ))}
   368→   368→          </div>
   369→   369→        </div>
   370→   370→      ) : complaints.length === 0 ? (
   371→   371→        <div className="flex flex-col items-center justify-center rounded-xl border border-dashed border-border py-16">
   372→   372→          <div className="flex h-16 w-16 items-center justify-center rounded-full bg-teal-50 dark:bg-teal-900/30 mb-4">
   373→   373→            <Thermometer className="h-8 w-8 text-teal-600 dark:text-teal-400" />
   374→   374→          </div>
   375→   375→          <h3 className="text-sm font-medium">
   376→   376→            {search || categoryFilter ? 'No complaints found' : 'No complaints added yet'}
   377→   377→          </h3>
   378→   378→          <p className="text-sm text-muted-foreground mt-1 text-center max-w-xs">
   379→   379→            {search || categoryFilter
   380→   380→              ? 'Try a different search term or adjust the filters'
   381→   381→              : 'Add chief complaints that you commonly encounter for quick selection in prescriptions'}
   382→   382→          </p>
   383→   383→          {!search && !categoryFilter && (
   384→   384→            <Button
   385→   385→              variant="outline"
   386→   386→              className="mt-4 text-teal-600 border-teal-300 hover:bg-teal-50 dark:text-teal-400 dark:border-teal-700 dark:hover:bg-teal-900/30"
   387→   387→              onClick={openCreateDialog}
   388→   388→            >
   389→   389→              <Plus className="mr-2 h-4 w-4" />
   390→   390→              Add Your First Complaint
   391→   391→            </Button>
   392→   392→          )}
   393→   393→        </div>
   394→   394→      ) : (
   395→   395→        <div className="rounded-xl border border-border overflow-hidden">
   396→   396→          <div className="overflow-x-auto">
   397→   397→            <Table>
   398→   398→              <TableHeader>
   399→   399→                <TableRow className="bg-muted/40 hover:bg-muted/40">
   400→   400→                  <TableHead className="w-[70px] text-xs">Code</TableHead>
   401→   401→                  <TableHead className="text-xs">Detail</TableHead>
   402→   402→                  <TableHead className="text-xs hidden md:table-cell">English</TableHead>
   403→   403→                  <TableHead className="text-xs hidden sm:table-cell">Category</TableHead>
   404→   404→                  <TableHead className="w-[80px] text-xs">Status</TableHead>
   405→   405→                  <TableHead className="w-[80px] text-xs text-right">Actions</TableHead>
   406→   406→                </TableRow>
   407→   407→              </TableHeader>
   408→   408→              <TableBody>
   409→   409→                <AnimatePresence mode="popLayout">
   410→   410→                  {complaints.map((comp, i) => (
   411→   411→                    <motion.tr
   412→   412→                      key={comp.id}
   413→   413→                      layout
   414→   414→                      initial={{ opacity: 0, y: 10 }}
   415→   415→                      animate={{ opacity: 1, y: 0 }}
   416→   416→                      exit={{ opacity: 0, x: -20 }}
   417→   417→                      transition={{ delay: i * 0.02 }}
   418→   418→                      className="group border-b border-border last:border-b-0 hover:bg-muted/30 transition-colors"
   419→   419→                    >
   420→   420→                      <TableCell className="py-3">
   421→   421→                        {comp.coCode ? (
   422→   422→                          <span className="inline-flex items-center gap-1 text-xs font-mono text-muted-foreground">
   423→   423→                            <Hash className="h-3 w-3" />
   424→   424→                            {comp.coCode}
   425→   425→                          </span>
   426→   426→                        ) : (
   427→   427→                          <span className="text-xs text-muted-foreground/40">—</span>
   428→   428→                        )}
   429→   429→                      </TableCell>
   430→   430→                      <TableCell className="py-3">
   431→   431→                        <span className="text-sm font-medium">{comp.coDetail}</span>
   432→   432→                      </TableCell>
   433→   433→                      <TableCell className="py-3 hidden md:table-cell">
   434→   434→                        {comp.coDetailEn ? (
   435→   435→                          <span className="flex items-center gap-1 text-xs text-muted-foreground">
   436→   436→                            <Languages className="h-3 w-3 shrink-0" />
   437→   437→                            {comp.coDetailEn}
   438→   438→                          </span>
   439→   439→                        ) : (
   440→   440→                          <span className="text-xs text-muted-foreground/40">—</span>
   441→   441→                        )}
   442→   442→                      </TableCell>
   443→   443→                      <TableCell className="py-3 hidden sm:table-cell">
   444→   444→                        {getCategoryName(comp) ? (
   445→   445→                          <Badge
   446→   446→                            variant="outline"
   447→   447→                            className="text-[10px] px-2 py-0 border-teal-200 text-teal-700 bg-teal-50 dark:border-teal-800 dark:text-teal-300 dark:bg-teal-900/20"
   448→   448→                          >
   449→   449→                            <Tag className="h-2.5 w-2.5 mr-1" />
   450→   450→                            {getCategoryName(comp)}
   451→   451→                          </Badge>
   452→   452→                        ) : (
   453→   453→                          <span className="text-xs text-muted-foreground/40">—</span>
   454→   454→                        )}
   455→   455→                      </TableCell>
   456→   456→                      <TableCell className="py-3">
