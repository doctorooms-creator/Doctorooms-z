    95→    95→
    96→    96→export default function LabTestMasterClient() {
    97→    97→  const queryClient = useQueryClient()
    98→    98→  const [search, setSearch] = useState('')
    99→    99→  const [categoryFilter, setCategoryFilter] = useState('')
   100→   100→  const [dialogOpen, setDialogOpen] = useState(false)
   101→   101→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   102→   102→  const [editingId, setEditingId] = useState<string | null>(null)
   103→   103→  const [deletingId, setDeletingId] = useState<string | null>(null)
   104→   104→
   105→   105→  // Form state
   106→   106→  const [formName, setFormName] = useState('')
   107→   107→  const [formShortCode, setFormShortCode] = useState('')
   108→   108→  const [formCategory, setFormCategory] = useState('')
   109→   109→  const [formDescription, setFormDescription] = useState('')
   110→   110→  const [formSpecimenType, setFormSpecimenType] = useState('')
   111→   111→  const [formReportDays, setFormReportDays] = useState(0)
   112→   112→  const [formRate, setFormRate] = useState(0)
   113→   113→  const [parameters, setParameters] = useState<LabTestParameter[]>([emptyParameter()])
   114→   114→
   115→   115→  // Fetch test masters
   116→   116→  const { data, isLoading } = useQuery<{ testMasters: LabTestMaster[] }>({
   117→   117→    queryKey: ['lab-test-masters', categoryFilter, search],
   118→   118→    queryFn: () => {
   119→   119→      const params = new URLSearchParams()
   120→   120→      if (categoryFilter) params.set('category', categoryFilter)
   121→   121→      params.set('status', 'Active')
   122→   122→      if (search) params.set('search', search)
   123→   123→      return fetch(`/api/lab-test-masters?${params}`).then((r) => r.json())
   124→   124→    },
   125→   125→  })
   126→   126→
   127→   127→  const testMasters = data?.testMasters || []
   128→   128→
   129→   129→  // Create / Update mutation
   130→   130→  const saveMutation = useMutation({
   131→   131→    mutationFn: async (body: Record<string, unknown>) => {
   132→   132→      if (editingId) {
   133→   133→        const res = await fetch(`/api/lab-test-masters/${editingId}`, { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   134→   134→        if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Update failed') }
   135→   135→        return res.json()
   136→   136→      } else {
   137→   137→        const res = await fetch('/api/lab-test-masters', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) })
   138→   138→        if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Create failed') }
   139→   139→        return res.json()
   140→   140→      }
   141→   141→    },
   142→   142→    onSuccess: () => {
   143→   143→      queryClient.invalidateQueries({ queryKey: ['lab-test-masters'] })
   144→   144→      toast.success(editingId ? 'Test updated successfully' : 'Test created successfully')
   145→   145→      closeDialog()
   146→   146→    },
   147→   147→    onError: (err: Error) => {
   148→   148→      toast.error(err.message)
   149→   149→    },
   150→   150→  })
   151→   151→
   152→   152→  // Delete mutation
   153→   153→  const deleteMutation = useMutation({
   154→   154→    mutationFn: async (id: string) => {
   155→   155→      const res = await fetch(`/api/lab-test-masters/${id}`, { method: 'DELETE' })
   156→   156→      if (!res.ok) { const err = await res.json(); throw new Error(err.error || 'Delete failed') }
   157→   157→      return res.json()
   158→   158→    },
   159→   159→    onSuccess: () => {
   160→   160→      queryClient.invalidateQueries({ queryKey: ['lab-test-masters'] })
   161→   161→      toast.success('Test deleted successfully')
   162→   162→      setDeleteDialogOpen(false)
   163→   163→      setDeletingId(null)
   164→   164→    },
   165→   165→    onError: (err: Error) => {
   166→   166→      toast.error(err.message)
   167→   167→    },
   168→   168→  })
   169→   169→
   170→   170→  // Fetch detail when editing
   171→   171→  const { data: detailData, isLoading: detailLoading } = useQuery<{ testMaster: LabTestMaster & { parameters: LabTestParameter[] } }>({
   172→   172→    queryKey: ['lab-test-master-detail', editingId],
   173→   173→    queryFn: () => fetch(`/api/lab-test-masters/${editingId}`).then((r) => r.json()),
   174→   174→    enabled: !!editingId,
   175→   175→  })
   176→   176→
   177→   177→  // Populate form on edit
   178→   178→  useEffect(() => {
   179→   179→    if (detailData?.testMaster && editingId) {
   180→   180→      const t = detailData.testMaster
   181→   181→      setFormName(t.name)
   182→   182→      setFormShortCode(t.shortCode)
   183→   183→      setFormCategory(t.category)
   184→   184→      setFormDescription(t.description)
   185→   185→      setFormSpecimenType(t.specimenType)
   186→   186→      setFormReportDays(t.reportDays)
   187→   187→      setFormRate(t.rate)
   188→   188→      setParameters(t.parameters.length > 0 ? t.parameters : [emptyParameter()])
   189→   189→    }
   190→   190→  }, [detailData, editingId])
   191→   191→
   192→   192→  function openCreate() {
   193→   193→    setEditingId(null)
   194→   194→    setFormName('')
   195→   195→    setFormShortCode('')
   196→   196→    setFormCategory('')
   197→   197→    setFormDescription('')
   198→   198→    setFormSpecimenType('')
   199→   199→    setFormReportDays(0)
   200→   200→    setFormRate(0)
   201→   201→    setParameters([emptyParameter()])
   202→   202→    setDialogOpen(true)
   203→   203→  }
   204→   204→
   205→   205→  function openEdit(id: string) {
   206→   206→    setEditingId(id)
   207→   207→    setDialogOpen(true)
   208→   208→  }
   209→   209→
   210→   210→  function closeDialog() {
   211→   211→    setDialogOpen(false)
   212→   212→    setEditingId(null)
   213→   213→  }
   214→   214→
   215→   215→  function handleSave() {
   216→   216→    if (!formName.trim()) { toast.error('Test name is required'); return }
   217→   217→
   218→   218→    const validParams = parameters.filter((p) => p.paramName.trim() !== '')
   219→   219→    saveMutation.mutate({
   220→   220→      name: formName,
   221→   221→      shortCode: formShortCode,
   222→   222→      category: formCategory,
   223→   223→      description: formDescription,
   224→   224→      specimenType: formSpecimenType,
   225→   225→      reportDays: formReportDays,
   226→   226→      rate: formRate,
   227→   227→      parameters: validParams.map((p, idx) => ({ ...p, sortOrder: idx })),
   228→   228→    })
   229→   229→  }
   230→   230→
   231→   231→  function addParameter() {
   232→   232→    setParameters([...parameters, { ...emptyParameter(), sortOrder: parameters.length }])
   233→   233→  }
   234→   234→
   235→   235→  function removeParameter(index: number) {
   236→   236→    if (parameters.length <= 1) return
   237→   237→    setParameters(parameters.filter((_, i) => i !== index))
   238→   238→  }
   239→   239→
   240→   240→  function updateParameter(index: number, field: keyof LabTestParameter, value: string | number) {
   241→   241→    const updated = [...parameters]
   242→   242→    updated[index] = { ...updated[index], [field]: value }
   243→   243→    setParameters(updated)
   244→   244→  }
   245→   245→
   246→   246→  function getCategoryBadgeColor(cat: string) {
   247→   247→    const colors: Record<string, string> = {
   248→   248→      Haematology: 'bg-red-100 text-red-700 dark:bg-red-950/50 dark:text-red-400',
   249→   249→      Biochemistry: 'bg-emerald-100 text-emerald-700 dark:bg-emerald-950/50 dark:text-emerald-400',
   250→   250→      Microbiology: 'bg-amber-100 text-amber-700 dark:bg-amber-950/50 dark:text-amber-400',
   251→   251→      Serology: 'bg-violet-100 text-violet-700 dark:bg-violet-950/50 dark:text-violet-400',
   252→   252→      Urine: 'bg-yellow-100 text-yellow-700 dark:bg-yellow-950/50 dark:text-yellow-400',
   253→   253→      Thyroid: 'bg-teal-100 text-teal-700 dark:bg-teal-950/50 dark:text-teal-400',
   254→   254→      Hormones: 'bg-pink-100 text-pink-700 dark:bg-pink-950/50 dark:text-pink-400',
   255→   255→    }
   256→   256→    return colors[cat] || 'bg-gray-100 text-gray-700 dark:bg-gray-800 dark:text-gray-300'
   257→   257→  }
   258→   258→
   259→   259→  return (
   260→   260→    <div className="space-y-6">
   261→   261→      {/* Header */}
   262→   262→      <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   263→   263→        <div className="space-y-1">
   264→   264→          <h1 className="text-2xl font-bold tracking-tight">Lab Test Master</h1>
   265→   265→          <p className="text-sm text-muted-foreground">Manage lab tests and their parameters</p>
   266→   266→        </div>
   267→   267→        <Button onClick={openCreate} className="bg-teal-600 hover:bg-teal-700">
   268→   268→          <Plus className="mr-2 h-4 w-4" /> Add Test
   269→   269→        </Button>
   270→   270→      </motion.div>
   271→   271→
   272→   272→      {/* Filters */}
   273→   273→      <Card>
   274→   274→        <CardContent className="p-4">
   275→   275→          <div className="flex flex-col gap-3 sm:flex-row">
   276→   276→            <div className="relative flex-1">
   277→   277→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   278→   278→              <Input
   279→   279→                placeholder="Search tests..."
   280→   280→                value={search}
   281→   281→                onChange={(e) => setSearch(e.target.value)}
   282→   282→                className="pl-9"
   283→   283→              />
   284→   284→            </div>
   285→   285→            <Select value={categoryFilter} onValueChange={(v) => setCategoryFilter(v === '_all' ? '' : v)}>
   286→   286→              <SelectTrigger className="w-full sm:w-48">
   287→   287→                <SelectValue placeholder="All Categories" />
   288→   288→              </SelectTrigger>
   289→   289→              <SelectContent>
   290→   290→                <SelectItem value="_all">All Categories</SelectItem>
   291→   291→                {CATEGORIES.map((c) => (
   292→   292→                  <SelectItem key={c} value={c}>{c}</SelectItem>
   293→   293→                ))}
   294→   294→              </SelectContent>
   295→   295→            </Select>
   296→   296→          </div>
   297→   297→        </CardContent>
   298→   298→      </Card>
   299→   299→
   300→   300→      {/* Table */}
   301→   301→      <Card>
   302→   302→        <CardContent className="p-0">
   303→   303→          {isLoading ? (
   304→   304→            <div className="space-y-3 p-4">
   305→   305→              {Array.from({ length: 5 }).map((_, i) => (
   306→   306→                <Skeleton key={i} className="h-12 w-full" />
   307→   307→              ))}
   308→   308→            </div>
   309→   309→          ) : testMasters.length === 0 ? (
   310→   310→            <div className="flex flex-col items-center justify-center py-12">
   311→   311→              <FlaskConical className="mb-3 h-12 w-12 text-muted-foreground/40" />
   312→   312→              <p className="text-sm text-muted-foreground">No lab tests found</p>
   313→   313→              <p className="text-xs text-muted-foreground">Click &quot;Add Test&quot; to create one</p>
   314→   314→            </div>
   315→   315→          ) : (
   316→   316→            <div className="max-h-[480px] overflow-y-auto">
   317→   317→              <Table>
   318→   318→                <TableHeader>
   319→   319→                  <TableRow>
   320→   320→                    <TableHead className="w-12">#</TableHead>
   321→   321→                    <TableHead>Name</TableHead>
   322→   322→                    <TableHead className="hidden sm:table-cell">Code</TableHead>
   323→   323→                    <TableHead className="hidden md:table-cell">Category</TableHead>
   324→   324→                    <TableHead className="hidden lg:table-cell">Specimen</TableHead>
   325→   325→                    <TableHead className="hidden md:table-cell text-right">Rate</TableHead>
   326→   326→                    <TableHead className="text-center">Params</TableHead>
   327→   327→                    <TableHead className="text-right">Actions</TableHead>
   328→   328→                  </TableRow>
   329→   329→                </TableHeader>
   330→   330→                <TableBody>
   331→   331→                  {testMasters.map((test, idx) => (
   332→   332→                    <TableRow key={test.id}>
   333→   333→                      <TableCell className="text-muted-foreground text-xs">{idx + 1}</TableCell>
   334→   334→                      <TableCell>
   335→   335→                        <div>
   336→   336→                          <p className="font-medium">{test.name}</p>
   337→   337→                          <p className="text-xs text-muted-foreground sm:hidden">{test.shortCode}</p>
   338→   338→                        </div>
   339→   339→                      </TableCell>
   340→   340→                      <TableCell className="hidden sm:table-cell">
   341→   341→                        <code className="rounded bg-muted px-1.5 py-0.5 text-xs">{test.shortCode}</code>
   342→   342→                      </TableCell>
   343→   343→                      <TableCell className="hidden md:table-cell">
   344→   344→                        {test.category && (
   345→   345→                          <Badge variant="secondary" className={getCategoryBadgeColor(test.category)}>
   346→   346→                            {test.category}
   347→   347→                          </Badge>
   348→   348→                        )}
   349→   349→                      </TableCell>
   350→   350→                      <TableCell className="hidden lg:table-cell text-sm">{test.specimenType || '—'}</TableCell>
   351→   351→                      <TableCell className="hidden md:table-cell text-right font-medium">
   352→   352→                        {test.rate > 0 ? `₹${test.rate}` : '—'}
   353→   353→                      </TableCell>
   354→   354→                      <TableCell className="text-center">
   355→   355→                        <Badge variant="outline" className="text-xs">
   356→   356→                          <ListOrdered className="mr-1 h-3 w-3" />
   357→   357→                          {test._count.parameters}
   358→   358→                        </Badge>
   359→   359→                      </TableCell>
   360→   360→                      <TableCell className="text-right">
   361→   361→                        <div className="flex items-center justify-end gap-1">
   362→   362→                          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => openEdit(test.id)}>
   363→   363→                            <Pencil className="h-4 w-4" />
   364→   364→                          </Button>
   365→   365→                          <Button
   366→   366→                            variant="ghost"
   367→   367→                            size="icon"
   368→   368→                            className="h-8 w-8 text-red-500 hover:text-red-600 hover:bg-red-50 dark:hover:bg-red-950/50"
   369→   369→                            onClick={() => { setDeletingId(test.id); setDeleteDialogOpen(true) }}
   370→   370→                          >
   371→   371→                            <Trash2 className="h-4 w-4" />
   372→   372→                          </Button>
   373→   373→                        </div>
   374→   374→                      </TableCell>
   375→   375→                    </TableRow>
   376→   376→                  ))}
   377→   377→                </TableBody>
   378→   378→              </Table>
   379→   379→            </div>
   380→   380→          )}
   381→   381→        </CardContent>
   382→   382→      </Card>
   383→   383→
   384→   384→      {/* Add/Edit Dialog */}
   385→   385→      <Dialog open={dialogOpen} onOpenChange={(open) => { if (!open) closeDialog() }}>
   386→   386→        <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   387→   387→          <DialogHeader>
   388→   388→            <DialogTitle>{editingId ? 'Edit Lab Test' : 'Add Lab Test'}</DialogTitle>
   389→   389→            <DialogDescription>
   390→   390→              {editingId ? 'Update test details and parameters' : 'Create a new lab test with its parameters'}
   391→   391→            </DialogDescription>
   392→   392→          </DialogHeader>
   393→   393→
   394→   394→          {detailLoading && editingId ? (
   395→   395→            <div className="space-y-3 py-4">
   396→   396→              {Array.from({ length: 6 }).map((_, i) => (
   397→   397→                <Skeleton key={i} className="h-10 w-full" />
   398→   398→              ))}
   399→   399→            </div>
   400→   400→          ) : (
   401→   401→            <div className="space-y-4 py-2">
   402→   402→              {/* Basic Info Grid */}
   403→   403→              <div className="grid gap-4 sm:grid-cols-2">
   404→   404→                <div className="space-y-2">
   405→   405→                  <Label htmlFor="name">Test Name *</Label>
   406→   406→                  <Input id="name" value={formName} onChange={(e) => setFormName(e.target.value)} placeholder="e.g. Complete Blood Count" />
   407→   407→                </div>
   408→   408→                <div className="space-y-2">
   409→   409→                  <Label htmlFor="shortCode">Short Code</Label>
   410→   410→                  <Input id="shortCode" value={formShortCode} onChange={(e) => setFormShortCode(e.target.value)} placeholder="e.g. CBC" className="uppercase" />
   411→   411→                </div>
   412→   412→                <div className="space-y-2">
   413→   413→                  <Label>Category</Label>
   414→   414→                  <Select value={formCategory} onValueChange={setFormCategory}>
   415→   415→                    <SelectTrigger><SelectValue placeholder="Select category" /></SelectTrigger>
   416→   416→                    <SelectContent>
   417→   417→                      {CATEGORIES.map((c) => (
   418→   418→                        <SelectItem key={c} value={c}>{c}</SelectItem>
   419→   419→                      ))}
   420→   420→                    </SelectContent>
   421→   421→                  </Select>
   422→   422→                </div>
   423→   423→                <div className="space-y-2">
   424→   424→                  <Label>Specimen Type</Label>
   425→   425→                  <Select value={formSpecimenType} onValueChange={setFormSpecimenType}>
   426→   426→                    <SelectTrigger><SelectValue placeholder="Select specimen" /></SelectTrigger>
   427→   427→                    <SelectContent>
   428→   428→                      {SPECIMEN_TYPES.map((s) => (
   429→   429→                        <SelectItem key={s} value={s}>{s}</SelectItem>
   430→   430→                      ))}
   431→   431→                    </SelectContent>
   432→   432→                  </Select>
   433→   433→                </div>
   434→   434→                <div className="space-y-2">
   435→   435→                  <Label htmlFor="reportDays">Report Days</Label>
   436→   436→                  <Input id="reportDays" type="number" min={0} value={formReportDays} onChange={(e) => setFormReportDays(Number(e.target.value))} />
   437→   437→                </div>
   438→   438→                <div className="space-y-2">
   439→   439→                  <Label htmlFor="rate">Rate (₹)</Label>
