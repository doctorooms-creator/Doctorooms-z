   101→  const [disease, setDisease] = useState('')

   104→  const [bookingCity, setBookingCity] = useState('')

   108→  const [slotStatuses, setSlotStatuses] = useState<SlotAvailability[]>([])
   109→
   110→  // ── Fetch doctor info ──
   111→  const {
   112→    data: doctorData,
   113→    isLoading: doctorLoading,
   114→  } = useQuery({
   115→    queryKey: ['doctor-info', doctorId],
   116→    queryFn: () => fetch(`/api/doctors/${doctorId}`).then((r) => r.json()),

   124→    data: scheduleData,

   127→    queryKey: ['doctor-schedule', doctorId],
   128→    queryFn: () => fetch(`/api/doctors/${doctorId}/schedule`).then((r) => r.json()),
   129→    enabled: !!doctorId,

   132→  const fees = doctor?.doctor?.fees ?? scheduleData?.fees ?? 0
   133→

   135→  const scheduleForSelectedDay = useMemo(() => {
   136→    if (!selectedDate || !scheduleData?.schedules) return null
   137→    const dayName = format(selectedDate, 'EEEE')
   138→    return scheduleData.schedules.find((s) => s.day === dayName) || null
   139→  }, [selectedDate, scheduleData])
   140→
   141→  const timeSlots = useMemo(() => {
   142→    if (!scheduleForSelectedDay) return []
   143→    return scheduleForSelectedDay.timeSlots
   144→  }, [scheduleForSelectedDay])
   145→
   146→  // ── Check if selected date is a holiday ──
   147→  const isHoliday = useMemo(() => {
   148→    if (!selectedDate || !scheduleData?.holidays) return false
   149→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   150→    return scheduleData.holidays.some((h) => h.date === dateStr)

   153→  // When date changes, reset slot selection
   154→  const handleDateSelect = useCallback(
   155→    (date: Date | undefined) => {

   157→      setSelectedSlot(null)

   160→        setSlotStatuses([])

   162→    },
   163→    []

   166→  // Fetch slot availability when date or slots change (useEffect avoids stale closure)

   168→    if (!selectedDate || timeSlots.length === 0 || !scheduleData?.doctorId) {
   169→      return
   170→    }

   177→    fetch(`/api/patient/bookings/slots-availability?doctorId=${currentDoctorId}&date=${dateStr}`)
   178→      .then((r) => {
   179→        if (!r.ok) {
   180→          // Auth or server error — show all slots as available but log warning
   181→          console.warn('Slots availability API returned', r.status)
   182→          return null
   183→        }
   184→        return r.json()
   185→      })
   186→      .then((data) => {
   187→        if (cancelled) return
   188→        if (!data) {
   189→          // API returned non-OK (e.g. 401) — still show slots so user can see them
   190→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: true, loading: false })))
   191→          return
   192→        }
   193→        if (data.available === false) {
   194→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: false, loading: false })))

   197→          setSlotStatuses(
   198→            currentSlots.map((slot) => ({
   199→              slot,
   200→              available: !bookedSet.has(slot),
   201→              loading: false,
   202→            }))
   203→          )
   204→        }
   205→      })
   206→      .catch(() => {
   207→        if (!cancelled) {
   208→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: true, loading: false })))
   209→        }
   210→      })
   211→
   212→    return () => {
   213→      cancelled = true
   214→    }

   216→

   240→    },
   241→    onError: (err: Error) => {
   242→      toast.error(err.message || 'Failed to book appointment')

   244→  })

   258→    }
   259→
   260→    if (!scheduleData?.doctorId) {
   261→      toast.error('Schedule data not loaded. Please wait and try again.')
   262→      return
   263→    }
   264→
   265→    const selectedGender = bookingGender || user?.gender || 'Male'
   266→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   267→    bookMutation.mutate({
   268→      doctorId: scheduleData.doctorId,
   269→      bookingDate: dateStr,
   270→      timeSlot: selectedSlot,
   271→      bookingMode,
   272→      gender: selectedGender,
   273→      disease: disease.trim(),
   274→      description: description.trim(),
   275→      state: bookingState.trim(),
   276→      city: bookingCity.trim(),
   277→    })
   278→  }
   279→
   280→  // ── Calendar disabled dates ──
   281→  const disablePastDates = (date: Date) => {
   282→    return isBefore(startOfDay(date), startOfDay(new Date()))

   284→

   286→    if (!scheduleData?.schedules) return false // Allow all if no schedule loaded yet
   287→    const dayName = format(date, 'EEEE')
   288→    const hasSchedule = scheduleData.schedules.some((s) => s.day === dayName)
   289→    return !hasSchedule
   290→  }
   291→
   292→  // ── Loading state ──
   293→  if (doctorLoading || scheduleLoading) {
   294→    return <BookingPageSkeleton />

   296→
   297→  if (!doctor) {
   298→    return (
   299→      <div className="flex flex-col items-center justify-center py-20 text-center">
   300→        <AlertCircle className="h-12 w-12 text-muted-foreground/40 mb-4" />

   305→            <ChevronLeft className="h-4 w-4 mr-1" />

   308→        </Button>
   309→      </div>
   310→    )
   311→  }
   312→
   313→  const doc = doctor.doctor
   314→  const initials = doctor.name
   315→    .split(' ')
   316→    .map((n) => n[0])
   317→    .join('')
   318→    .slice(0, 2)
   319→    .toUpperCase()

   321→  return (
   322→    <div className="max-w-4xl mx-auto space-y-6">
   323→      {/* Breadcrumb */}
   324→      <motion.div {...fadeIn} className="flex items-center gap-2 text-sm text-muted-foreground">
   325→        <Link href="/dashboard/patient" className="hover:text-teal-600 transition-colors">Dashboard</Link>
   326→        <span>/</span>
   327→        <Link href="/dashboard/patient/appointments" className="hover:text-teal-600 transition-colors">Appointments</Link>
   328→        <span>/</span>
   329→        <span className="text-foreground font-medium truncate">Book Appointment</span>
   330→      </motion.div>
   331→
   332→      {/* Doctor Info Card */}
   333→      <motion.div {...fadeIn} transition={{ delay: 0.05 }}>
   334→        <Card className="border-teal-200/50 dark:border-teal-800/30 overflow-hidden">
   335→          <div className="bg-gradient-to-r from-teal-500 to-emerald-500 h-2" />
   336→          <CardContent className="p-5 md:p-6">
   337→            <div className="flex items-center gap-4">
   338→              <Avatar className="h-16 w-16 border-2 border-teal-200 dark:border-teal-800 shrink-0">
   339→                <AvatarImage src={doctor.profileImg !== 'default.png' ? doctor.profileImg : ''} />
   340→                <AvatarFallback className="bg-gradient-to-br from-teal-100 to-teal-200 text-teal-700 font-bold text-xl">
   341→                  {initials}
   342→                </AvatarFallback>
   343→              </Avatar>
   344→              <div className="flex-1 min-w-0">
   345→                <div className="flex items-center gap-2 flex-wrap">
   346→                  <h2 className="text-xl font-bold">{doctor.name}</h2>

   352→                <p className="text-teal-600 dark:text-teal-400 font-medium text-sm mt-0.5">
   353→                  {doc?.specialization || 'General Physician'}
   354→                </p>
   355→                <div className="flex items-center gap-4 mt-1.5 text-xs text-muted-foreground flex-wrap">
   356→                  {doc?.experience && <span className="flex items-center gap-1"><Clock className="h-3 w-3" />{doc.experience}</span>}
   357→                  {doc?.city && <span className="flex items-center gap-1"><MapPin className="h-3 w-3" />{doc.city}</span>}
   358→                  <span className="flex items-center gap-1 font-semibold text-teal-700 dark:text-teal-400">
   359→                    <IndianRupee className="h-3 w-3" />{fees} consultation
   360→                  </span>

   378→          {/* Step 1: Select Date */}

   383→                  <CalendarDays className="h-5 w-5 text-teal-500" />

   447→                        {' · '}
   448→                        {scheduleForSelectedDay.slotDuration}min slots

   450→                      {scheduleData?.dailyLimit && (

   489→                    <Separator />

   491→                    {/* Time slots grid */}
