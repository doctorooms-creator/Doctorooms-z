   104→   104→function getRelativeTime(isoString: string): string {
   105→   105→  try {
   106→   106→    return formatDistanceToNow(new Date(isoString), { addSuffix: true })
   107→   107→  } catch {
   108→   108→    return ''
   109→   109→  }
   110→   110→}
   111→   111→
   112→   112→function getOpdColor(percentage: number): string {
   113→   113→  if (percentage < 80) return 'bg-emerald-500'
   114→   114→  if (percentage <= 95) return 'bg-amber-500'
   115→   115→  return 'bg-red-500'
   116→   116→}
