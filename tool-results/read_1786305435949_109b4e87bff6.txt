   106→   106→function formatFileSize(bytes: number) {
   107→   107→  if (!bytes || bytes === 0) return null
   108→   108→  const k = 1024
   109→   109→  const sizes = ['B', 'KB', 'MB', 'GB']
   110→   110→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   111→   111→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`
   112→   112→}
