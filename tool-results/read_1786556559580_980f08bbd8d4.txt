    61→function FadeUpSection({
    62→  children,
    63→  className,
    64→  delay = 0,
    65→}: {
    66→  children: React.ReactNode
    67→  className?: string
    68→  delay?: number
    69→}) {
    70→  const ref = useRef<HTMLDivElement>(null)
    71→  const inView = useInView(ref, { once: true, margin: '-60px' })
    72→  return (
    73→    <motion.div
    74→      ref={ref}
    75→      initial={{ opacity: 0, y: 30 }}
    76→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    77→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    78→      className={className}
    79→    >
    80→      {children}
    81→    </motion.div>
    82→  )
    83→}

   109→// ── Count-up hook ────────────────────────────────────────────────────────────
   110→
   111→function useCountUp(target: number, inView: boolean) {
   112→  const [value, setValue] = useState(0)
   113→  const rafRef = useRef<number | null>(null)
   114→
   115→  const animate = useCallback(() => {
   116→    const duration = 1800
   117→    const start = performance.now()
   118→    const from = 0
   119→    const to = target
   120→
   121→    const step = (now: number) => {
   122→      const elapsed = now - start
   123→      const progress = Math.min(elapsed / duration, 1)
   124→      const eased = 1 - Math.pow(1 - progress, 3)
   125→      setValue(Math.floor(from + (to - from) * eased))
   126→      if (progress < 1) {
   127→        rafRef.current = requestAnimationFrame(step)
   128→      }
   129→    }
   130→    rafRef.current = requestAnimationFrame(step)
   131→  }, [target])
   132→
   133→  useEffect(() => {
   134→    if (inView) animate()
   135→    return () => {
   136→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   137→    }
   138→  }, [inView, animate])
   139→
   140→  return value
   141→}

   306→      ))}
   307→      <span className="ml-1 text-sm font-medium text-muted-foreground">{rating}</span>
