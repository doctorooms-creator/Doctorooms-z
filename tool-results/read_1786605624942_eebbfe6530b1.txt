   102→

   104→  const queryClient = useQueryClient()
   105→  const queueRef = useRef<HTMLDivElement>(null)
   106→
   107→  // Form state
   108→  const [patientName, setPatientName] = useState('')
   109→  const [mobileNo, setMobileNo] = useState('')
   110→  const [gender, setGender] = useState('')
   111→  const [age, setAge] = useState('')
   112→  const [disease, setDisease] = useState('')
   113→  const [timeSlot, setTimeSlot] = useState('')
   114→  const [bookingMode, setBookingMode] = useState<string>('InPerson')
   115→  const [lastAddedId, setLastAddedId] = useState<string | null>(null)
   116→
   117→  // Hospital mode state
   118→  const [selectedDepartmentId, setSelectedDepartmentId] = useState<string>('')
   119→  const [selectedDoctorId, setSelectedDoctorId] = useState<string>('')
   120→  const [queueDoctorFilter, setQueueDoctorFilter] = useState<string>('all')
   121→
   122→  // Fetch today's queue (returns isHospitalMode flag)
   123→  const { data: queueData, isLoading: queueLoading } = useQuery<QueueData>({
   124→    queryKey: ['walkin-queue'],
   125→    queryFn: () =>
   126→      fetch('/api/dashboard/receptionist/walk-in').then((r) => r.json()),
   127→    refetchInterval: 15_000,
   128→  })
   129→
   130→  const isHospitalMode = queueData?.isHospitalMode === true
   131→
   132→  // Fetch departments & doctors for hospital mode
   133→  const { data: hospitalScheduleData, isLoading: hospitalScheduleLoading } = useQuery<{
   134→    isHospitalMode: boolean
   135→    departments: Department[]
   136→    todayName: string
   137→  }>({
   138→    queryKey: ['walkin-hospital-schedule'],
   139→    queryFn: () =>
   140→      fetch('/api/dashboard/receptionist/schedule').then((r) => r.json()),
   141→    enabled: isHospitalMode,
   142→  })
   143→
   144→  const departments = hospitalScheduleData?.departments ?? []
   145→
   146→  // Get selected department's doctors
   147→  const departmentDoctors = useMemo(() => {
   148→    if (!selectedDepartmentId) return []
   149→    const dept = departments.find(
   150→      (d) => d.department.id === selectedDepartmentId
   151→    )
   152→    return dept?.doctors ?? []
   153→  }, [departments, selectedDepartmentId])
   154→
   155→  // Fetch clinic-mode doctor schedule
   156→  const { data: clinicScheduleResponse } = useQuery<{
   157→    schedules: (ScheduleSlot | null)[]
   158→    todayName: string
   159→  }>({
   160→    queryKey: ['walkin-doctor-schedule'],
   161→    queryFn: () =>
   162→      fetch('/api/dashboard/receptionist/schedule').then((r) => r.json()),
   163→    enabled: !isHospitalMode,
   164→  })
   165→
   166→  // In hospital mode, fetch specific doctor's schedule when department + doctor selected
   167→  const { data: doctorScheduleResponse } = useQuery<{
   168→    isHospitalMode: boolean
   169→    departments: Department[]
   170→    todayName: string
   171→  }>({
   172→    queryKey: [
   173→      'walkin-doctor-schedule-detail',
   174→      selectedDepartmentId,
   175→      selectedDoctorId,
   176→    ],
   177→    queryFn: () => {
   178→      const params = new URLSearchParams()
   179→      if (selectedDepartmentId) params.set('departmentId', selectedDepartmentId)
   180→      if (selectedDoctorId) params.set('doctorId', selectedDoctorId)
   181→      return fetch(
   182→        `/api/dashboard/receptionist/schedule?${params.toString()}`
   183→      ).then((r) => r.json())
   184→    },
   185→    enabled: isHospitalMode && !!selectedDepartmentId && !!selectedDoctorId,
   186→  })
   187→

   189→  const availableSlots = useMemo(() => {
   190→    if (isHospitalMode) {
   191→      // Hospital mode: use doctor-specific schedule
   192→      if (!doctorScheduleResponse?.departments?.length) return []
   193→      const dept = doctorScheduleResponse.departments[0]
   194→      const doctor = dept?.doctors.find(
   195→        (d) => d.id === selectedDoctorId
   196→      )
   197→      if (!doctor?.schedules) return []
   198→
   199→      const todayName = doctorScheduleResponse.todayName
   200→      const todaySchedule = doctor.schedules.find(
   201→        (s) => s?.day === todayName
   202→      )
   203→      if (!todaySchedule?.timeSlots?.length) return []
   204→
   205→      // Filter out booked slots for this specific doctor
   206→      const bookedSlots = new Set(
   207→        (queueData?.queue ?? [])
   208→          .filter(
   209→            (q: QueueItem) =>
   210→              q.timeSlot && q.doctorId === selectedDoctorId
   211→          )
   212→          .map((q: QueueItem) => q.timeSlot)
   213→      )
   214→
   215→      return todaySchedule.timeSlots.filter(
   216→        (slot: string) => !bookedSlots.has(slot)
   217→      )
   218→    }
   219→
   220→    // Clinic mode: use original schedule logic
   221→    if (!clinicScheduleResponse?.schedules || !queueData) return []
   222→
   223→    const todaySchedule = clinicScheduleResponse.schedules.find(
   224→      (s) => s?.day === clinicScheduleResponse.todayName
   225→    )
   226→
   227→    if (!todaySchedule?.timeSlots?.length) return []
   228→
   229→    const bookedSlots = new Set(
   230→      queueData.queue
   231→        .filter((q: QueueItem) => q.timeSlot)
   232→        .map((q: QueueItem) => q.timeSlot)
   233→    )
   234→
   235→    return todaySchedule.timeSlots.filter((slot: string) => !bookedSlots.has(slot))
   236→  }, [
   237→    isHospitalMode,
   238→    doctorScheduleResponse,
   239→    clinicScheduleResponse,
   240→    queueData,
   241→    selectedDoctorId,
   242→  ])
   243→

   245→  const walkInMutation = useMutation({
   246→    mutationFn: (body: Record<string, unknown>) =>
   247→      fetch('/api/dashboard/receptionist/walk-in', {
   248→        method: 'POST',
   249→        headers: { 'Content-Type': 'application/json' },
   250→        body: JSON.stringify(body),
   251→      }).then((r) => r.json()),
   252→    onSuccess: (result) => {
   253→      if (result.success) {
   254→        if (isHospitalMode && result.booking?.tokenNumber) {
   255→          toast.success(
   256→            `Token ${result.booking.tokenNumber} assigned! Queue #${result.queuePosition}`
   257→          )
   258→        } else {
   259→          toast.success(
   260→            `Patient added to queue #${result.queuePosition}!`
   261→          )
   262→        }
   263→        setLastAddedId(result.booking.id)
   264→        // Reset form
   265→        setPatientName('')
   266→        setMobileNo('')
   267→        setGender('')
   268→        setAge('')
   269→        setDisease('')
   270→        setTimeSlot('')
   271→        setBookingMode('InPerson')

   273→        setSelectedDepartmentId('')
   274→        setSelectedDoctorId('')
   275→        // Invalidate and refetch queue
   276→        queryClient.invalidateQueries({ queryKey: ['walkin-queue'] })

   278→        queryClient.invalidateQueries({
   279→          queryKey: ['receptionist-appointments'],
   280→        })
   281→        // Scroll to queue after a tick
   282→        setTimeout(() => {
   283→          queueRef.current?.scrollIntoView({
   284→            behavior: 'smooth',
   285→            block: 'start',
   286→          })
   287→        }, 300)
   288→      } else {
   289→        toast.error(result.error || 'Failed to register patient')
   290→      }
   291→    },
   292→    onError: () => {
   293→      toast.error('Failed to register walk-in patient')
   294→    },
   295→  })
   296→

   298→    (e: React.FormEvent) => {

   300→      if (!patientName.trim() || !disease.trim()) {
   301→        toast.error('Patient name and disease/reason are required')
   302→        return
   303→      }
   304→      if (isHospitalMode && (!selectedDepartmentId || !selectedDoctorId)) {
   305→        toast.error('Please select both department and doctor')
   306→        return
   307→      }
   308→
   309→      const body: Record<string, unknown> = {
   310→        patientName: patientName.trim(),
   311→        mobileNo: mobileNo.trim(),
   312→        gender,
   313→        age: age ? parseInt(age, 10) : null,
   314→        disease: disease.trim(),
   315→        timeSlot: timeSlot || null,
   316→        bookingMode,
   317→      }
   318→
   319→      if (isHospitalMode) {
   320→        body.departmentId = selectedDepartmentId
   321→        body.doctorId = selectedDoctorId
   322→      }
   323→
   324→      walkInMutation.mutate(body)
   325→    },
   326→    [
   327→      patientName,
   328→      mobileNo,
   329→      gender,
   330→      age,
   331→      disease,
   332→      timeSlot,
   333→      bookingMode,
   334→      walkInMutation,
   335→      isHospitalMode,
   336→      selectedDepartmentId,
   337→      selectedDoctorId,
   338→    ]
   339→  )
   340→
   341→  const isSubmitting = walkInMutation.isPending
   342→  const queue = queueData?.queue ?? []
   343→  const opdLimit = queueData?.opdLimit ?? 0
   344→  const totalInQueue = queueData?.totalInQueue ?? 0
   345→  const opdCompletedToday = queueData?.opdCompletedToday ?? 0
   346→

   349→    if (!isHospitalMode || queueDoctorFilter === 'all') return queue

   352→

   355→    if (!isHospitalMode) return []

   358→      if (item.doctorId && item.doctorName) {
   359→        seen.set(item.doctorId, item.doctorName)
   360→      }
   361→    }
   362→    return Array.from(seen.entries()).map(([id, name]) => ({
   363→      id,
   364→      name,
   365→    }))
   366→  }, [queue, isHospitalMode])
   367→
   368→  // Check if form is valid for submit
   369→  const canSubmit =
   370→    patientName.trim() &&
   371→    disease.trim() &&
   372→    (!isHospitalMode || (selectedDepartmentId && selectedDoctorId))
   373→
   374→  return (
   375→    <div className="space-y-6">
   376→      {/* Header */}
   377→      <div className="flex items-center gap-3">
   378→        <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/40">
   379→          <UserPlus className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   380→        </div>
   381→        <div>
   382→          <h2 className="text-xl font-semibold tracking-tight">
   383→            Walk-in Registration
   384→          </h2>
   385→          <p className="text-sm text-muted-foreground">
   386→            {isHospitalMode
   387→              ? 'Register walk-in patients to a department and doctor'
   388→              : 'Register walk-in patients directly to the queue'}
   389→          </p>
   390→        </div>
   391→        {isHospitalMode && (
   392→          <Badge className="ml-2 bg-amber-100 text-amber-700 hover:bg-amber-100 dark:bg-amber-900/40 dark:text-amber-400">
   393→            <Building2 className="mr-1 h-3 w-3" />
   394→            Hospital Mode
   395→          </Badge>
   396→        )}
   397→      </div>
   398→
   399→      <div className="grid gap-6 lg:grid-cols-[380px_1fr]">
   400→        {/* ============ LEFT: Registration Form ============ */}
   401→        <motion.div
   402→          initial={{ opacity: 0, x: -20 }}
   403→          animate={{ opacity: 1, x: 0 }}
   404→          transition={{ duration: 0.3 }}
   405→        >
   406→          <Card>
   407→            <CardHeader className="pb-3">
   408→              <CardTitle className="text-base">Quick Registration</CardTitle>
   409→            </CardHeader>
   410→            <CardContent>
   411→              <form onSubmit={handleSubmit} className="space-y-3">
   412→                {/* ── Hospital Mode: Department + Doctor Selection ── */}
   413→                {isHospitalMode && (
   414→                  <motion.div
   415→                    initial={{ opacity: 0, height: 0 }}
   416→                    animate={{ opacity: 1, height: 'auto' }}
   417→                    transition={{ duration: 0.3 }}
   418→                    className="space-y-3"
   419→                  >
   420→                    {/* Department Select */}

   422→                      <Label
   423→                        htmlFor="department"
   424→                        className="text-xs font-medium"
   425→                      >
   426→                        <Building2 className="mr-1 inline h-3 w-3" />
   427→                        Department <span className="text-red-500">*</span>
   428→                      </Label>
   429→                      <Select
   430→                        value={selectedDepartmentId}

   432→                          setSelectedDepartmentId(val)
   433→                          setSelectedDoctorId('')
   434→                          setTimeSlot('')
   435→                        }}
   436→                        disabled={
   437→                          isSubmitting || hospitalScheduleLoading
   438→                        }
   439→                      >
   440→                        <SelectTrigger className="h-9 w-full">
   441→                          <SelectValue placeholder="Select department" />
   442→                        </SelectTrigger>
   443→                        <SelectContent>
   444→                          {departments.length === 0 &&
   445→                            !hospitalScheduleLoading && (
   446→                              <div className="px-2 py-1.5 text-xs text-muted-foreground">
   447→                                No departments available
   448→                              </div>
   449→                            )}
   450→                          {hospitalScheduleLoading && (
   451→                            <div className="flex items-center gap-2 px-2 py-1.5">
   452→                              <Loader2 className="h-3 w-3 animate-spin" />
   453→                              <span className="text-xs text-muted-foreground">
   454→                                Loading departments...
   455→                              </span>
   456→                            </div>
   457→                          )}
   458→                          {departments.map((dept) => (
   459→                            <SelectItem
   460→                              key={dept.department.id}
   461→                              value={dept.department.id}
   462→                            >
   463→                              <span className="flex items-center gap-2">
   464→                                {dept.department.icon && (
   465→                                  <span>{dept.department.icon}</span>
   466→                                )}
   467→                                <span className="font-medium">
   468→                                  {dept.department.name}
   469→                                </span>
   470→                                {dept.department.shortCode && (
   471→                                  <span className="text-muted-foreground">
   472→                                    ({dept.department.shortCode})
   473→                                  </span>
   474→                                )}
   475→                              </span>
   476→                            </SelectItem>
   477→                          ))}
   478→                        </SelectContent>
   479→                      </Select>
   480→                    </div>
   481→
   482→                    {/* Doctor Select */}
   483→                    <div className="space-y-1">
   484→                      <Label
   485→                        htmlFor="doctor"
   486→                        className="text-xs font-medium"
   487→                      >
   488→                        <Stethoscope className="mr-1 inline h-3 w-3" />
   489→                        Doctor <span className="text-red-500">*</span>
   490→                      </Label>
   491→                      <Select
   492→                        value={selectedDoctorId}
   493→                        onValueChange={(val) => {
   494→                          setSelectedDoctorId(val)
   495→                          setTimeSlot('')
   496→                        }}
   497→                        disabled={
   498→                          isSubmitting || !selectedDepartmentId

   500→                      >
   501→                        <SelectTrigger className="h-9 w-full">

   508→                        <SelectContent>

   510→                            selectedDepartmentId && (
   511→                              <div className="px-2 py-1.5 text-xs text-muted-foreground">
   512→                                No doctors in this department
   513→                              </div>
   514→                            )}
   515→                          {departmentDoctors.map((doc) => (
   516→                            <SelectItem key={doc.id} value={doc.id}>
   517→                              <span className="flex items-center gap-2">
   518→                                <span className="font-medium">
   519→                                  {doc.name}
   520→                                </span>
   521→                                {doc.specialization && (
   522→                                  <span className="text-xs text-muted-foreground">
   523→                                    {doc.specialization}
   524→                                  </span>
   525→                                )}
   526→                              </span>
   527→                            </SelectItem>
   528→                          ))}
   529→                        </SelectContent>
   530→                      </Select>
   531→                    </div>
   532→
   533→                    {/* Divider */}
   534→                    <div className="flex items-center gap-2 pt-1">
   535→                      <div className="h-px flex-1 bg-border" />
   536→                      <span className="text-[10px] uppercase tracking-wider text-muted-foreground">
   537→                        Patient Details
   538→                      </span>
   539→                      <div className="h-px flex-1 bg-border" />
   540→                    </div>
   541→                  </motion.div>
   542→                )}
   543→
   544→                {/* Patient Name */}
   545→                <div className="space-y-1">
   546→                  <Label htmlFor="patientName" className="text-xs font-medium">
   547→                    Patient Name <span className="text-red-500">*</span>

   549→                  <Input
   550→                    id="patientName"
   551→                    placeholder="Full name"
   552→                    value={patientName}
   553→                    onChange={(e) => setPatientName(e.target.value)}

   555→                    autoFocus

   557→                  />

   559→
   560→                {/* Mobile + Gender row */}

   563→                    <Label htmlFor="mobileNo" className="text-xs font-medium">
   564→                      Mobile No.
