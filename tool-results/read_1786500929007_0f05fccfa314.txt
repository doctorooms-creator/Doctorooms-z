    84→/*  Dose Tag Input (inline component)                                  */
    85→/* ------------------------------------------------------------------ */
    86→
    87→function DoseTagInput({
    88→  tags,
    89→  onChange,
    90→}: {
    91→  tags: string[]
    92→  onChange: (tags: string[]) => void
    93→}) {
    94→  const [input, setInput] = useState('')
    95→  const inputRef = useRef<HTMLInputElement>(null)
    96→
    97→  const addTag = (value: string) => {
    98→    const trimmed = value.trim()
    99→    if (!trimmed) return
   100→    if (tags.includes(trimmed)) {
   101→      toast.error('This dose option already exists')
   102→      return
   103→    }
   104→    onChange([...tags, trimmed])
   105→    setInput('')
   106→  }
   107→
   108→  const removeTag = (index: number) => {
   109→    const next = [...tags]
   110→    next.splice(index, 1)
   111→    onChange(next)
   112→  }
   113→
   114→  const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
   115→    if (e.key === 'Enter' || e.key === ',') {
   116→      e.preventDefault()
   117→      addTag(input)
   118→    }
   119→    if (e.key === 'Backspace' && input === '' && tags.length > 0) {
   120→      removeTag(tags.length - 1)
   121→    }
   122→  }
   123→
   124→  return (
   125→    <div
   126→      className="flex flex-wrap items-center gap-1.5 min-h-[42px] rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-sm focus-within:ring-1 focus-within:ring-ring cursor-text"
   127→      onClick={() => inputRef.current?.focus()}
   128→    >
   129→      {tags.map((tag, i) => (
   130→        <span
   131→          key={i}
   132→          className={cn(
   133→            'inline-flex items-center gap-1 rounded-md px-2 py-0.5 text-xs font-medium transition-colors',
   134→            i === 0
   135→              ? 'bg-teal-100 text-teal-800 dark:bg-teal-900/40 dark:text-teal-300 border border-teal-200 dark:border-teal-800'
   136→              : 'bg-muted text-muted-foreground border border-border'
   137→          )}
   138→        >
   139→          {tag}
   140→          <button
   141→            type="button"
   142→            className="ml-0.5 rounded-full p-0.5 hover:bg-black/10 dark:hover:bg-white/10 transition-colors"
   143→            onClick={(e) => {
   144→              e.stopPropagation()
   145→              removeTag(i)
   146→            }}
   147→          >
   148→            <X className="h-3 w-3" />
   149→          </button>
   150→          {i === 0 && (
   151→            <span className="ml-0.5 text-[9px] font-bold uppercase tracking-wider opacity-60">default</span>
   152→          )}
   153→        </span>
   154→      ))}
   155→      <input
   156→        ref={inputRef}
   157→        value={input}
   158→        onChange={(e) => setInput(e.target.value)}
   159→        onKeyDown={handleKeyDown}
   160→        onBlur={() => {
   161→          if (input.trim()) addTag(input)
   162→        }}
   163→        placeholder={tags.length === 0 ? 'e.g. 500mg, then Enter' : 'Add more...'}
   164→        className="flex-1 min-w-[100px] bg-transparent outline-none placeholder:text-muted-foreground text-sm"
   165→      />
   166→    </div>
   167→  )
   168→}

   267→                dose: JSON.stringify(body.doseArray),
