    61→function FadeUpSection({
    62→  children,
    63→  className,
    64→  delay = 0,
    65→}: {
    66→  children: React.ReactNode
    67→  className?: string
    68→  delay?: number
    69→}) {
    70→  const ref = useRef<HTMLDivElement>(null)
    71→  const inView = useInView(ref, { once: true, margin: '-60px' })
    72→  return (
    73→    <motion.div
    74→      ref={ref}
    75→      initial={{ opacity: 0, y: 30 }}
    76→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    77→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    78→      className={className}
    79→    >
    80→      {children}
    81→    </motion.div>
    82→  )
    83→}

   109→// ── Count-up hook ────────────────────────────────────────────────────────────
   110→
   111→function useCountUp(target: number, inView: boolean) {
   112→  const [value, setValue] = useState(0)
   113→  const rafRef = useRef<number | null>(null)
   114→
   115→  const animate = useCallback(() => {
   116→    const duration = 1800
   117→    const start = performance.now()
   118→    const from = 0
   119→    const to = target
   120→
   121→    const step = (now: number) => {
   122→      const elapsed = now - start
   123→      const progress = Math.min(elapsed / duration, 1)
   124→      const eased = 1 - Math.pow(1 - progress, 3)
   125→      setValue(Math.floor(from + (to - from) * eased))
   126→      if (progress < 1) {
   127→        rafRef.current = requestAnimationFrame(step)
   128→      }
   129→    }
   130→    rafRef.current = requestAnimationFrame(step)
   131→  }, [target])
   132→
   133→  useEffect(() => {
   134→    if (inView) animate()
   135→    return () => {
   136→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   137→    }
   138→  }, [inView, animate])
   139→
   140→  return value
   141→}

   755→function StatsGrid({ stats }: { stats: Stats }) {
   756→  const ref = useRef<HTMLDivElement>(null)
   757→  const inView = useInView(ref, { once: true, margin: '100px' })
   758→  const [delayPassed, setDelayPassed] = useState(false)
   759→
   760→  useEffect(() => {
   761→    const id = setTimeout(() => setDelayPassed(true), 800)
   762→    return () => clearTimeout(id)
   763→  }, [])
   764→
   765→  const shouldAnimate = inView || delayPassed
   766→
   767→  const doctorCount = useCountUp(stats.doctorCount, shouldAnimate)
   768→  const hospitalCount = useCountUp(stats.hospitalCount, shouldAnimate)
   769→  const patientCount = useCountUp(stats.patientCount, shouldAnimate)
   770→  const bookingCount = useCountUp(stats.bookingCount, shouldAnimate)
   771→
   772→  const items = [
   773→    { icon: Stethoscope, value: doctorCount, label: 'Verified Doctors', suffix: '+' },
   774→    { icon: Building2, value: hospitalCount, label: 'Partner Hospitals', suffix: '+' },
   775→    { icon: Users, value: patientCount, label: 'Happy Patients', suffix: '+' },
   776→    { icon: CalendarCheck, value: bookingCount, label: 'Appointments Booked', suffix: '+' },
   777→  ]
   778→
   779→  return (
   780→    <div ref={ref} className="grid grid-cols-2 gap-6 lg:grid-cols-4">
   781→      {items.map((item, i) => (
   782→        <FadeUpItem key={item.label} index={i} className="flex flex-col items-center text-center">
   783→          <div className="mb-3 flex h-14 w-14 items-center justify-center rounded-full bg-teal-100 text-teal-600 dark:bg-teal-900/40 dark:text-teal-400">
   784→            <item.icon className="h-7 w-7" />
   785→          </div>
   786→          <p className="text-3xl font-bold text-gray-900 dark:text-white sm:text-4xl">
   787→            {item.value.toLocaleString('en-IN')}
   788→            {item.suffix}
   789→          </p>
   790→          <p className="mt-1 text-sm text-gray-500 dark:text-gray-400">{item.label}</p>
   791→          <div className="mt-2 h-1 w-8 rounded-full bg-teal-500" />
   792→        </FadeUpItem>

   794→    </div>
   795→  )
