     6→// ============ HELPER: Parse OPD timings string into schedule entries ============
     7→function parseOpdTimings(opdTimings: string): { day: string; startTime: string; endTime: string }[] {
     8→  // Examples: "Mon/Wed/Fri 10:00-1:00" → [{day:"Monday", startTime:"10:00", endTime:"13:00"}]
     9→  //           "Tue/Thu/Sat 9:00-12:00" → [{day:"Tuesday", startTime:"09:00", endTime:"12:00"}]
    10→  //           "Mon-Sat 10:00-2:00" → [Mon..Sat]
    11→  const dayMap: Record<string, string> = {
    12→    Mon: 'Monday', Tue: 'Tuesday', Wed: 'Wednesday', Thu: 'Thursday',
    13→    Fri: 'Friday', Sat: 'Saturday', Sun: 'Sunday',
    14→  }
    15→
    16→  const schedules: { day: string; startTime: string; endTime: string }[] = []
    17→  const parts = opdTimings.split(' ')
    18→  if (parts.length < 2) return schedules
    19→
    20→  const daysPart = parts[0]
    21→  const timePart = parts[1]
    22→
    23→  // Parse time: "10:00-1:00" or "10:00-13:00"
    24→  const [startStr, endStr] = timePart.split('-')
    25→  const startTime = padTime(startStr)
    26→  const endTime = padTime(endStr)
    27→
    28→  // Parse days: "Mon/Wed/Fri" (slash = explicit list) or "Mon-Sat" (dash = range)
    29→  const dayOrder = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun']
    30→
    31→  let dayAbbrs: string[]
    32→  if (daysPart.includes('/')) {
    33→    dayAbbrs = daysPart.split('/')
    34→  } else {
    35→    // Range: "Mon-Sat" → expand to Mon..Sat
    36→    const [start, end] = daysPart.split('-')
    37→    const startIdx = dayOrder.indexOf(start)
    38→    const endIdx = dayOrder.indexOf(end)
    39→    if (startIdx !== -1 && endIdx !== -1 && startIdx <= endIdx) {
    40→      dayAbbrs = dayOrder.slice(startIdx, endIdx + 1)
    41→    } else {
    42→      dayAbbrs = [daysPart] // fallback
    43→    }
    44→  }
    45→
    46→  for (const abbr of dayAbbrs) {
    47→    const fullDay = dayMap[abbr.trim()]
    48→    if (fullDay) {
    49→      schedules.push({ day: fullDay, startTime, endTime })
    50→    }
    51→  }
    52→
    53→  return schedules
    54→}
    55→
    56→function padTime(t: string): string {
    57→  const parts = t.split(':')
    58→  if (parts.length === 2) return `${parts[0].padStart(2, '0')}:${parts[1].padStart(2, '0')}`
    59→  // If no colon, assume it's like "1:00" → "01:00" or just "1" → "01:00"
    60→  return `${parts[0].padStart(2, '0')}:00`
    61→}
