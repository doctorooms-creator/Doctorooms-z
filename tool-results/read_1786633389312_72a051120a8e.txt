   171→   171→function isAbnormal(param: string, value: number): boolean {
   172→   172→  const T = VITAL_THRESHOLDS as Record<string, Record<string, number>>
   173→   173→  const t = T[param]
   174→   174→  if (!t) return false
   175→   175→  if (param === 'spo2') return value < t.warning
   176→   176→  if (param === 'bpSystolic') return value < (t.criticalLow ?? 999) || value > (t.warningHigh ?? 999)
   177→   177→  if (param === 'bpDiastolic') return value > (t.warningHigh ?? 999)
   178→   178→  if (param === 'pulse') return value < (t.criticalLow ?? 999) || value > (t.warningHigh ?? 999)
   179→   179→  if (param === 'temperature') return value > (t.warningHigh ?? 999)
   180→   180→  if (param === 'respiratoryRate') return value < (t.criticalLow ?? 999) || value > (t.criticalHigh ?? 999)
   181→   181→  return false
   182→   182→}
