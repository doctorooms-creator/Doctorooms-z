    90→

   101→// ==================== MAIN COMPONENT ====================
   102→
   103→export default function MedicinesPage() {
   104→  const queryClient = useQueryClient()
   105→  const [search, setSearch] = useState('')
   106→  const [dialogOpen, setDialogOpen] = useState(false)
   107→  const [editingMedicine, setEditingMedicine] = useState<Medicine | null>(null)
   108→  const [form, setForm] = useState<FormData>(emptyForm)
   109→  const [deleteId, setDeleteId] = useState<string | null>(null)
   110→  const [debouncedSearch, setDebouncedSearch] = useState('')
   111→
   112→  // Debounce search input
   113→  useEffect(() => {
   114→    const timer = setTimeout(() => setDebouncedSearch(search), 300)
   115→    return () => clearTimeout(timer)
   116→  }, [search])
   117→
   118→  // ---------- Query ----------
   119→
   120→  const { data, isLoading } = useQuery<{ medicines: Medicine[] }>({
   121→    queryKey: ['receptionist-medicines', debouncedSearch],
   122→    queryFn: () =>
   123→      fetch(
   124→        `/api/receptionist/medicines${debouncedSearch ? `?search=${encodeURIComponent(debouncedSearch)}` : ''}`
   125→      ).then((r) => r.json()),
   126→  })
   127→
   128→  const medicines = data?.medicines || []
   129→
   130→  // ---------- Mutations ----------
   131→
   132→  const createMutation = useMutation({
   133→    mutationFn: (body: FormData) =>
   134→      fetch('/api/receptionist/medicines', {
   135→        method: 'POST',
   136→        headers: { 'Content-Type': 'application/json' },
   137→        body: JSON.stringify(body),
   138→      }).then((r) => r.json()),
   139→    onSuccess: (data) => {
   140→      if (data.error) {
   141→        toast.error(data.error)
   142→        return
   143→      }
   144→      queryClient.invalidateQueries({ queryKey: ['receptionist-medicines'] })
   145→      toast.success('Medicine added successfully')
   146→      closeDialog()
   147→    },
   148→    onError: () => {
   149→      toast.error('Failed to add medicine')
   150→    },
   151→  })
   152→
   153→  const updateMutation = useMutation({
   154→    mutationFn: ({ id, body }: { id: string; body: FormData }) =>
   155→      fetch(`/api/receptionist/medicines/${id}`, {
   156→        method: 'PUT',
   157→        headers: { 'Content-Type': 'application/json' },
   158→        body: JSON.stringify(body),
   159→      }).then((r) => r.json()),
   160→    onSuccess: (data) => {
   161→      if (data.error) {
   162→        toast.error(data.error)
   163→        return
   164→      }
   165→      queryClient.invalidateQueries({ queryKey: ['receptionist-medicines'] })
   166→      toast.success('Medicine updated successfully')
   167→      closeDialog()
   168→    },
   169→    onError: () => {
   170→      toast.error('Failed to update medicine')
   171→    },
   172→  })
   173→
   174→  const deleteMutation = useMutation({
   175→    mutationFn: (id: string) =>
   176→      fetch(`/api/receptionist/medicines/${id}`, { method: 'DELETE' }).then((r) =>
   177→        r.json()
   178→      ),
   179→    onSuccess: () => {
   180→      queryClient.invalidateQueries({ queryKey: ['receptionist-medicines'] })
   181→      toast.success('Medicine deleted successfully')
   182→      setDeleteId(null)
   183→    },
   184→    onError: () => {
   185→      toast.error('Failed to delete medicine')
   186→    },
   187→  })
   188→
   189→  const toggleMutation = useMutation({
   190→    mutationFn: (id: string) =>
   191→      fetch(`/api/receptionist/medicines/${id}/toggle`, { method: 'PATCH' }).then(
   192→        (r) => r.json()
   193→      ),
   194→    onSuccess: (data) => {
   195→      if (data.error) {
   196→        toast.error(data.error)
   197→        return
   198→      }
   199→      queryClient.invalidateQueries({ queryKey: ['receptionist-medicines'] })
   200→      toast.success(
   201→        data.status === 'Active'
   202→          ? 'Medicine activated'
   203→          : 'Medicine deactivated'
   204→      )
   205→    },
   206→    onError: () => {
   207→      toast.error('Failed to toggle status')
   208→    },
   209→  })
   210→
   211→  // ---------- Handlers ----------
   212→
   213→  const openAddDialog = useCallback(() => {
   214→    setEditingMedicine(null)
   215→    setForm(emptyForm)
   216→    setDialogOpen(true)
   217→  }, [])
   218→
   219→  const openEditDialog = useCallback((med: Medicine) => {
   220→    setEditingMedicine(med)
   221→    setForm({
   222→      name: med.name,
   223→      morning: med.morning,
   224→      afternoon: med.afternoon,
   225→      evening: med.evening,
   226→      dose: med.dose,
   227→      tab: med.tab,
   228→      description: med.description,
   229→    })
   230→    setDialogOpen(true)
   231→  }, [])
   232→
   233→  const closeDialog = useCallback(() => {
   234→    setDialogOpen(false)
   235→    setEditingMedicine(null)
   236→    setForm(emptyForm)
   237→  }, [])
   238→
   239→  const handleSave = useCallback(() => {
   240→    if (!form.name.trim()) {
   241→      toast.error('Medicine name is required')
   242→      return
   243→    }
   244→    if (editingMedicine) {
   245→      updateMutation.mutate({ id: editingMedicine.id, body: form })
   246→    } else {
   247→      createMutation.mutate(form)
   248→    }
   249→  }, [form, editingMedicine, createMutation, updateMutation])
   250→
   251→  const isSaving = createMutation.isPending || updateMutation.isPending
   252→
   253→  // ==================== RENDER ====================
   254→
   255→  return (
   256→    <div className="space-y-6">
   257→      {/* ========== A. PAGE HEADER ========== */}
   258→      <motion.div
   259→        initial={{ opacity: 0, y: -10 }}
   260→        animate={{ opacity: 1, y: 0 }}
   261→        transition={{ duration: 0.35 }}
   262→        className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between"
   263→      >
   264→        <div>
   265→          <h1 className="text-2xl font-bold tracking-tight sm:text-3xl">
   266→            Medicine List
   267→          </h1>
   268→          <p className="text-sm text-muted-foreground mt-1">
   269→            Manage the doctor&apos;s medicine master list
   270→          </p>
   271→        </div>
   272→        <Button
   273→          onClick={openAddDialog}
   274→          className="bg-gradient-to-r from-teal-500 to-teal-600 hover:from-teal-600 hover:to-teal-700 text-white shadow-md shadow-teal-500/20 mt-2 sm:mt-0"
   275→        >
   276→          <Plus className="mr-2 h-4 w-4" />
   277→          Add Medicine
   278→        </Button>
   279→      </motion.div>
   280→
   281→      {/* ========== B. SEARCH + COUNT ========== */}
   282→      <motion.div
   283→        initial={{ opacity: 0, y: 8 }}
   284→        animate={{ opacity: 1, y: 0 }}
   285→        transition={{ duration: 0.3, delay: 0.05 }}
   286→        className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between"
   287→      >
   288→        <div className="relative w-full sm:max-w-xs">
   289→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   290→          <Input
   291→            placeholder="Search medicines..."
   292→            value={search}
   293→            onChange={(e) => setSearch(e.target.value)}
   294→            className="pl-9"
   295→          />
   296→        </div>
   297→        <p className="text-sm text-muted-foreground">
   298→          Showing{' '}
   299→          <span className="font-medium text-foreground">{medicines.length}</span>{' '}

   308→            <div key={i} className="rounded-lg border border-border p-4">

   310→                <Skeleton className="h-5 w-40" />

   313→                  <Skeleton className="h-8 w-20" />
   314→                  <Skeleton className="h-8 w-20" />
   315→                </div>
   316→              </div>
   317→            </div>
   318→          ))}
   319→        </div>

   321→
   322→      {/* ========== D. EMPTY STATE ========== */}
   323→      {!isLoading && medicines.length === 0 && (
   324→        <motion.div
   325→          initial={{ opacity: 0, y: 20 }}
   326→          animate={{ opacity: 1, y: 0 }}
   327→          className="flex flex-col items-center justify-center rounded-xl border border-dashed border-border py-20"
   328→        >
   329→          <div className="flex h-16 w-16 items-center justify-center rounded-full bg-muted">
   330→            <Pill className="h-8 w-8 text-muted-foreground/40" />
   331→          </div>
   332→          <p className="mt-4 text-sm font-medium text-muted-foreground">
   333→            {debouncedSearch
   334→              ? 'No medicines match your search'
   335→              : 'No medicines added yet'}
   336→          </p>
   337→          <p className="mt-1 text-xs text-muted-foreground">
   338→            {debouncedSearch
   339→              ? 'Try a different search term'
   340→              : 'Start by adding medicines to the master list'}
   341→          </p>
   342→          {!debouncedSearch && (
   343→            <Button
   344→              onClick={openAddDialog}
   345→              className="mt-4 bg-teal-500 hover:bg-teal-600 text-white"
   346→            >
   347→              <Plus className="mr-2 h-4 w-4" />
   348→              Add Your First Medicine

   350→          )}
   351→        </motion.div>
   352→      )}
   353→
   354→      {/* ========== E. DESKTOP TABLE ========== */}
   355→      {!isLoading && medicines.length > 0 && (
   356→        <>
   357→          {/* Desktop Table */}
   358→          <motion.div
   359→            variants={containerVariants}
   360→            initial="hidden"
   361→            animate="show"
   362→            className="hidden md:block rounded-lg border border-border overflow-hidden"
   363→          >
   364→            <Table>
   365→              <TableHeader>
   366→                <TableRow className="bg-muted/50 hover:bg-muted/50">
   367→                  <TableHead className="pl-4">Name</TableHead>
   368→                  <TableHead className="hidden lg:table-cell">
   369→                    <span className="inline-flex items-center gap-1">
   370→                      <Sun className="h-3.5 w-3.5 text-amber-500" />
   371→                      Morning
   372→                    </span>
   373→                  </TableHead>
   374→                  <TableHead className="hidden lg:table-cell">
   375→                    <span className="inline-flex items-center gap-1">
   376→                      <CloudSun className="h-3.5 w-3.5 text-orange-500" />
   377→                      Afternoon
   378→                    </span>
   379→                  </TableHead>
   380→                  <TableHead className="hidden lg:table-cell">
   381→                    <span className="inline-flex items-center gap-1">
   382→                      <Moon className="h-3.5 w-3.5 text-violet-500" />
   383→                      Evening

   390→                </TableRow>
   391→              </TableHeader>
   392→              <TableBody>
   393→                {medicines.map((med) => (
   394→                  <motion.tr
   395→                    key={med.id}
   396→                    variants={itemVariants}
   397→                    className="border-b transition-colors hover:bg-muted/50"
   398→                  >
   399→                    <TableCell className="pl-4 font-medium">
   400→                      {med.name}
   401→                    </TableCell>
   402→                    <TableCell className="hidden lg:table-cell">
   403→                      {med.morning || '—'}
   404→                    </TableCell>
   405→                    <TableCell className="hidden lg:table-cell">
   406→                      {med.afternoon || '—'}
   407→                    </TableCell>

   409→                      {med.evening || '—'}

   411→                    <TableCell className="hidden xl:table-cell">
   412→                      {med.dose || '—'}

   414→                    <TableCell className="hidden xl:table-cell">
   415→                      {med.tab}
   416→                    </TableCell>
   417→                    <TableCell>
   418→                      <button
   419→                        onClick={() => toggleMutation.mutate(med.id)}
   420→                        disabled={toggleMutation.isPending}
   421→                        className="cursor-pointer"
   422→                      >
   423→                        <Badge
   424→                          variant="secondary"
   425→                          className={
   426→                            med.status === 'Active'
   427→                              ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-400'
   428→                              : 'bg-secondary text-muted-foreground'
   429→                          }
   430→                        >
   431→                          {med.status}
   432→                        </Badge>
   433→                      </button>
   434→                    </TableCell>
   435→                    <TableCell className="text-right pr-4">
   436→                      <div className="flex items-center justify-end gap-1">
   437→                        <Button
   438→                          variant="ghost"
   439→                          size="icon"
   440→                          className="h-8 w-8"
   441→                          onClick={() => openEditDialog(med)}
   442→                        >
   443→                          <Pencil className="h-4 w-4" />
   444→                          <span className="sr-only">Edit</span>

   451→                        >
   452→                          <AlertDialogTrigger asChild>

   456→                              className="h-8 w-8 text-destructive hover:text-destructive"

   488→
   489→          {/* Mobile Card Layout */}
   490→          <motion.div

   501→                    <div className="flex items-start justify-between gap-2">
