    84→    84→/*  Dose Tag Input (inline component)                                  */
    85→    85→/* ------------------------------------------------------------------ */
    86→    86→
    87→    87→function DoseTagInput({
    88→    88→  tags,
    89→    89→  onChange,
    90→    90→}: {
    91→    91→  tags: string[]
    92→    92→  onChange: (tags: string[]) => void
    93→    93→}) {
    94→    94→  const [input, setInput] = useState('')
    95→    95→  const inputRef = useRef<HTMLInputElement>(null)
    96→    96→
    97→    97→  const addTag = (value: string) => {
    98→    98→    const trimmed = value.trim()
    99→    99→    if (!trimmed) return
   100→   100→    if (tags.includes(trimmed)) {
   101→   101→      toast.error('This dose option already exists')
   102→   102→      return
   103→   103→    }
   104→   104→    onChange([...tags, trimmed])
   105→   105→    setInput('')
   106→   106→  }
   107→   107→
   108→   108→  const removeTag = (index: number) => {
   109→   109→    const next = [...tags]
   110→   110→    next.splice(index, 1)
   111→   111→    onChange(next)
   112→   112→  }
   113→   113→
   114→   114→  const handleKeyDown = (e: KeyboardEvent<HTMLInputElement>) => {
   115→   115→    if (e.key === 'Enter' || e.key === ',') {
   116→   116→      e.preventDefault()
   117→   117→      addTag(input)
   118→   118→    }
   119→   119→    if (e.key === 'Backspace' && input === '' && tags.length > 0) {
   120→   120→      removeTag(tags.length - 1)
   121→   121→    }
   122→   122→  }
   123→   123→
   124→   124→  return (
   125→   125→    <div
   126→   126→      className="flex flex-wrap items-center gap-1.5 min-h-[42px] rounded-md border border-input bg-transparent px-3 py-2 text-sm shadow-sm focus-within:ring-1 focus-within:ring-ring cursor-text"
   127→   127→      onClick={() => inputRef.current?.focus()}
   128→   128→    >
   129→   129→      {tags.map((tag, i) => (
   130→   130→        <span
   131→   131→          key={i}
   132→   132→          className={cn(
   133→   133→            'inline-flex items-center gap-1 rounded-md px-2 py-0.5 text-xs font-medium transition-colors',
   134→   134→            i === 0
   135→   135→              ? 'bg-teal-100 text-teal-800 dark:bg-teal-900/40 dark:text-teal-300 border border-teal-200 dark:border-teal-800'
   136→   136→              : 'bg-muted text-muted-foreground border border-border'
   137→   137→          )}
   138→   138→        >
   139→   139→          {tag}
   140→   140→          <button
   141→   141→            type="button"
   142→   142→            className="ml-0.5 rounded-full p-0.5 hover:bg-black/10 dark:hover:bg-white/10 transition-colors"
   143→   143→            onClick={(e) => {
   144→   144→              e.stopPropagation()
   145→   145→              removeTag(i)
   146→   146→            }}
   147→   147→          >
   148→   148→            <X className="h-3 w-3" />
   149→   149→          </button>
   150→   150→          {i === 0 && (
   151→   151→            <span className="ml-0.5 text-[9px] font-bold uppercase tracking-wider opacity-60">default</span>
   152→   152→          )}
   153→   153→        </span>
   154→   154→      ))}
   155→   155→      <input
   156→   156→        ref={inputRef}
   157→   157→        value={input}
   158→   158→        onChange={(e) => setInput(e.target.value)}
   159→   159→        onKeyDown={handleKeyDown}
   160→   160→        onBlur={() => {
   161→   161→          if (input.trim()) addTag(input)
   162→   162→        }}
   163→   163→        placeholder={tags.length === 0 ? 'e.g. 500mg, then Enter' : 'Add more...'}
   164→   164→        className="flex-1 min-w-[100px] bg-transparent outline-none placeholder:text-muted-foreground text-sm"
   165→   165→      />
   166→   166→    </div>
   167→   167→  )
   168→   168→}
