   296→   296→        protected function _generateSeoURL($string, $wordLimit = 0)
   297→   297→        {
   298→   298→                $separator = '-';
   299→   299→
   300→   300→                if ($wordLimit != 0) {
   301→   301→                        $wordArr = explode(' ', $string);
   302→   302→                        $string = implode(' ', array_slice($wordArr, 0, $wordLimit));
   303→   303→                }
   304→   304→
   305→   305→                $quoteSeparator = preg_quote($separator, '#');
   306→   306→
   307→   307→                $trans = array(
   308→   308→                        '&.+?;' => '',
   309→   309→                        '[^\w\d _-]' => '',
   310→   310→                        '\s+' => $separator,
   311→   311→                        '(' . $quoteSeparator . ')+' => $separator
   312→   312→                );
   313→   313→
   314→   314→                $string = strip_tags($string);
   315→   315→                foreach ($trans as $key => $val) {
   316→   316→                        $string = preg_replace('#' . $key . '#i' . (UTF8_ENABLED ? 'u' : ''), $val, $string);
   317→   317→                }
   318→   318→
   319→   319→                $string = strtolower($string);
   320→   320→
   321→   321→                return trim(trim($string, $separator));
   322→   322→        }
