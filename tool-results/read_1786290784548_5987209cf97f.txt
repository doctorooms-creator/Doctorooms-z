   108→   108→// ── Count-up hook ────────────────────────────────────────────────────────────
   109→   109→
   110→   110→function useCountUp(target: number, inView: boolean) {
   111→   111→  const [value, setValue] = useState(0)
   112→   112→  const rafRef = useRef<number | null>(null)
   113→   113→
   114→   114→  const animate = useCallback(() => {
   115→   115→    const duration = 1800
   116→   116→    const start = performance.now()
   117→   117→    const from = 0
   118→   118→    const to = target
   119→   119→
   120→   120→    const step = (now: number) => {
   121→   121→      const elapsed = now - start
   122→   122→      const progress = Math.min(elapsed / duration, 1)
   123→   123→      const eased = 1 - Math.pow(1 - progress, 3)
   124→   124→      setValue(Math.floor(from + (to - from) * eased))
   125→   125→      if (progress < 1) {
   126→   126→        rafRef.current = requestAnimationFrame(step)
   127→   127→      }
   128→   128→    }
   129→   129→    rafRef.current = requestAnimationFrame(step)
   130→   130→  }, [target])
   131→   131→
   132→   132→  useEffect(() => {
   133→   133→    if (inView) animate()
   134→   134→    return () => {
   135→   135→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   136→   136→    }
   137→   137→  }, [inView, animate])
   138→   138→
   139→   139→  return value
   140→   140→}
