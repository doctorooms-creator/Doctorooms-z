    79→
    80→export default function ReceptionistSchedulePage() {
    81→  const queryClient = useQueryClient()
    82→
    83→  // Schedule query
    84→  const { data, isLoading } = useQuery<ScheduleData>({
    85→    queryKey: ['receptionist-schedule'],
    86→    queryFn: () => fetch('/api/dashboard/receptionist/schedule').then(r => r.json()),
    87→  })
    88→
    89→  // Holidays query (separate, includes past + future)

    91→    queryKey: ['receptionist-holidays'],
    92→    queryFn: () => fetch('/api/receptionist/holidays').then(r => r.json()),
    93→  })
    94→
    95→  // Booking days query
    96→  const { data: bookingDaysData } = useQuery<{ bookingDays: number }>({
    97→    queryKey: ['receptionist-booking-days'],
    98→    queryFn: () => fetch('/api/receptionist/booking-days').then(r => r.json()),
    99→  })
   100→
   101→  // State
   102→  const [addHolidayOpen, setAddHolidayOpen] = useState(false)
   103→  const [batchHolidayOpen, setBatchHolidayOpen] = useState(false)

   106→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(undefined)
   107→  const [remark, setRemark] = useState('')

   110→    { date: undefined, remark: '' },
   111→  ])

   115→    mutationFn: async (payload: { date: string; remark: string }) => {
   116→      const res = await fetch('/api/receptionist/holidays', {
   117→        method: 'POST',
   118→        headers: { 'Content-Type': 'application/json' },
   119→        body: JSON.stringify(payload),
   120→      })
   121→      const data = await res.json()

   124→    },

   126→      queryClient.invalidateQueries({ queryKey: ['receptionist-holidays'] })
   127→      queryClient.invalidateQueries({ queryKey: ['receptionist-schedule'] })

   129→      setAddHolidayOpen(false)
   130→      setSelectedDate(undefined)
   131→      setRemark('')

   138→    mutationFn: async (id: string) => {
   139→      const res = await fetch(`/api/receptionist/holidays/${id}`, { method: 'DELETE' })
   140→      const data = await res.json()
   141→      if (!res.ok) throw new Error(data.error || 'Failed to delete holiday')
   142→      return data
   143→    },
   144→    onSuccess: () => {
   145→      queryClient.invalidateQueries({ queryKey: ['receptionist-holidays'] })
   146→      queryClient.invalidateQueries({ queryKey: ['receptionist-schedule'] })

   157→        method: 'PUT',

   166→      queryClient.invalidateQueries({ queryKey: ['receptionist-booking-days'] })

   209→          if (res.ok) {
   210→            successCount++
   211→          } else {
   212→            failCount++

   216→          failCount++
   217→        })

   254→          {Array.from({ length: 7 }).map((_, i) => (
   255→            <div key={i} className="h-36 animate-pulse rounded-xl border border-border bg-card" />

   257→        </div>

   303→                setBookingDaysOpen(true)

   305→            >
   306→              <Pencil className="mr-1.5 h-3.5 w-3.5" />
   307→              Edit

   309→          </CardContent>

   312→

   319→              <motion.div

   356→                        'h-2 w-2 rounded-full',
   357→                        isToday ? 'bg-teal-500' : 'bg-emerald-500'

   389→                              {slot}
   390→                            </span>
   391→                          ))}
   392→                        </div>
   393→                      </div>
   394→                    )}
   395→                  </div>
   396→                </CardContent>
   397→              </Card>
   398→            </motion.div>

   405→        initial={{ opacity: 0, y: 12 }}

   411→          <div className="flex items-center gap-2">
   412→            <CalendarOff className="h-5 w-5 text-red-500" />

   528→          <div className="space-y-4 py-2">
