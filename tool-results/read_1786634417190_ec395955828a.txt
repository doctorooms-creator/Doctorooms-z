   558→function TrendIndicator({ current, previous }: { current: number; previous?: number }) {
   559→  if (!previous || previous === current) {
   560→    return <Minus className="h-3 w-3 text-muted-foreground" />
   561→  }
   562→  if (current > previous) {
   563→    return <ArrowUp className="h-3 w-3 text-amber-500" />
   564→  }
   565→  return <ArrowDown className="h-3 w-3 text-teal-500" />
   566→}

   656→    // I/O data
   657→    const ioData = recent
   658→      .filter((v) => v.inputMl > 0 || v.urineMl > 0 || v.outputMl > 0)
   659→      .map((v) => ({
   660→        time: v.recordedAt,
   661→        input: v.inputMl,
   662→        urine: v.urineMl,
   663→        otherOutput: Math.max(0, v.outputMl - v.urineMl),
   664→      }))
   665→
   666→    // Latest values
   667→    const latest = sorted.length > 0 ? sorted[sorted.length - 1] : null
   668→    const prev = sorted.length > 1 ? sorted[sorted.length - 2] : null
   669→
   670→    // Min/Max calculations
   671→    const bpSysVals = sorted.filter((v) => v.bpSystolic > 0).map((v) => v.bpSystolic)
   672→    const bpDiaVals = sorted.filter((v) => v.bpDiastolic > 0).map((v) => v.bpDiastolic)
   673→    const pulseVals = sorted.filter((v) => v.pulse > 0).map((v) => v.pulse)
   674→    const spo2Vals = sorted.filter((v) => v.spo2 > 0).map((v) => v.spo2)
   675→    const tempVals = sorted.filter((v) => v.temperature > 0).map((v) => v.temperature)
   676→
   677→    return {
   678→      bpData,
   679→      pulseData,
   680→      spo2Data,
   681→      tempData,
   682→      ioData,
   683→      latest,
   684→      prev,
   685→      stats: {
   686→        bpSysMin: bpSysVals.length > 0 ? Math.min(...bpSysVals) : undefined,
   687→        bpSysMax: bpSysVals.length > 0 ? Math.max(...bpSysVals) : undefined,
   688→        bpDiaMin: bpDiaVals.length > 0 ? Math.min(...bpDiaVals) : undefined,
   689→        bpDiaMax: bpDiaVals.length > 0 ? Math.max(...bpDiaVals) : undefined,
   690→        pulseMin: pulseVals.length > 0 ? Math.min(...pulseVals) : undefined,
   691→        pulseMax: pulseVals.length > 0 ? Math.max(...pulseVals) : undefined,
   692→        spo2Min: spo2Vals.length > 0 ? Math.min(...spo2Vals) : undefined,
   693→        spo2Max: spo2Vals.length > 0 ? Math.max(...spo2Vals) : undefined,
   694→        tempMin: tempVals.length > 0 ? Math.min(...tempVals) : undefined,
   695→        tempMax: tempVals.length > 0 ? Math.max(...tempVals) : undefined,
   696→      },
   697→      hasData: sorted.length >= 2,
   698→    }
   699→  }, [vitals])
   700→
   701→  if (!chartData.hasData) {
   702→    return (
   703→      <motion.div
   704→        initial={{ opacity: 0 }}
   705→        animate={{ opacity: 1 }}
   706→        transition={{ delay: 0.15 }}
   707→      >
   708→        <Card className="border-dashed">
   709→          <CardContent className="flex flex-col items-center justify-center py-10 text-center">
   710→            <Activity className="mb-2 h-8 w-8 text-muted-foreground/50" />
   711→            <p className="text-sm text-muted-foreground">At least 2 vital records are needed to display trend charts.</p>
   712→            <p className="mt-1 text-xs text-muted-foreground/70">Record vitals a few times to see trends.</p>
   713→          </CardContent>
   714→        </Card>
   715→      </motion.div>
   716→    )
   717→  }
   718→
   719→  const { latest, prev, stats, bpData, pulseData, spo2Data, tempData, ioData } = chartData
   720→
   721→  // Helper to determine status
   722→  const bpStatus = (() => {
   723→    if (!latest) return 'normal'
   724→    const T = VITAL_THRESHOLDS
   725→    if (latest.bpSystolic < T.bpSystolic.criticalLow || latest.bpSystolic > T.bpSystolic.criticalHigh) return 'critical'
   726→    if (latest.bpSystolic > T.bpSystolic.warningHigh) return 'warning'
   727→    return 'normal'
   728→  })()
   729→
   730→  const pulseStatus = (() => {
   731→    if (!latest) return 'normal'
   732→    const T = VITAL_THRESHOLDS
   733→    if (latest.pulse < T.pulse.criticalLow || latest.pulse > T.pulse.criticalHigh) return 'critical'
   734→    if (latest.pulse > T.pulse.warningHigh) return 'warning'
   735→    return 'normal'
   736→  })()
   737→
   738→  const spo2Status = (() => {
   739→    if (!latest) return 'normal'
   740→    const T = VITAL_THRESHOLDS
   741→    if (latest.spo2 < T.spo2.critical) return 'critical'
   742→    if (latest.spo2 < T.spo2.warning) return 'warning'
   743→    return 'normal'
   744→  })()
   745→
   746→  const tempStatus = (() => {
   747→    if (!latest) return 'normal'
   748→    const T = VITAL_THRESHOLDS
   749→    if (latest.temperature > T.temperature.warningHigh) return 'warning'
   750→    return 'normal'
   751→  })()
   752→
   753→  return (
   754→    <div className="space-y-4">
   755→      <div className="flex items-center gap-2 px-1">
   756→        <TrendingUp className="h-4 w-4 text-teal-500" />
   757→        <h3 className="text-sm font-semibold">Vital Trends</h3>
   758→        <Badge variant="outline" className="text-[10px] text-muted-foreground">
   759→          Last {chartData.ioData.length} readings
   760→        </Badge>
   761→      </div>
   762→
   763→      {/* 2-column grid on desktop, 1-column on mobile */}
   764→      <div className="grid gap-4 md:grid-cols-2">
   765→        {/* BP Chart */}
   766→        <ChartCard
   767→          title="Blood Pressure"
   768→          icon={<Droplets className="h-4 w-4 text-teal-500" />}
   769→          latestValue={latest ? `${latest.bpSystolic}/${latest.bpDiastolic} mmHg` : '—'}
   770→          latestRaw={latest?.bpSystolic}
   771→          prevRaw={prev?.bpSystolic}
   772→          minVal={stats.bpSysMin !== undefined ? `${stats.bpSysMin}/${stats.bpDiaMin}` : undefined}
   773→          maxVal={stats.bpSysMax !== undefined ? `${stats.bpSysMax}/${stats.bpDiaMax}` : undefined}
   774→          status={bpStatus}
   775→          delay={0}
   776→        >
   777→          {bpData.length >= 2 ? (
   778→            <BPChart data={bpData} />
   779→          ) : (
   780→            <p className="py-6 text-center text-xs text-muted-foreground">Not enough BP data</p>
   781→          )}
   782→        </ChartCard>
   783→
   784→        {/* Pulse Chart */}
   785→        <ChartCard
   786→          title="Pulse Rate"
   787→          icon={<Heart className="h-4 w-4 text-rose-500" />}
   788→          latestValue={latest ? `${latest.pulse} bpm` : '—'}
   789→          latestRaw={latest?.pulse}
   790→          prevRaw={prev?.pulse}
   791→          minVal={stats.pulseMin !== undefined ? `${stats.pulseMin} bpm` : undefined}
   792→          maxVal={stats.pulseMax !== undefined ? `${stats.pulseMax} bpm` : undefined}
   793→          status={pulseStatus}
   794→          delay={1}
   795→        >
   796→          {pulseData.length >= 2 ? (
   797→            <SingleLineChart
   798→              data={pulseData}
   799→              normalLow={60}
   800→              normalHigh={100}
   801→              criticalLow={50}
   802→              criticalHigh={130}
   803→              unit="bpm"
   804→              lineColor="#f43f5e"
   805→              gradientId="pulse-grad"
   806→            />
   807→          ) : (
   808→            <p className="py-6 text-center text-xs text-muted-foreground">Not enough pulse data</p>
   809→          )}
   810→        </ChartCard>
   811→
   812→        {/* SpO2 Chart */}
   813→        <ChartCard
   814→          title="SpO2"
   815→          icon={<Wind className="h-4 w-4 text-sky-500" />}
   816→          latestValue={latest ? `${latest.spo2}%` : '—'}
   817→          latestRaw={latest?.spo2}
   818→          prevRaw={prev?.spo2}
   819→          minVal={stats.spo2Min !== undefined ? `${stats.spo2Min}%` : undefined}
   820→          maxVal={stats.spo2Max !== undefined ? `${stats.spo2Max}%` : undefined}
   821→          status={spo2Status}
   822→          delay={2}
   823→        >
   824→          {spo2Data.length >= 2 ? (
   825→            <SingleLineChart
   826→              data={spo2Data}
   827→              normalLow={94}
   828→              normalHigh={100}
   829→              criticalLow={90}
   830→              redZoneBelow={90}
   831→              unit="%"
   832→              lineColor="#0ea5e9"
   833→              gradientId="spo2-grad"
   834→            />
   835→          ) : (
   836→            <p className="py-6 text-center text-xs text-muted-foreground">Not enough SpO2 data</p>
   837→          )}
   838→        </ChartCard>
   839→
   840→        {/* Temperature Chart */}
   841→        <ChartCard
   842→          title="Temperature"
   843→          icon={<Thermometer className="h-4 w-4 text-amber-500" />}
   844→          latestValue={latest ? `${latest.temperature}°F` : '—'}
   845→          latestRaw={latest?.temperature}
   846→          prevRaw={prev?.temperature}
   847→          minVal={stats.tempMin !== undefined ? `${stats.tempMin}°F` : undefined}
   848→          maxVal={stats.tempMax !== undefined ? `${stats.tempMax}°F` : undefined}
   849→          status={tempStatus}
   850→          delay={3}
   851→        >
   852→          {tempData.length >= 2 ? (
   853→            <SingleLineChart
   854→              data={tempData}
   855→              normalLow={97}
   856→              normalHigh={99.5}
   857→              feverLine={102.2}
   858→              unit="°F"
   859→              lineColor="#f59e0b"
   860→              gradientId="temp-grad"
   861→            />
   862→          ) : (
   863→            <p className="py-6 text-center text-xs text-muted-foreground">Not enough temperature data</p>
   864→          )}
   865→        </ChartCard>
   866→
   867→        {/* Input/Output Chart - spans full width */}
   868→        <ChartCard
   869→          title="Fluid Input / Output"
   870→          icon={<Droplets className="h-4 w-4 text-emerald-500" />}
   871→          latestValue={latest ? `I: ${latest.inputMl}ml / U: ${latest.urineMl}ml` : '—'}
   872→          status="normal"
   873→          delay={4}
   874→          className="md:col-span-2"
   875→        >
   876→          {ioData.length >= 2 ? (
   877→            <IOBarChart data={ioData} />
   878→          ) : (
   879→            <p className="py-6 text-center text-xs text-muted-foreground">Not enough I/O data</p>
   880→          )}
   881→        </ChartCard>
   882→      </div>
   883→    </div>
   884→  )
   885→}
