   658→function WeightTimeline({ visits }: { visits: Visit[] }) {
   659→  const weights = visits.filter((v) => v.weight > 0)
   660→  if (weights.length === 0) return null
   661→
   662→  const maxW = Math.max(...weights.map((v) => v.weight))
   663→  const minW = Math.min(...weights.map((v) => v.weight))
   664→  const range = maxW - minW || 1
   665→
   666→  return (
   667→    <Card>
   668→      <CardHeader className="pb-3">
   669→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   670→          <Dumbbell className="h-4 w-4 text-teal-600 dark:text-teal-400" />
   671→          Weight Trend (kg)
   672→        </CardTitle>
   673→      </CardHeader>
   674→      <CardContent className="space-y-3">
   675→        {weights.map((v) => {
   676→          const pct = ((v.weight - minW) / range) * 100
   677→          return (
   678→            <div key={v.id} className="flex items-center gap-3">
   679→              <span className="text-xs text-muted-foreground w-20 shrink-0">
   680→                {format(parseISO(v.bookingDate), 'MMM d')}
   681→              </span>
   682→              <div className="flex-1 h-6 bg-muted/50 rounded-full relative overflow-hidden">
   683→                <motion.div
   684→                  className="absolute inset-y-0 left-0 bg-gradient-to-r from-teal-500 to-teal-400 rounded-full"
   685→                  initial={{ width: 0 }}
   686→                  animate={{ width: `${Math.max(pct, 8)}%` }}
   687→                  transition={{ duration: 0.6, delay: 0.1 }}
   688→                />
   689→              </div>
   690→              <span className="text-xs font-semibold w-12 text-right">{v.weight}</span>
   691→            </div>
   692→          )
   693→        })}
   694→        <div className="flex justify-between text-[10px] text-muted-foreground pt-1">
   695→          <span>{minW} kg</span>
   696→          <span>{maxW} kg</span>
   697→        </div>
   698→      </CardContent>
   699→    </Card>
   700→  )
   701→}

   703→function BPTimeline({ visits }: { visits: Visit[] }) {
   704→  const bpData = visits
   705→    .filter((v) => v.prescriptions.some((p) => p.bp))
   706→    .map((v) => {
   707→      const rx = v.prescriptions.find((p) => p.bp)
   708→      return { ...v, bp: rx!.bp }
   709→    })
   710→
   711→  if (bpData.length === 0) return null
   712→
   713→  // Parse systolic values
   714→  const systolics = bpData
   715→    .map((v) => parseInt(v.bp.split('/')[0], 10))
   716→    .filter((n) => !isNaN(n))
   717→
   718→  if (systolics.length === 0) return null
   719→
   720→  const maxSys = Math.max(...systolics)
   721→  const minSys = Math.min(...systolics)
   722→  const range = maxSys - minSys || 1
   723→
   724→  return (
   725→    <Card>
   726→      <CardHeader className="pb-3">
   727→        <CardTitle className="text-sm font-semibold flex items-center gap-2">
   728→          <Heart className="h-4 w-4 text-red-500" />
   729→          Blood Pressure Trend (mmHg)
   730→        </CardTitle>
   731→      </CardHeader>
   732→      <CardContent className="space-y-3">
   733→        {bpData.map((v) => {
   734→          const sys = parseInt(v.bp.split('/')[0], 10) || 0
   735→          const pct = ((sys - minSys) / range) * 100
   736→          // Color based on BP level
   737→          const barColor = sys > 140 ? 'from-red-500 to-red-400' : sys > 120 ? 'from-amber-500 to-amber-400' : 'from-green-500 to-green-400'
   738→
   739→          return (
   740→            <div key={v.id} className="flex items-center gap-3">
   741→              <span className="text-xs text-muted-foreground w-20 shrink-0">
   742→                {format(parseISO(v.bookingDate), 'MMM d')}
   743→              </span>
   744→              <div className="flex-1 h-6 bg-muted/50 rounded-full relative overflow-hidden">
   745→                <motion.div
   746→                  className={cn('absolute inset-y-0 left-0 bg-gradient-to-r rounded-full', barColor)}
   747→                  initial={{ width: 0 }}
   748→                  animate={{ width: `${Math.max(pct, 8)}%` }}
   749→                  transition={{ duration: 0.6, delay: 0.1 }}
   750→                />
   751→              </div>
   752→              <span className="text-xs font-semibold w-16 text-right">{v.bp}</span>
   753→            </div>
   754→          )
   755→        })}
   756→        <div className="flex justify-between text-[10px] text-muted-foreground pt-1">
   757→          <span>{minSys} mmHg</span>
   758→          <span>{maxSys} mmHg</span>
   759→        </div>
   760→      </CardContent>
   761→    </Card>
   762→  )
   763→}

   784→          <Thermometer className="h-4 w-4 text-orange-500" />
   785→          Temperature Trend (°F)
   786→        </CardTitle>
   787→      </CardHeader>
   788→      <CardContent className="space-y-3">
   789→        {tempData.map((v) => {
   790→          const pct = ((v.temp - minT) / range) * 100
   791→          const barColor = v.temp > 99 ? 'from-red-500 to-red-400' : v.temp > 98 ? 'from-amber-500 to-amber-400' : 'from-green-500 to-green-400'
   792→
   793→          return (
   794→            <div key={v.id} className="flex items-center gap-3">
