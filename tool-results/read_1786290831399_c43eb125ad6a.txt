   102→function formatCurrency(amount: number): string {
   103→  return new Intl.NumberFormat('en-IN', {
   104→    style: 'currency',
   105→    currency: 'INR',
   106→    minimumFractionDigits: 0,
   107→  }).format(amount)
   108→}
   109→
   110→function formatTime(timeStr: string): string {
   111→  const [h, m] = timeStr.split(':').map(Number)
   112→  const period = h >= 12 ? 'PM' : 'AM'
   113→  const hour = h % 12 || 12
   114→  return `${hour}:${m.toString().padStart(2, '0')} ${period}`
   115→}
   116→
   117→function parseAwards(str: string): string[] {
   118→  if (!str) return []
   119→  try {
   120→    const parsed = JSON.parse(str)
   121→    if (Array.isArray(parsed)) return parsed.filter(Boolean)
   122→  } catch {
   123→    // not JSON, treat as newline/comma-separated
   124→    return str.split(/[\n,]/).map((s) => s.trim()).filter(Boolean)
   125→  }
   126→  return []
   127→}
