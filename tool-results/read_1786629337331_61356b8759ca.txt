   130→function formatTimeAgo(dateStr: string) {
   131→  const now = new Date()
   132→  const recorded = new Date(dateStr)
   133→  const diffMs = now.getTime() - recorded.getTime()
   134→  const diffMin = Math.floor(diffMs / 60000)
   135→  if (diffMin < 1) return 'Just now'
   136→  if (diffMin < 60) return `${diffMin}m ago`
   137→  const diffHr = Math.floor(diffMin / 60)
   138→  if (diffHr < 24) return `${diffHr}h ago`
   139→  return `${Math.floor(diffHr / 24)}d ago`
   140→}
   141→
   142→function getVitalTimeColor(dateStr: string) {
   143→  const now = new Date()
   144→  const recorded = new Date(dateStr)
   145→  const diffHr = (now.getTime() - recorded.getTime()) / 3600000
   146→  if (diffHr < 1) return 'text-emerald-500'
   147→  if (diffHr < 2) return 'text-amber-500'
   148→  return 'text-red-500'
   149→}

   151→function isAbnormal(param: string, value: number) {
   152→  const thresholds = {
   153→    spo2: { critical: 90, warning: 94 },
   154→    bpSystolic: { criticalLow: 90, criticalHigh: 180, warningHigh: 160 },
   155→    bpDiastolic: { criticalHigh: 120, warningHigh: 100 },
   156→    pulse: { criticalLow: 50, criticalHigh: 130, warningHigh: 110 },
   157→    temperature: { warningHigh: 102.2 },
   158→    respiratoryRate: { criticalLow: 10, criticalHigh: 30 },
   159→  } as const
   160→
   161→  const t = thresholds[param as keyof typeof thresholds]
   162→  if (!t) return false
   163→
   164→  if (param === 'spo2') return value < t.warning
   165→  if (param === 'bpSystolic') return value < (t as { criticalLow: number }).criticalLow || value > (t as { warningHigh: number }).warningHigh
   166→  if (param === 'bpDiastolic') return value > (t as { warningHigh: number }).warningHigh
   167→  if (param === 'pulse') return value < (t as { criticalLow: number }).criticalLow || value > (t as { warningHigh: number }).warningHigh
   168→  if (param === 'temperature') return value > (t as { warningHigh: number }).warningHigh
   169→  if (param === 'respiratoryRate') return value < (t as { criticalLow: number }).criticalLow || value > (t as { criticalHigh: number }).criticalHigh
   170→  return false
   171→}

   175→function VitalMini({ vital, alerts }: { vital: LatestVital; alerts: VitalAlert[] }) {
   176→  const alertParams = new Set(alerts.map((a) => a.parameter))
   177→
   178→  return (
   179→    <div className="mt-3 grid grid-cols-2 gap-x-3 gap-y-1.5">
   180→      <VitalPill
   181→        icon={<Thermometer className="h-3 w-3" />}
   182→        label="Temp"
   183→        value={vital.temperature > 0 ? `${vital.temperature.toFixed(1)}°F` : '—'}
   184→        abnormal={isAbnormal('temperature', vital.temperature) || alertParams.has('Temperature')}
   185→      />
   186→      <VitalPill
   187→        icon={<Heart className="h-3 w-3" />}
   188→        label="Pulse"
   189→        value={vital.pulse > 0 ? `${vital.pulse}` : '—'}
   190→        unit="bpm"
   191→        abnormal={isAbnormal('pulse', vital.pulse) || alertParams.has('Pulse')}
   192→      />
   193→      <VitalPill
   194→        icon={<Droplets className="h-3 w-3" />}
   195→        label="BP"
   196→        value={vital.bpSystolic > 0 ? `${vital.bpSystolic}/${vital.bpDiastolic}` : '—'}
   197→        unit="mmHg"
   198→        abnormal={isAbnormal('bpSystolic', vital.bpSystolic) || alertParams.has('BP')}
   199→      />
   200→      <VitalPill
   201→        icon={<Wind className="h-3 w-3" />}
   202→        label="SpO2"
   203→        value={vital.spo2 > 0 ? `${Math.round(vital.spo2)}` : '—'}
   204→        unit="%"
   205→        abnormal={isAbnormal('spo2', vital.spo2) || alertParams.has('SpO2')}
   206→      />
   207→    </div>
   208→  )
   209→}
