   227→function isAbnormal(param: string, value: number): boolean {
   228→  const T = VITAL_THRESHOLDS as Record<string, Record<string, number>>
   229→  const t = T[param]
   230→  if (!t) return false
   231→  if (param === 'spo2') return value < t.warning
   232→  if (param === 'bpSystolic') return value < (t.criticalLow ?? 999) || value > (t.warningHigh ?? 999)
   233→  if (param === 'bpDiastolic') return value > (t.warningHigh ?? 999)
   234→  if (param === 'pulse') return value < (t.criticalLow ?? 999) || value > (t.warningHigh ?? 999)
   235→  if (param === 'temperature') return value > (t.warningHigh ?? 999)
   236→  if (param === 'respiratoryRate') return value < (t.criticalLow ?? 999) || value > (t.criticalHigh ?? 999)
   237→  return false
   238→}

   286→function TrendArrow({ current, previous, param }: { current: number; previous: number; param: string }) {
   287→  if (!previous || previous === 0) return null
   288→  const diff = current - previous
   289→  if (Math.abs(diff) === 0) return null
   290→  // For SpO2: up is good, down is bad
   291→  // For BP systolic/pulse/temp: up or down can be bad depending on thresholds
   292→  const isWorse = (param === 'spo2' && diff < 0) || (param === 'spo2' && diff > 0) === false
   293→  const isWorseActual = (param === 'spo2' && diff < 0) || (param !== 'spo2' && Math.abs(diff) > 0)
   294→  return (
   295→    <span className={cn('text-xs', isWorseActual && isAbnormal(param, current) ? 'text-red-500' : 'text-muted-foreground')}>
   296→      {diff > 0 ? <ArrowUp className="inline h-3 w-3" /> : <ArrowDown className="inline h-3 w-3" />}
   297→    </span>
   298→  )
   299→}
