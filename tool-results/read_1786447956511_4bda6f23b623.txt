   107→function formatFileSize(bytes: number) {
   108→  if (!bytes || bytes === 0) return null
   109→  const k = 1024
   110→  const sizes = ['B', 'KB', 'MB', 'GB']
   111→  const i = Math.floor(Math.log(bytes) / Math.log(k))
   112→  return `${parseFloat((bytes / Math.pow(k, i)).toFixed(1))} ${sizes[i]}`
   113→}
