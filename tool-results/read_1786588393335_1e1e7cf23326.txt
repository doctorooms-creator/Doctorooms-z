   537→                key={item.id}
   538→                initial={{ opacity: 0, x: -8 }}
   539→                animate={{ opacity: 1, x: 0 }}
   540→                transition={{ delay: i * 0.03 }}
   541→                className={cn(
   542→                  'flex items-center gap-3 px-4 py-3 border-b border-border last:border-b-0 transition-colors hover:bg-muted/40',
   543→                  item.status === 'Visited' && 'bg-teal-50/50 dark:bg-teal-950/20',
   544→                  item.status === 'Finish' && 'opacity-60',
   545→                )}
   546→              >
   547→                {/* Token number badge */}
   548→                <Badge
   549→                  className={cn(
   550→                    'shrink-0 font-mono text-xs px-2.5 py-0.5 border-0 min-w-[3.5rem] justify-center',

   574→                    {item.bookingType !== 'By Self' && (
   575→                      <Badge variant="outline" className="text-[9px] px-1 py-0 border-dashed">

   577→                      </Badge>
   578→                    )}
   579→                  </div>
   580→                </div>
   581→
   582→                <Badge
   583→                  className={cn(
   584→                    'text-[10px] px-1.5 py-0 shrink-0',
   585→                    statusColors[item.status] || 'bg-gray-100 text-gray-700'
   586→                  )}
   587→                >
   588→                  {item.status === 'Visited' ? 'In Progress' : item.status}
   589→                </Badge>
   590→              </motion.div>
   591→            ))}
   592→          </div>
   593→        </CardContent>
   594→      </Card>
   595→
   596→      {/* Next Patient Button (visual only for now) */}
   597→      {waitingItems.length > 0 && (
   598→        <div className="flex justify-center">
   599→          <Button
   600→            className="bg-teal-600 hover:bg-teal-700 text-white px-8"
   601→            disabled={inConsultationItems.length > 0}
   602→          >
   603→            <UserPlus className="h-4 w-4 mr-2" />
   604→            {inConsultationItems.length > 0
   605→              ? 'Patient in consultation'
   606→              : 'Call Next Patient'
   607→            }
   608→          </Button>
   609→        </div>
   610→      )}
   611→    </motion.div>
   612→  )
   613→}
