   162→export default function WardsClient() {
   163→  const queryClient = useQueryClient()
   164→  const [selectedHospital, setSelectedHospital] = useState<string>('all')
   165→
   166→  // Dialogs
   167→  const [viewWard, setViewWard] = useState<WardDetail | null>(null)
   168→  const [wardDialog, setWardDialog] = useState<{ open: boolean; edit: WardItem | null }>({ open: false, edit: null })
   169→  const [deleteWard, setDeleteWard] = useState<WardItem | null>(null)
   170→  const [bedDialog, setBedDialog] = useState<{ open: boolean; edit: BedItem | null; wardId: string }>({ open: false, edit: null, wardId: '' })
   171→  const [deleteBed, setDeleteBed] = useState<BedItem | null>(null)
   172→
   173→  // Ward form
   174→  const [wardForm, setWardForm] = useState({ hospitalId: '', name: '', nameHi: '', wardType: 'General', floorNo: '', totalBeds: '0', nurseRatio: '6' })
   175→  // Bed form
   176→  const [bedForm, setBedForm] = useState({ bedNumber: '', bedType: 'General', dailyRate: '' })
   177→
   178→  // ─── Queries ──────────────────────────────────────────────────────
   179→
   180→  const { data: hospitalsData } = useQuery<{ hospitals: HospitalOption[] }>({
   181→    queryKey: ['admin-hospitals-list'],
   182→    queryFn: () => fetch('/api/dashboard/admin/hospitals').then((r) => r.json()),
   183→    staleTime: 5 * 60 * 1000,
   184→  })
   185→
   186→  const hospitals = hospitalsData?.hospitals || []
   187→
   188→  const { data: wardsData, isLoading } = useQuery<{ wards: WardItem[]; total: number }>({
   189→    queryKey: ['admin-wards', selectedHospital],
   190→    queryFn: () => {
   191→      const params = selectedHospital !== 'all' ? `?hospitalId=${selectedHospital}` : ''
   192→      return fetch(`/api/dashboard/admin/wards${params}`).then((r) => r.json())
   193→    },
   194→  })
   195→
   196→  const wards = wardsData?.wards || []
   197→
   198→  const { data: wardDetail, isLoading: isDetailLoading } = useQuery<{ ward: WardDetail }>({
   199→    queryKey: ['admin-ward-detail', viewWard?.id],
   200→    queryFn: () => fetch(`/api/dashboard/admin/wards/${viewWard!.id}`).then((r) => r.json()),
   201→    enabled: !!viewWard?.id,
   202→  })
   203→
   204→  // When detail loads, update viewWard with full data
   205→  const detailWard = wardDetail?.ward || viewWard
   206→
   207→  // ─── Mutations ────────────────────────────────────────────────────
   208→
   209→  const createWard = useMutation({
   210→    mutationFn: (data: typeof wardForm) =>
   211→      fetch('/api/dashboard/admin/wards', {
   212→        method: 'POST',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({
   215→          hospitalId: data.hospitalId,
   216→          name: data.name,
   217→          nameHi: data.nameHi,
   218→          wardType: data.wardType,
   219→          floorNo: data.floorNo,
   220→          totalBeds: parseInt(data.totalBeds) || 0,
   221→          nurseRatio: parseInt(data.nurseRatio) || 6,
   222→        }),
   223→      }).then((r) => r.json()),
   224→    onSuccess: () => {
   225→      queryClient.invalidateQueries({ queryKey: ['admin-wards'] })
   226→      setWardDialog({ open: false, edit: null })
   227→      toast.success('Ward created successfully')
   228→    },
   229→    onError: (err) => toast.error(err.message || 'Failed to create ward'),
   230→  })
   231→
   232→  const updateWard = useMutation({
   233→    mutationFn: ({ id, data }: { id: string; data: Partial<typeof wardForm> }) =>
   234→      fetch(`/api/dashboard/admin/wards/${id}`, {
   235→        method: 'PUT',
   236→        headers: { 'Content-Type': 'application/json' },
   237→        body: JSON.stringify({
   238→          ...(data.name !== undefined && { name: data.name }),
   239→          ...(data.nameHi !== undefined && { nameHi: data.nameHi }),
   240→          ...(data.wardType !== undefined && { wardType: data.wardType }),
   241→          ...(data.floorNo !== undefined && { floorNo: data.floorNo }),
   242→          ...(data.totalBeds !== undefined && { totalBeds: parseInt(data.totalBeds) || 0 }),
   243→          ...(data.nurseRatio !== undefined && { nurseRatio: parseInt(data.nurseRatio) || 6 }),
   244→        }),
   245→      }).then((r) => r.json()),
   246→    onSuccess: () => {
   247→      queryClient.invalidateQueries({ queryKey: ['admin-wards'] })
   248→      queryClient.invalidateQueries({ queryKey: ['admin-ward-detail'] })
   249→      setWardDialog({ open: false, edit: null })
   250→      toast.success('Ward updated successfully')
   251→    },
   252→    onError: (err) => toast.error(err.message || 'Failed to update ward'),
   253→  })
   254→
   255→  const deleteWardMut = useMutation({
   256→    mutationFn: (id: string) =>
   257→      fetch(`/api/dashboard/admin/wards/${id}`, { method: 'DELETE' }).then((r) => r.json()),
   258→    onSuccess: () => {
   259→      queryClient.invalidateQueries({ queryKey: ['admin-wards'] })
   260→      setDeleteWard(null)
   261→      setViewWard(null)
   262→      toast.success('Ward deleted successfully')
   263→    },
   264→    onError: (err) => toast.error(err.error || 'Failed to delete ward'),
   265→  })
   266→
   267→  const createBed = useMutation({
   268→    mutationFn: ({ wardId, data }: { wardId: string; data: typeof bedForm }) =>
   269→      fetch(`/api/dashboard/admin/wards/${wardId}/beds`, {
   270→        method: 'POST',
   271→        headers: { 'Content-Type': 'application/json' },
   272→        body: JSON.stringify({
   273→          bedNumber: data.bedNumber,
   274→          bedType: data.bedType,
   275→          dailyRate: parseFloat(data.dailyRate) || 0,
   276→        }),
   277→      }).then((r) => r.json()),
   278→    onSuccess: () => {
   279→      queryClient.invalidateQueries({ queryKey: ['admin-ward-detail'] })
   280→      queryClient.invalidateQueries({ queryKey: ['admin-wards'] })
   281→      setBedDialog({ open: false, edit: null, wardId: '' })
   282→      toast.success('Bed added successfully')
   283→    },
   284→    onError: (err) => toast.error(err.error || 'Failed to add bed'),
   285→  })
   286→
   287→  const updateBed = useMutation({
   288→    mutationFn: ({ id, data }: { id: string; data: Partial<typeof bedForm> & { status?: string } }) =>
   289→      fetch(`/api/dashboard/admin/wards/beds/${id}`, {
   290→        method: 'PUT',
   291→        headers: { 'Content-Type': 'application/json' },
   292→        body: JSON.stringify(data),
   293→      }).then((r) => r.json()),
   294→    onSuccess: () => {
   295→      queryClient.invalidateQueries({ queryKey: ['admin-ward-detail'] })
   296→      queryClient.invalidateQueries({ queryKey: ['admin-wards'] })
   297→      setBedDialog({ open: false, edit: null, wardId: '' })
   298→      toast.success('Bed updated successfully')
   299→    },
   300→    onError: (err) => toast.error(err.error || 'Failed to update bed'),
   301→  })
   302→
   303→  const deleteBedMut = useMutation({
   304→    mutationFn: (id: string) =>
   305→      fetch(`/api/dashboard/admin/wards/beds/${id}`, { method: 'DELETE' }).then((r) => r.json()),
   306→    onSuccess: () => {
   307→      queryClient.invalidateQueries({ queryKey: ['admin-ward-detail'] })
   308→      queryClient.invalidateQueries({ queryKey: ['admin-wards'] })
   309→      setDeleteBed(null)
   310→      toast.success('Bed deleted successfully')
   311→    },
   312→    onError: (err) => toast.error(err.error || 'Failed to delete bed'),
   313→  })
   314→
   315→  // ─── Handlers ─────────────────────────────────────────────────────
   316→
   317→  const openCreateWard = () => {
   318→    setWardForm({
   319→      hospitalId: selectedHospital !== 'all' ? selectedHospital : (hospitals[0]?.id || ''),
   320→      name: '',
   321→      nameHi: '',
   322→      wardType: 'General',
   323→      floorNo: '',
   324→      totalBeds: '0',
   325→      nurseRatio: '6',
   326→    })
   327→    setWardDialog({ open: true, edit: null })
   328→  }
   329→
   330→  const openEditWard = (w: WardItem) => {
   331→    setWardForm({
   332→      hospitalId: w.hospitalId,
   333→      name: w.name,
   334→      nameHi: w.nameHi,
   335→      wardType: w.wardType,
   336→      floorNo: w.floorNo,
   337→      totalBeds: String(w.totalBeds),
   338→      nurseRatio: String(w.nurseRatio),
   339→    })
   340→    setWardDialog({ open: true, edit: w })
   341→  }
   342→
   343→  const openAddBed = (wardId: string) => {
   344→    setBedForm({ bedNumber: '', bedType: 'General', dailyRate: '' })
   345→    setBedDialog({ open: true, edit: null, wardId })
   346→  }
   347→
   348→  const openEditBed = (b: BedItem) => {
   349→    setBedForm({ bedNumber: b.bedNumber, bedType: b.bedType, dailyRate: String(b.dailyRate) })
   350→    setBedDialog({ open: true, edit: b, wardId: b.wardId })
   351→  }
   352→
   353→  const handleWardSubmit = () => {
   354→    if (!wardForm.name.trim() || !wardForm.hospitalId) {
   355→      toast.error('Hospital and ward name are required')
   356→      return
   357→    }
   358→    if (wardDialog.edit) {
   359→      updateWard.mutate({ id: wardDialog.edit.id, data: wardForm })
   360→    } else {
   361→      createWard.mutate(wardForm)
   362→    }
   363→  }
   364→
   365→  const handleBedSubmit = () => {
   366→    if (!bedForm.bedNumber.trim()) {
   367→      toast.error('Bed number is required')
   368→      return
   369→    }
   370→    if (bedDialog.edit) {
   371→      updateBed.mutate({ id: bedDialog.edit.id, data: bedForm })
   372→    } else {
   373→      createBed.mutate({ wardId: bedDialog.wardId, data: bedForm })
   374→    }
   375→  }
   376→
   377→  // ─── Computed stats ───────────────────────────────────────────────
   378→
   379→  const totalBeds = wards.reduce((s, w) => s + (w.bedCounts?.total || 0), 0)
   380→  const totalOccupied = wards.reduce((s, w) => s + (w.bedCounts?.occupied || 0), 0)
   381→  const totalAvailable = wards.reduce((s, w) => s + (w.bedCounts?.available || 0), 0)
   382→  const totalNurses = wards.reduce((s, w) => s + w.nurseCount, 0)
   383→
   384→  return (
   385→    <div className="space-y-6">
   386→      {/* Header with hospital selector */}
   387→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   388→        <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:gap-3">
   389→          <div className="flex items-center gap-2">
   390→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/50">
   391→              <BedDouble className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   392→            </div>
   393→            <h1 className="text-lg font-semibold">Wards Management</h1>
   394→          </div>
   395→          <Select value={selectedHospital} onValueChange={setSelectedHospital}>
   396→            <SelectTrigger className="w-full sm:w-56">
   397→              <Building2 className="mr-2 h-4 w-4 text-muted-foreground" />
   398→              <SelectValue placeholder="All Hospitals" />
   399→            </SelectTrigger>
   400→            <SelectContent>
   401→              <SelectItem value="all">All Hospitals</SelectItem>
   402→              {hospitals.map((h) => (
   403→                <SelectItem key={h.id} value={h.id}>
   404→                  {h.hospitalName}
   405→                </SelectItem>
   406→              ))}
   407→            </SelectContent>
   408→          </Select>
   409→        </div>
   410→        <Button onClick={openCreateWard} className="gap-2 bg-teal-600 hover:bg-teal-700">
   411→          <Plus className="h-4 w-4" />
   412→          Add Ward
   413→        </Button>
   414→      </div>
   415→
   416→      {/* Stat cards */}
   417→      <div className="grid gap-4 grid-cols-2 lg:grid-cols-4">
   418→        <Card>
   419→          <CardContent className="p-4">
   420→            <div className="flex items-center gap-3">
   421→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/50">
   422→                <BedDouble className="h-5 w-5 text-amber-600 dark:text-amber-400" />
   423→              </div>
   424→              <div>
   425→                <p className="text-2xl font-bold">{wards.length}</p>
   426→                <p className="text-xs text-muted-foreground">Total Wards</p>
   427→              </div>
   428→            </div>
   429→          </CardContent>
   430→        </Card>
   431→        <Card>
   432→          <CardContent className="p-4">
   433→            <div className="flex items-center gap-3">
   434→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-teal-100 dark:bg-teal-900/50">
   435→                <Bed className="h-5 w-5 text-teal-600 dark:text-teal-400" />
   436→              </div>
   437→              <div>
   438→                <p className="text-2xl font-bold">{totalBeds}</p>
   439→                <p className="text-xs text-muted-foreground">Total Beds</p>
   440→              </div>
   441→            </div>
   442→          </CardContent>
   443→        </Card>
   444→        <Card>
   445→          <CardContent className="p-4">
   446→            <div className="flex items-center gap-3">
   447→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/50">
   448→                <Activity className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   449→              </div>
   450→              <div>
   451→                <p className="text-2xl font-bold">
   452→                  <span className="text-emerald-600">{totalAvailable}</span>
   453→                  <span className="text-muted-foreground text-sm"> / {totalOccupied} occ.</span>
   454→                </p>
   455→                <p className="text-xs text-muted-foreground">Available / Occupied</p>
   456→              </div>
   457→            </div>
   458→          </CardContent>
   459→        </Card>
   460→        <Card>
   461→          <CardContent className="p-4">
   462→            <div className="flex items-center gap-3">
   463→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-violet-100 dark:bg-violet-900/50">
   464→                <Users className="h-5 w-5 text-violet-600 dark:text-violet-400" />
   465→              </div>
   466→              <div>
   467→                <p className="text-2xl font-bold">{totalNurses}</p>
   468→                <p className="text-xs text-muted-foreground">Nurses Assigned</p>
   469→              </div>
   470→            </div>
   471→          </CardContent>
   472→        </Card>
   473→      </div>
   474→
   475→      {/* Wards grid */}
   476→      {isLoading ? (
   477→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   478→          {Array.from({ length: 6 }).map((_, i) => (
   479→            <Card key={i}>
   480→              <CardContent className="p-4">
   481→                <div className="space-y-3">
   482→                  <div className="h-5 w-28 animate-pulse rounded bg-muted" />
   483→                  <div className="h-4 w-20 animate-pulse rounded bg-muted" />
   484→                  <div className="flex gap-2">
   485→                    <div className="h-6 w-16 animate-pulse rounded-full bg-muted" />
   486→                    <div className="h-6 w-16 animate-pulse rounded bg-muted" />
   487→                  </div>
   488→                </div>
   489→              </CardContent>
   490→            </Card>
   491→          ))}
   492→        </div>
   493→      ) : wards.length === 0 ? (
   494→        <Card>
   495→          <CardContent className="flex flex-col items-center justify-center py-16">
   496→            <BedDouble className="mb-3 h-12 w-12 text-muted-foreground/30" />
   497→            <p className="text-sm font-medium text-muted-foreground">No wards found</p>
   498→            <p className="mt-1 text-xs text-muted-foreground/70">Create a ward to get started</p>
   499→          </CardContent>
   500→        </Card>
   501→      ) : (
   502→        <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   503→          <AnimatePresence>
   504→            {wards.map((ward, i) => (
   505→              <motion.div
   506→                key={ward.id}
   507→                initial={{ opacity: 0, y: 10 }}
   508→                animate={{ opacity: 1, y: 0 }}
   509→                exit={{ opacity: 0 }}
   510→                transition={{ delay: i * 0.04 }}
   511→              >
   512→                <Card
   513→                  className="group cursor-pointer transition-all hover:shadow-md hover:border-teal-200 dark:hover:border-teal-800"
   514→                  onClick={() => {
   515→                    setViewWard(ward)
   516→                  }}
   517→                >
   518→                  <CardContent className="p-4">
   519→                    <div className="flex items-start justify-between gap-2">
