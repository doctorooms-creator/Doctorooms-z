   283→                                                redirect("hospital/dashboard", "refresh");

   285→                                                redirect("admin/dashboard", "refresh");
   286→                                        } elseif ($ud['role'] == DOCTOR_ASSISTANT) {
   287→                                                redirect("assistant/dashboard", "refresh");

   289→                                                redirect("pharmacist/pharmacist/listPrescriptions", "refresh");
   290→                                        }
   291→                                }
