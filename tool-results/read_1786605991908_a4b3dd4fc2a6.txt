   313→/** Small step indicator dot */
   314→function StepDot({ label, active, done }: { label: string; active: boolean; done: boolean }) {
   315→  return (
   316→    <div className="flex flex-col items-center gap-1">
   317→      <motion.div
   318→        animate={done ? { scale: [1, 1.2, 1] } : active ? { scale: [1, 1.1, 1] } : {}}
   319→        transition={{ repeat: active ? Infinity : 0, duration: 1.5 }}
   320→        className={cn(
   321→          'h-3 w-3 rounded-full border-2 transition-colors',
   322→          done
   323→            ? 'bg-violet-500 border-violet-500'
   324→            : active
   325→              ? 'bg-violet-200 border-violet-500 dark:bg-violet-800'
   326→              : 'bg-transparent border-muted-foreground/30'
   327→        )}
   328→      />
   329→      <span className={cn(
   330→        'text-[10px] font-medium transition-colors',
   331→        done ? 'text-violet-600 dark:text-violet-400' : active ? 'text-foreground' : 'text-muted-foreground'
   332→      )}>
   333→        {label}
   334→      </span>
   335→    </div>
   336→  )
   337→}
