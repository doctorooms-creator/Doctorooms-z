    51→// ============ Types ============
    52→
    53→interface ChargeCategory {
    54→  id: string
    55→  name: string
    56→  description: string
    57→  isTaxable: boolean
    58→  taxPercent: number
    59→  status: string
    60→  sortOrder: number
    61→  chargeItemCount: number
    62→  createdAt: string
    63→  updatedAt: string
    64→}
    65→
    66→interface ChargeItem {
    67→  id: string
    68→  categoryId: string
    69→  name: string
    70→  shortCode: string
    71→  unitType: string
    72→  rate: number
    73→  isTaxable: boolean
    74→  taxPercent: number
    75→  status: string
    76→  categoryName: string
    77→  createdAt: string
    78→  updatedAt: string
    79→}
    80→
    81→interface CategoryFormData {
    82→  name: string
    83→  description: string
    84→  isTaxable: boolean
    85→  taxPercent: number

    87→
    88→interface ChargeItemFormData {
    89→  categoryId: string
    90→  name: string
    91→  shortCode: string
    92→  unitType: string
    93→  rate: number
    94→  isTaxable: boolean
    95→  taxPercent: number
    96→}
    97→

    99→  name: '',
   100→  description: '',
   101→  isTaxable: false,
   102→  taxPercent: 0,
   103→}
   104→
   105→const EMPTY_ITEM_FORM: ChargeItemFormData = {
   106→  categoryId: '',
   107→  name: '',
   108→  shortCode: '',
   109→  unitType: 'Per Day',
   110→  rate: 0,
   111→  isTaxable: false,
   112→  taxPercent: 0,
   113→}
   114→
   115→const UNIT_TYPES = ['Per Day', 'Per Hour', 'Per Session', 'Per Unit', 'Per Visit', 'Flat', 'Per Kg', 'Per Liter']

   117→// ============ Animation Variants ============

   121→  show: {

   124→  },
   125→}
   126→
   127→const itemVariants = {
   128→  hidden: { opacity: 0, y: 16 },
   129→  show: { opacity: 1, y: 0 },
   130→}

   133→  'bg-rose-100 dark:bg-rose-900/40 text-rose-600 dark:text-rose-400',

   143→// ============ Main Component ============
   144→
   145→export default function ChargeMasterClient() {
   146→  const queryClient = useQueryClient()
   147→  const [activeTab, setActiveTab] = useState('categories')
   148→
   149→  // --- Category Dialog State ---
   150→  const [catDialogOpen, setCatDialogOpen] = useState(false)
   151→  const [editingCategory, setEditingCategory] = useState<ChargeCategory | null>(null)
   152→  const [catForm, setCatForm] = useState<CategoryFormData>(EMPTY_CATEGORY_FORM)
   153→
   154→  // --- Item Dialog State ---
   155→  const [itemDialogOpen, setItemDialogOpen] = useState(false)
   156→  const [editingItem, setEditingItem] = useState<ChargeItem | null>(null)
   157→  const [itemForm, setItemForm] = useState<ChargeItemFormData>(EMPTY_ITEM_FORM)
   158→
   159→  // --- Delete Dialog State ---
   160→  const [deleteDialogOpen, setDeleteDialogOpen] = useState(false)
   161→  const [deletingTarget, setDeletingTarget] = useState<{ id: string; name: string; type: 'category' | 'item' } | null>(null)
   162→
   163→  // --- Filter State for Items ---
   164→  const [filterCategory, setFilterCategory] = useState<string>('all')
   165→  const [searchQuery, setSearchQuery] = useState('')
   166→
   167→  // ============ Queries ============
   168→
   169→  const {
   170→    data: categoriesData,
   171→    isLoading: categoriesLoading,
   172→  } = useQuery<{ categories: ChargeCategory[] }>({
   173→    queryKey: ['charge-categories'],
   174→    queryFn: () => fetch('/api/charge-categories').then((r) => r.json()),
   175→  })
   176→
   177→  const {
   178→    data: itemsData,
   179→    isLoading: itemsLoading,
   180→  } = useQuery<{ chargeItems: ChargeItem[] }>({
   181→    queryKey: ['charge-items', filterCategory, searchQuery],
   182→    queryFn: () => {
   183→      const params = new URLSearchParams()
   184→      if (filterCategory && filterCategory !== 'all') params.set('categoryId', filterCategory)
   185→      if (searchQuery) params.set('search', searchQuery)
   186→      const qs = params.toString()
   187→      return fetch(`/api/charge-items${qs ? `?${qs}` : ''}`).then((r) => r.json())
   188→    },
   189→  })
   190→
   191→  const categories = categoriesData?.categories || []
   192→  const chargeItems = itemsData?.chargeItems || []
   193→
   194→  // ============ Mutations ============
   195→
   196→  // --- Category Mutations ---
   197→  const createCategoryMutation = useMutation({
   198→    mutationFn: (payload: CategoryFormData) =>
   199→      fetch('/api/charge-categories', {
   200→        method: 'POST',
   201→        headers: { 'Content-Type': 'application/json' },
   202→        body: JSON.stringify(payload),
   203→      }).then((r) => r.json()),
   204→    onSuccess: () => {
   205→      queryClient.invalidateQueries({ queryKey: ['charge-categories'] })
   206→      toast.success('Category created successfully')
   207→      closeCatDialog()
   208→    },
   209→    onError: (err: { error?: string }) => {
   210→      toast.error(err.error || 'Failed to create category')
   211→    },
   212→  })
   213→
   214→  const updateCategoryMutation = useMutation({
   215→    mutationFn: ({ id, payload }: { id: string; payload: Partial<CategoryFormData> }) =>
   216→      fetch(`/api/charge-categories/${id}`, {
   217→        method: 'PUT',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify(payload),
   220→      }).then((r) => r.json()),
   221→    onSuccess: () => {
   222→      queryClient.invalidateQueries({ queryKey: ['charge-categories'] })
   223→      toast.success('Category updated successfully')
   224→      closeCatDialog()
   225→    },
   226→    onError: (err: { error?: string }) => {
   227→      toast.error(err.error || 'Failed to update category')
   228→    },
   229→  })
   230→
   231→  const deleteCategoryMutation = useMutation({
   232→    mutationFn: (id: string) =>
   233→      fetch(`/api/charge-categories/${id}`, { method: 'DELETE' }).then((r) => r.json()),
   234→    onSuccess: () => {
   235→      queryClient.invalidateQueries({ queryKey: ['charge-categories'] })
   236→      queryClient.invalidateQueries({ queryKey: ['charge-items'] })
   237→      toast.success('Category deactivated successfully')
   238→      closeDeleteDialog()
   239→    },
   240→    onError: (err: { error?: string }) => {
   241→      toast.error(err.error || 'Failed to deactivate category')
   242→    },
   243→  })
   244→
   245→  // --- Item Mutations ---
   246→  const createItemMutation = useMutation({
   247→    mutationFn: (payload: ChargeItemFormData) =>
   248→      fetch('/api/charge-items', {
   249→        method: 'POST',
   250→        headers: { 'Content-Type': 'application/json' },
   251→        body: JSON.stringify(payload),
   252→      }).then((r) => r.json()),
   253→    onSuccess: () => {
   254→      queryClient.invalidateQueries({ queryKey: ['charge-items'] })
   255→      queryClient.invalidateQueries({ queryKey: ['charge-categories'] })
   256→      toast.success('Charge item created successfully')
   257→      closeItemDialog()
   258→    },
   259→    onError: (err: { error?: string }) => {
   260→      toast.error(err.error || 'Failed to create charge item')
   261→    },
   262→  })
   263→
   264→  const updateItemMutation = useMutation({
   265→    mutationFn: ({ id, payload }: { id: string; payload: Partial<ChargeItemFormData> }) =>
   266→      fetch(`/api/charge-items/${id}`, {
   267→        method: 'PUT',
   268→        headers: { 'Content-Type': 'application/json' },
   269→        body: JSON.stringify(payload),
   270→      }).then((r) => r.json()),
   271→    onSuccess: () => {
   272→      queryClient.invalidateQueries({ queryKey: ['charge-items'] })
   273→      toast.success('Charge item updated successfully')
   274→      closeItemDialog()
   275→    },
   276→    onError: (err: { error?: string }) => {
   277→      toast.error(err.error || 'Failed to update charge item')
   278→    },
   279→  })
   280→
   281→  const deleteItemMutation = useMutation({
   282→    mutationFn: (id: string) =>
   283→      fetch(`/api/charge-items/${id}`, { method: 'DELETE' }).then((r) => r.json()),
   284→    onSuccess: () => {
   285→      queryClient.invalidateQueries({ queryKey: ['charge-items'] })
   286→      queryClient.invalidateQueries({ queryKey: ['charge-categories'] })
   287→      toast.success('Charge item deactivated successfully')
   288→      closeDeleteDialog()
   289→    },
   290→    onError: (err: { error?: string }) => {
   291→      toast.error(err.error || 'Failed to deactivate charge item')
   292→    },
   293→  })
   294→
   295→  // ============ Handlers ============
   296→
   297→  // --- Category Dialog Handlers ---
   298→  function openCreateCatDialog() {
   299→    setEditingCategory(null)
   300→    setCatForm(EMPTY_CATEGORY_FORM)
   301→    setCatDialogOpen(true)
   302→  }
   303→
   304→  function openEditCatDialog(cat: ChargeCategory) {
   305→    setEditingCategory(cat)
   306→    setCatForm({
   307→      name: cat.name,
   308→      description: cat.description,
   309→      isTaxable: cat.isTaxable,
   310→      taxPercent: cat.taxPercent,
   311→    })
   312→    setCatDialogOpen(true)
   313→  }
   314→
   315→  function closeCatDialog() {
   316→    setCatDialogOpen(false)
   317→    setEditingCategory(null)
   318→    setCatForm(EMPTY_CATEGORY_FORM)
   319→  }
   320→
   321→  function handleCatSubmit(e: React.FormEvent) {
   322→    e.preventDefault()
   323→    if (!catForm.name.trim()) {
   324→      toast.error('Category name is required')
   325→      return
   326→    }
   327→    if (editingCategory) {
   328→      updateCategoryMutation.mutate({ id: editingCategory.id, payload: catForm })
   329→    } else {
   330→      createCategoryMutation.mutate(catForm)
   331→    }
   332→  }
   333→
   334→  // --- Item Dialog Handlers ---
   335→  function openCreateItemDialog() {
   336→    setEditingItem(null)
   337→    setItemForm(EMPTY_ITEM_FORM)
   338→    if (filterCategory && filterCategory !== 'all') {
   339→      setItemForm((f) => ({ ...f, categoryId: filterCategory }))
   340→    }
   341→    setItemDialogOpen(true)
   342→  }
   343→
   344→  function openEditItemDialog(item: ChargeItem) {
   345→    setEditingItem(item)
   346→    setItemForm({
   347→      categoryId: item.categoryId,
   348→      name: item.name,
   349→      shortCode: item.shortCode,
   350→      unitType: item.unitType,
   351→      rate: item.rate,
   352→      isTaxable: item.isTaxable,
   353→      taxPercent: item.taxPercent,
   354→    })
   355→    setItemDialogOpen(true)
   356→  }
   357→
   358→  function closeItemDialog() {
   359→    setItemDialogOpen(false)
   360→    setEditingItem(null)
   361→    setItemForm(EMPTY_ITEM_FORM)
   362→  }
   363→
   364→  function handleItemSubmit(e: React.FormEvent) {
   365→    e.preventDefault()
   366→    if (!itemForm.categoryId) {
   367→      toast.error('Please select a category')
   368→      return
   369→    }
   370→    if (!itemForm.name.trim()) {
   371→      toast.error('Item name is required')
   372→      return
   373→    }
   374→    if (editingItem) {
   375→      updateItemMutation.mutate({ id: editingItem.id, payload: itemForm })
   376→    } else {
   377→      createItemMutation.mutate(itemForm)
   378→    }
   379→  }
   380→
   381→  // --- Delete Dialog Handlers ---
   382→  function openDeleteDialog(target: { id: string; name: string; type: 'category' | 'item' }) {
   383→    setDeletingTarget(target)
   384→    setDeleteDialogOpen(true)
   385→  }
   386→
   387→  function closeDeleteDialog() {
   388→    setDeleteDialogOpen(false)
   389→    setDeletingTarget(null)
   390→  }
   391→
   392→  function confirmDelete() {
   393→    if (!deletingTarget) return
   394→    if (deletingTarget.type === 'category') {
   395→      deleteCategoryMutation.mutate(deletingTarget.id)
   396→    } else {
   397→      deleteItemMutation.mutate(deletingTarget.id)
   398→    }
   399→  }
   400→
   401→  const isCatSaving = createCategoryMutation.isPending || updateCategoryMutation.isPending
   402→  const isItemSaving = createItemMutation.isPending || updateItemMutation.isPending
   403→
   404→  // ============ Render ============
   405→
   406→  return (
   407→    <div className="space-y-6">
   408→      {/* Header */}
   409→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   410→        <div>
   411→          <h1 className="text-2xl font-bold tracking-tight">Charge Master</h1>
   412→          <p className="text-sm text-muted-foreground">
   413→            Manage charge categories and items for billing
   414→          </p>
   415→        </div>
   416→      </div>
   417→
   418→      {/* Tabs */}
   419→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   420→        <TabsList className="grid w-full max-w-md grid-cols-2">
   421→          <TabsTrigger value="categories" className="gap-2">
   422→            <Tags className="h-4 w-4" />
   423→            Categories
   424→          </TabsTrigger>
   425→          <TabsTrigger value="items" className="gap-2">
   426→            <Receipt className="h-4 w-4" />
   427→            Charge Items
   428→          </TabsTrigger>
   429→        </TabsList>
   430→
   431→        {/* ============ Categories Tab ============ */}
   432→        <TabsContent value="categories" className="mt-6 space-y-6">
   433→          <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   434→            <p className="text-sm text-muted-foreground">
   435→              {categories.length} {categories.length === 1 ? 'category' : 'categories'} found
   436→            </p>
   437→            <Button onClick={openCreateCatDialog} className="shrink-0 bg-teal-600 hover:bg-teal-700">
   438→              <Plus className="mr-2 h-4 w-4" />
   439→              Add Category
   440→            </Button>
   441→          </div>
   442→
   443→          {/* Loading Skeleton */}
   444→          {categoriesLoading && <CategoriesSkeleton />}
   445→
   446→          {/* Empty State */}
   447→          {!categoriesLoading && categories.length === 0 && (
   448→            <motion.div
   449→              initial={{ opacity: 0, y: 20 }}
   450→              animate={{ opacity: 1, y: 0 }}
   451→              className="flex flex-col items-center justify-center rounded-xl border border-dashed border-border py-16"
   452→            >
   453→              <div className="mb-4 flex h-16 w-16 items-center justify-center rounded-2xl bg-teal-50 dark:bg-teal-900/30">
   454→                <Tags className="h-8 w-8 text-teal-500" />
   455→              </div>
   456→              <h3 className="mb-1 text-lg font-semibold">No categories yet</h3>
   457→              <p className="mb-6 max-w-sm text-center text-sm text-muted-foreground">
   458→                Create your first charge category to organize billing items.
   459→              </p>
   460→              <Button onClick={openCreateCatDialog} className="bg-teal-600 hover:bg-teal-700">
   461→                <Plus className="mr-2 h-4 w-4" />
   462→                Add Category
   463→              </Button>
   464→            </motion.div>
   465→          )}
   466→
   467→          {/* Category Cards Grid */}
   468→          {!categoriesLoading && categories.length > 0 && (
   469→            <motion.div
   470→              className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-4"
   471→              variants={containerVariants}
   472→              initial="hidden"
   473→              animate="show"
   474→            >
   475→              {categories.map((cat, index) => (
   476→                <motion.div key={cat.id} variants={itemVariants}>
   477→                  <Card className="group relative overflow-hidden transition-all duration-200 hover:shadow-lg hover:-translate-y-0.5">
   478→                    <CardHeader className="pb-3">
   479→                      <div className="flex items-start justify-between">
   480→                        <div className={`flex h-10 w-10 items-center justify-center rounded-xl ${CARD_COLORS[index % CARD_COLORS.length]}`}>
   481→                          <Tags className="h-5 w-5" />
   482→                        </div>
   483→                        <div className="flex items-center gap-1.5">
   484→                          <Badge
   485→                            className={`text-[10px] font-medium ${
   486→                              cat.status === 'Active'
   487→                                ? 'bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-400'
   488→                                : 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400'
   489→                            }`}
   490→                          >
   491→                            {cat.status}
   492→                          </Badge>
   493→                        </div>
   494→                      </div>
   495→                      <CardTitle className="mt-2 text-base">{cat.name}</CardTitle>
   496→                    </CardHeader>
   497→                    <CardContent className="pt-0">
   498→                      {cat.description && (
   499→                        <p className="mb-3 line-clamp-2 text-sm text-muted-foreground">
   500→                          {cat.description}
   501→                        </p>
   502→                      )}
   503→                      <div className="flex items-center justify-between">
   504→                        <div className="flex items-center gap-3 text-xs text-muted-foreground">
   505→                          <span className="flex items-center gap-1">
   506→                            <Receipt className="h-3.5 w-3.5" />
   507→                            {cat.chargeItemCount} {cat.chargeItemCount === 1 ? 'item' : 'items'}
   508→                          </span>
   509→                          {cat.isTaxable && (
   510→                            <span className="flex items-center gap-1">
   511→                              Tax: {cat.taxPercent}%
   512→                            </span>
   513→                          )}
   514→                        </div>
   515→                        <div className="flex items-center gap-1">
   516→                          <Button
   517→                            variant="ghost"
   518→                            size="icon"
   519→                            className="h-8 w-8"
   520→                            onClick={() => openEditCatDialog(cat)}
   521→                          >
   522→                            <Pencil className="h-3.5 w-3.5" />
   523→                          </Button>
   524→                          <Button
   525→                            variant="ghost"
   526→                            size="icon"
   527→                            className="h-8 w-8 text-red-500 hover:text-red-600"
   528→                            onClick={() => openDeleteDialog({ id: cat.id, name: cat.name, type: 'category' })}
   529→                          >
   530→                            <Trash2 className="h-3.5 w-3.5" />
   531→                          </Button>
   532→                        </div>
   533→                      </div>
   534→                    </CardContent>
   535→                  </Card>
   536→                </motion.div>
   537→              ))}
   538→            </motion.div>
   539→          )}
   540→        </TabsContent>
   541→
   542→        {/* ============ Charge Items Tab ============ */}
   543→        <TabsContent value="items" className="mt-6 space-y-6">
   544→          {/* Filters */}
   545→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   546→            <div className="flex flex-1 flex-col gap-3 sm:flex-row sm:items-center">
   547→              <div className="relative flex-1 sm:max-w-xs">
   548→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   549→                <Input
   550→                  placeholder="Search items..."
   551→                  value={searchQuery}
   552→                  onChange={(e) => setSearchQuery(e.target.value)}
   553→                  className="pl-9"
   554→                />
   555→              </div>
   556→              <Select value={filterCategory} onValueChange={setFilterCategory}>
   557→                <SelectTrigger className="w-full sm:w-[180px]">
   558→                  <SelectValue placeholder="All Categories" />
   559→                </SelectTrigger>
   560→                <SelectContent>
   561→                  <SelectItem value="all">All Categories</SelectItem>
   562→                  {categories.map((cat) => (
   563→                    <SelectItem key={cat.id} value={cat.id}>
   564→                      {cat.name}
   565→                    </SelectItem>
   566→                  ))}
   567→                </SelectContent>
   568→              </Select>
   569→            </div>
   570→            <div className="flex items-center gap-3">
   571→              <p className="hidden text-sm text-muted-foreground sm:block">
   572→                {chargeItems.length} {chargeItems.length === 1 ? 'item' : 'items'}
   573→              </p>
   574→              <Button onClick={openCreateItemDialog} className="shrink-0 bg-teal-600 hover:bg-teal-700">
   575→                <Plus className="mr-2 h-4 w-4" />
   576→                Add Item
   577→              </Button>
   578→            </div>
