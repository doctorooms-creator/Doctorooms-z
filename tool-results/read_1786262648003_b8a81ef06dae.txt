    60→function FadeUpSection({
    61→  children,
    62→  className,
    63→  delay = 0,
    64→}: {
    65→  children: React.ReactNode
    66→  className?: string
    67→  delay?: number
    68→}) {
    69→  const ref = useRef<HTMLDivElement>(null)
    70→  const inView = useInView(ref, { once: true, margin: '-60px' })
    71→  return (
    72→    <motion.div
    73→      ref={ref}
    74→      initial={{ opacity: 0, y: 30 }}
    75→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    76→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    77→      className={className}
    78→    >
    79→      {children}
    80→    </motion.div>
    81→  )
    82→}

   108→// ── Count-up hook ────────────────────────────────────────────────────────────
   109→
   110→function useCountUp(target: number, inView: boolean) {
   111→  const [value, setValue] = useState(0)
   112→  const rafRef = useRef<number | null>(null)
   113→
   114→  const animate = useCallback(() => {
   115→    const duration = 1800
   116→    const start = performance.now()
   117→    const from = 0
   118→    const to = target
   119→
   120→    const step = (now: number) => {
   121→      const elapsed = now - start
   122→      const progress = Math.min(elapsed / duration, 1)
   123→      const eased = 1 - Math.pow(1 - progress, 3)
   124→      setValue(Math.floor(from + (to - from) * eased))
   125→      if (progress < 1) {
   126→        rafRef.current = requestAnimationFrame(step)
   127→      }
   128→    }
   129→    rafRef.current = requestAnimationFrame(step)
   130→  }, [target])
   131→
   132→  useEffect(() => {
   133→    if (inView) animate()
   134→    return () => {
   135→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   136→    }
   137→  }, [inView, animate])
   138→
   139→  return value
   140→}

   270→function StarRating({ rating = 4.8 }: { rating?: number }) {
   271→  const fullStars = Math.floor(rating)
   272→  const hasHalf = rating - fullStars >= 0.5
   273→  return (
   274→    <div className="flex items-center gap-0.5">
   275→      {Array.from({ length: 5 }).map((_, i) => (
   276→        <Star
   277→          key={i}
   278→          className={`h-4 w-4 ${i < fullStars || (i === fullStars && hasHalf) ? 'fill-amber-400 text-amber-400' : 'text-gray-200 dark:text-gray-700'}`}
   279→        />
   280→      ))}
   281→      <span className="ml-1 text-sm font-medium text-muted-foreground">{rating}</span>
   282→    </div>
   283→  )
   284→}

   727→function StatsGrid({ stats }: { stats: Stats }) {
   728→  const ref = useRef<HTMLDivElement>(null)
   729→  const inView = useInView(ref, { once: true, margin: '100px' })
   730→  const [delayPassed, setDelayPassed] = useState(false)
   731→
   732→  useEffect(() => {
   733→    const id = setTimeout(() => setDelayPassed(true), 800)
   734→    return () => clearTimeout(id)
   735→  }, [])
   736→
   737→  const shouldAnimate = inView || delayPassed
   738→
   739→  const doctorCount = useCountUp(stats.doctorCount, shouldAnimate)
   740→  const hospitalCount = useCountUp(stats.hospitalCount, shouldAnimate)
   741→  const patientCount = useCountUp(stats.patientCount, shouldAnimate)
   742→  const bookingCount = useCountUp(stats.bookingCount, shouldAnimate)
   743→
   744→  const items = [
   745→    { icon: Stethoscope, value: doctorCount, label: 'Verified Doctors', suffix: '+' },
   746→    { icon: Building2, value: hospitalCount, label: 'Partner Hospitals', suffix: '+' },

   748→    { icon: CalendarCheck, value: bookingCount, label: 'Appointments Booked', suffix: '+' },
   749→  ]
   750→
   751→  return (
   752→    <div ref={ref} className="grid grid-cols-2 gap-6 lg:grid-cols-4">
   753→      {items.map((item, i) => (
   754→        <FadeUpItem key={item.label} index={i} className="flex flex-col items-center text-center">
