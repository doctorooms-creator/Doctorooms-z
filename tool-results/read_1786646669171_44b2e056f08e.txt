   141→function statusBadge(status: string) {
   142→  switch (status) {
   143→    case 'Admitted':
   144→      return 'bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-900/30 dark:text-amber-300 dark:border-amber-800'
   145→    case 'Discharged':
   146→      return 'bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-900/30 dark:text-emerald-300 dark:border-emerald-800'
   147→    case 'DAMA':
   148→      return 'bg-orange-100 text-orange-800 border-orange-200 dark:bg-orange-900/30 dark:text-orange-300 dark:border-orange-800'
   149→    case 'Expired':
   150→      return 'bg-red-100 text-red-800 border-red-200 dark:bg-red-900/30 dark:text-red-300 dark:border-red-800'
   151→    case 'Transferred':
   152→      return 'bg-sky-100 text-sky-800 border-sky-200 dark:bg-sky-900/30 dark:text-sky-300 dark:border-sky-800'
   153→    default:
   154→      return 'bg-gray-100 text-gray-800 border-gray-200'
   155→  }
   156→}

   158→function formatDate(isoString: string) {
   159→  const d = new Date(isoString)
   160→  return d.toLocaleDateString('en-IN', {
   161→    day: '2-digit',
   162→    month: 'short',
   163→    year: 'numeric',
   164→  })
   165→}
