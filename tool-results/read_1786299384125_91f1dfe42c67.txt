   108→// ── Count-up hook ────────────────────────────────────────────────────────────
   109→
   110→function useCountUp(target: number, inView: boolean) {
   111→  const [value, setValue] = useState(0)
   112→  const rafRef = useRef<number | null>(null)
   113→
   114→  const animate = useCallback(() => {
   115→    const duration = 1800
   116→    const start = performance.now()
   117→    const from = 0
   118→    const to = target
   119→
   120→    const step = (now: number) => {
   121→      const elapsed = now - start
   122→      const progress = Math.min(elapsed / duration, 1)
   123→      const eased = 1 - Math.pow(1 - progress, 3)
   124→      setValue(Math.floor(from + (to - from) * eased))
   125→      if (progress < 1) {
   126→        rafRef.current = requestAnimationFrame(step)
   127→      }
   128→    }
   129→    rafRef.current = requestAnimationFrame(step)
   130→  }, [target])
   131→
   132→  useEffect(() => {
   133→    if (inView) animate()
   134→    return () => {
   135→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   136→    }
   137→  }, [inView, animate])
   138→
   139→  return value
   140→}

   738→function StatsGrid({ stats }: { stats: Stats }) {
   739→  const ref = useRef<HTMLDivElement>(null)
   740→  const inView = useInView(ref, { once: true, margin: '100px' })
   741→  const [delayPassed, setDelayPassed] = useState(false)
   742→
   743→  useEffect(() => {
   744→    const id = setTimeout(() => setDelayPassed(true), 800)
   745→    return () => clearTimeout(id)
   746→  }, [])
   747→
   748→  const shouldAnimate = inView || delayPassed
   749→
   750→  const doctorCount = useCountUp(stats.doctorCount, shouldAnimate)
   751→  const hospitalCount = useCountUp(stats.hospitalCount, shouldAnimate)
   752→  const patientCount = useCountUp(stats.patientCount, shouldAnimate)
   753→  const bookingCount = useCountUp(stats.bookingCount, shouldAnimate)
   754→
   755→  const items = [
   756→    { icon: Stethoscope, value: doctorCount, label: 'Verified Doctors', suffix: '+' },
   757→    { icon: Building2, value: hospitalCount, label: 'Partner Hospitals', suffix: '+' },
   758→    { icon: Users, value: patientCount, label: 'Happy Patients', suffix: '+' },
   759→    { icon: CalendarCheck, value: bookingCount, label: 'Appointments Booked', suffix: '+' },
   760→  ]
   761→
   762→  return (
   763→    <div ref={ref} className="grid grid-cols-2 gap-6 lg:grid-cols-4">
   764→      {items.map((item, i) => (
   765→        <FadeUpItem key={item.label} index={i} className="flex flex-col items-center text-center">

   768→          </div>

   771→            {item.suffix}
   772→          </p>

   774→          <div className="mt-2 h-1 w-8 rounded-full bg-teal-500" />
