    88→export default function PatientBookDoctorPage() {
    89→  const params = useParams()
    90→  const router = useRouter()
    91→  const queryClient = useQueryClient()
    92→  const doctorId = params.doctorId as string
    93→
    94→  // State
    95→  const [selectedDate, setSelectedDate] = useState<Date | undefined>(undefined)
    96→  const [selectedSlot, setSelectedSlot] = useState<string | null>(null)
    97→  const [bookingMode, setBookingMode] = useState<'InPerson' | 'VideoCall'>('InPerson')
    98→  const [disease, setDisease] = useState('')
    99→  const [description, setDescription] = useState('')
   100→  const [showBookingForm, setShowBookingForm] = useState(false)
   101→  const [slotStatuses, setSlotStatuses] = useState<SlotAvailability[]>([])
   102→
   103→  // ── Fetch doctor info ──
   104→  const {
   105→    data: doctorData,
   106→    isLoading: doctorLoading,
   107→  } = useQuery({
   108→    queryKey: ['doctor-info', doctorId],
   109→    queryFn: () => fetch(`/api/doctors/${doctorId}`).then((r) => r.json()),
   110→    enabled: !!doctorId,
   111→  })
   112→
   113→  const doctor: DoctorInfo | null = doctorData?.doctor || null
   114→
   115→  // ── Fetch doctor schedule (public, no auth) ──
   116→  const {
   117→    data: scheduleData,
   118→    isLoading: scheduleLoading,
   119→  } = useQuery<ScheduleData>({
   120→    queryKey: ['doctor-schedule', doctorId],
   121→    queryFn: () => fetch(`/api/doctors/${doctorId}/schedule`).then((r) => r.json()),
   122→    enabled: !!doctorId,
   123→  })

   125→  const fees = doctor?.doctor?.fees ?? scheduleData?.fees ?? 0
   126→

   128→  const scheduleForSelectedDay = useMemo(() => {
   129→    if (!selectedDate || !scheduleData?.schedules) return null
   130→    const dayName = format(selectedDate, 'EEEE')
   131→    return scheduleData.schedules.find((s) => s.day === dayName) || null
   132→  }, [selectedDate, scheduleData])
   133→
   134→  const timeSlots = useMemo(() => {
   135→    if (!scheduleForSelectedDay) return []
   136→    return scheduleForSelectedDay.timeSlots
   137→  }, [scheduleForSelectedDay])
   138→
   139→  // ── Check if selected date is a holiday ──
   140→  const isHoliday = useMemo(() => {
   141→    if (!selectedDate || !scheduleData?.holidays) return false
   142→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   143→    return scheduleData.holidays.some((h) => h.date === dateStr)
   144→  }, [selectedDate, scheduleData])
   145→

   147→  const checkSlotAvailabilities = useCallback(async () => {
   148→    if (!selectedDate || timeSlots.length === 0) {
   149→      setSlotStatuses([])
   150→      return
   151→    }
   152→
   153→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   154→    const currentDoctorId = scheduleData?.doctorId
   155→    const initial: SlotAvailability[] = timeSlots.map((slot) => ({
   156→      slot,
   157→      available: true,
   158→      loading: true,
   159→    }))
   160→    setSlotStatuses(initial)
   161→
   162→    // Check all slots in parallel
   163→    const results = await Promise.all(
   164→      timeSlots.map(async (slot) => {
   165→        try {
   166→          const res = await fetch(
   167→            `/api/patient/bookings/check-slot?doctorId=${currentDoctorId}&date=${dateStr}&timeSlot=${encodeURIComponent(slot)}`
   168→          )
   169→          const data = await res.json()
   170→          return { slot, available: data.available !== false, loading: false }
   171→        } catch {
   172→          return { slot, available: true, loading: false }
   173→        }
   174→      })
   175→    )
   176→
   177→    setSlotStatuses(results)
   178→  }, [selectedDate, timeSlots, scheduleData])

   184→      setSelectedSlot(null)

   187→        // Small delay to allow scheduleForSelectedDay to compute

   190→        setSlotStatuses([])

   192→    },

   200→        method: 'POST',
   201→        headers: { 'Content-Type': 'application/json' },
   202→        body: JSON.stringify(body),
   203→      }).then((r) => {
   204→        if (!r.ok) return r.json().then((d) => Promise.reject(new Error(d.error || 'Booking failed')))
   205→        return r.json()

   210→      queryClient.invalidateQueries({ queryKey: ['patient-stats'] })

   212→    },
   213→    onError: (err: Error) => {
   214→      toast.error(err.message || 'Failed to book appointment')
   215→    },
   216→  })

   228→      toast.error('Please fill in all required fields')
   229→      return
   230→    }
   231→
   232→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   233→    bookMutation.mutate({
   234→      doctorId: scheduleData?.doctorId || doctorId,
   235→      bookingDate: dateStr,
   236→      timeSlot: selectedSlot,
   237→      bookingMode,
   238→      disease: disease.trim(),
   239→      description: description.trim(),
   240→    })
   241→  }
   242→
   243→  // ── Calendar disabled dates ──
   244→  const disablePastDates = (date: Date) => {
   245→    return isBefore(startOfDay(date), startOfDay(new Date()))
   246→  }
   247→
   248→  const disableNonScheduleDays = (date: Date) => {
   249→    if (!scheduleData?.schedules) return false // Allow all if no schedule loaded yet
   250→    const dayName = format(date, 'EEEE')
   251→    const hasSchedule = scheduleData.schedules.some((s) => s.day === dayName)
   252→    return !hasSchedule
   253→  }
   254→
   255→  // ── Loading state ──
   256→  if (doctorLoading || scheduleLoading) {
   257→    return <BookingPageSkeleton />

   259→
   260→  if (!doctor) {
   261→    return (
   262→      <div className="flex flex-col items-center justify-center py-20 text-center">
   263→        <AlertCircle className="h-12 w-12 text-muted-foreground/40 mb-4" />

   271→        </Button>

   275→
   276→  const doc = doctor.doctor

   278→    .split(' ')
   279→    .map((n) => n[0])
   280→    .join('')
   281→    .slice(0, 2)
   282→    .toUpperCase()

   284→  return (
   285→    <div className="max-w-4xl mx-auto space-y-6">
   286→      {/* Breadcrumb */}
   287→      <motion.div {...fadeIn} className="flex items-center gap-2 text-sm text-muted-foreground">

   290→        <Link href="/dashboard/patient/appointments" className="hover:text-teal-600 transition-colors">Appointments</Link>

   295→      {/* Doctor Info Card */}
   296→      <motion.div {...fadeIn} transition={{ delay: 0.05 }}>

   298→          <div className="bg-gradient-to-r from-teal-500 to-emerald-500 h-2" />
   299→          <CardContent className="p-5 md:p-6">
   300→            <div className="flex items-center gap-4">
   301→              <Avatar className="h-16 w-16 border-2 border-teal-200 dark:border-teal-800 shrink-0">
   302→                <AvatarImage src={doctor.profileImg !== 'default.png' ? doctor.profileImg : ''} />
   303→                <AvatarFallback className="bg-gradient-to-br from-teal-100 to-teal-200 text-teal-700 font-bold text-xl">
   304→                  {initials}
   305→                </AvatarFallback>
   306→              </Avatar>
   307→              <div className="flex-1 min-w-0">
   308→                <div className="flex items-center gap-2 flex-wrap">
   309→                  <h2 className="text-xl font-bold">{doctor.name}</h2>
   310→                  <ShieldCheck className="h-5 w-5 text-teal-500 shrink-0" />
   311→                  {doc?.isEmergency && (

   317→                </p>
   318→                <div className="flex items-center gap-4 mt-1.5 text-xs text-muted-foreground flex-wrap">

   321→                  <span className="flex items-center gap-1 font-semibold text-teal-700 dark:text-teal-400">
   322→                    <IndianRupee className="h-3 w-3" />{fees} consultation

   325→              </div>
   326→              <Button variant="outline" size="icon" className="shrink-0 h-8 w-8" asChild>
   327→                <Link href={`/doctors/${doctorId}`}>
   328→                  <ChevronLeft className="h-4 w-4" />
   329→                  <span className="sr-only">Back to profile</span>

   331→              </Button>
   332→            </div>
   333→          </CardContent>
   334→        </Card>

   336→
   337→      {/* Main booking area */}
   338→      <div className="grid gap-6 lg:grid-cols-5">
   339→        {/* Left: Calendar + Slots */}
   340→        <div className="lg:col-span-3 space-y-6">
   341→          {/* Step 1: Select Date */}
   342→          <motion.div {...fadeIn} transition={{ delay: 0.1 }}>
   343→            <Card>
   344→              <CardHeader className="pb-3">
   345→                <CardTitle className="text-base font-semibold flex items-center gap-2">
   346→                  <CalendarDays className="h-5 w-5 text-teal-500" />
   347→                  <span className="flex items-center gap-2">

   358→                    onSelect={handleDateSelect}

   364→                      day_disabled: 'text-muted-foreground/30 line-through',

   369→                {/* Holiday notice */}

   389→          {/* Step 2: Select Slot + Mode */}
