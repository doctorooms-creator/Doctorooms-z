    96→

   100→  const [bookingMode, setBookingMode] = useState<'InPerson' | 'VideoCall'>('InPerson')
   101→  const [disease, setDisease] = useState('')
   102→  const [description, setDescription] = useState('')
   103→  const [bookingState, setBookingState] = useState('')
   104→  const [bookingCity, setBookingCity] = useState('')
   105→  const [bookingGender, setBookingGender] = useState('')

   115→    queryKey: ['doctor-info', doctorId],
   116→    queryFn: () => fetch(`/api/doctors/${doctorId}`).then((r) => r.json()),

   118→  })

   120→  const doctor: DoctorInfo | null = doctorData?.doctor || null
   121→

   123→  const {
   124→    data: scheduleData,
   125→    isLoading: scheduleLoading,
   126→  } = useQuery<ScheduleData>({
   127→    queryKey: ['doctor-schedule', doctorId],
   128→    queryFn: () => fetch(`/api/doctors/${doctorId}/schedule`).then((r) => r.json()),

   133→

   135→  const scheduleForSelectedDay = useMemo(() => {
   136→    if (!selectedDate || !scheduleData?.schedules) return null
   137→    const dayName = format(selectedDate, 'EEEE')
   138→    return scheduleData.schedules.find((s) => s.day === dayName) || null
   139→  }, [selectedDate, scheduleData])
   140→
   141→  const timeSlots = useMemo(() => {
   142→    if (!scheduleForSelectedDay) return []
   143→    return scheduleForSelectedDay.timeSlots

   154→  const handleDateSelect = useCallback(
   155→    (date: Date | undefined) => {

   157→      setSelectedSlot(null)
   158→      setShowBookingForm(false)
   159→      if (!date) {
   160→        setSlotStatuses([])

   168→    if (!selectedDate || timeSlots.length === 0 || !scheduleData?.doctorId) {
   169→      return
   170→    }
   171→
   172→    const dateStr = format(selectedDate, 'yyyy-MM-dd')
   173→    const currentDoctorId = scheduleData.doctorId
   174→    const currentSlots = [...timeSlots]
   175→    let cancelled = false
   176→
   177→    fetch(`/api/patient/bookings/slots-availability?doctorId=${currentDoctorId}&date=${dateStr}`)
   178→      .then((r) => r.json())
   179→      .then((data) => {
   180→        if (cancelled) return
   181→        if (data.available === false) {
   182→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: false, loading: false })))

   185→          setSlotStatuses(
   186→            currentSlots.map((slot) => ({
   187→              slot,
   188→              available: !bookedSet.has(slot),
   189→              loading: false,
   190→            }))
   191→          )
   192→        }
   193→      })

   195→        if (!cancelled) {
   196→          setSlotStatuses(currentSlots.map((slot) => ({ slot, available: true, loading: false })))

   198→      })

   200→    return () => {
   201→      cancelled = true
   202→    }

   217→        method: 'POST',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify(body),

   221→        if (!r.ok) return r.json().then((d) => Promise.reject(new Error(d.error || 'Booking failed')))
   222→        return r.json()

   249→    const selectedGender = bookingGender || user?.gender || 'Male'

   252→      doctorId: scheduleData?.doctorId || doctorId,
   253→      bookingDate: dateStr,
   254→      timeSlot: selectedSlot,
   255→      bookingMode,
   256→      gender: selectedGender,
   257→      disease: disease.trim(),
   258→      description: description.trim(),
   259→      state: bookingState.trim(),
   260→      city: bookingCity.trim(),
   261→    })
   262→  }
   263→
   264→  // ── Calendar disabled dates ──
   265→  const disablePastDates = (date: Date) => {
   266→    return isBefore(startOfDay(date), startOfDay(new Date()))
   267→  }
   268→
   269→  const disableNonScheduleDays = (date: Date) => {
   270→    if (!scheduleData?.schedules) return false // Allow all if no schedule loaded yet
   271→    const dayName = format(date, 'EEEE')
   272→    const hasSchedule = scheduleData.schedules.some((s) => s.day === dayName)
   273→    return !hasSchedule
   274→  }
   275→
   276→  // ── Loading state ──
   277→  if (doctorLoading || scheduleLoading) {
   278→    return <BookingPageSkeleton />

   297→  const doc = doctor.doctor
   298→  const initials = doctor.name
   299→    .split(' ')
   300→    .map((n) => n[0])

   302→    .slice(0, 2)

   309→        <Link href="/dashboard/patient" className="hover:text-teal-600 transition-colors">Dashboard</Link>
   310→        <span>/</span>
   311→        <Link href="/dashboard/patient/appointments" className="hover:text-teal-600 transition-colors">Appointments</Link>

   315→
   316→      {/* Doctor Info Card */}
   317→      <motion.div {...fadeIn} transition={{ delay: 0.05 }}>

   324→                <AvatarFallback className="bg-gradient-to-br from-teal-100 to-teal-200 text-teal-700 font-bold text-xl">
   325→                  {initials}

   327→              </Avatar>
   328→              <div className="flex-1 min-w-0">

   330→                  <h2 className="text-xl font-bold">{doctor.name}</h2>

   344→                  </span>
   345→                </div>
   346→              </div>
   347→              <Button variant="outline" size="icon" className="shrink-0 h-8 w-8" asChild>
   348→                <Link href={`/doctors/${doctorId}`}>
   349→                  <ChevronLeft className="h-4 w-4" />

   352→              </Button>
   353→            </div>
   354→          </CardContent>
   355→        </Card>

   358→      {/* Main booking area */}
