   111→   111→function useCountUp(target: number, inView: boolean) {
   112→   112→  const [value, setValue] = useState(0)
   113→   113→  const rafRef = useRef<number | null>(null)
   114→   114→
   115→   115→  const animate = useCallback(() => {
   116→   116→    const duration = 1800
   117→   117→    const start = performance.now()
   118→   118→    const from = 0
   119→   119→    const to = target
   120→   120→
   121→   121→    const step = (now: number) => {
   122→   122→      const elapsed = now - start
   123→   123→      const progress = Math.min(elapsed / duration, 1)
   124→   124→      const eased = 1 - Math.pow(1 - progress, 3)
   125→   125→      setValue(Math.floor(from + (to - from) * eased))
   126→   126→      if (progress < 1) {
   127→   127→        rafRef.current = requestAnimationFrame(step)
   128→   128→      }
   129→   129→    }
   130→   130→    rafRef.current = requestAnimationFrame(step)
   131→   131→  }, [target])
   132→   132→
   133→   133→  useEffect(() => {
   134→   134→    if (inView) animate()
   135→   135→    return () => {
   136→   136→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   137→   137→    }
   138→   138→  }, [inView, animate])
   139→   139→
   140→   140→  return value
   141→   141→}
