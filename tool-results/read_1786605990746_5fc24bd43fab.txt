   117→const DESIGNATION_COLORS: Record<string, string> = {
   118→  HOD: 'bg-violet-100 text-violet-700 dark:bg-violet-900/50 dark:text-violet-400',
   119→  Director: 'bg-amber-100 text-amber-700 dark:bg-amber-900/50 dark:text-amber-400',
   120→  Professor: 'bg-violet-100 text-violet-700 dark:bg-violet-900/50 dark:text-violet-400',
   121→  Senior: 'bg-teal-100 text-teal-700 dark:bg-teal-900/50 dark:text-teal-400',
   122→  Consultant: 'bg-teal-100 text-teal-700 dark:bg-teal-900/50 dark:text-teal-400',
   123→  Junior: 'bg-sky-100 text-sky-700 dark:bg-sky-900/50 dark:text-sky-400',
   124→  Resident: 'bg-sky-100 text-sky-700 dark:bg-sky-900/50 dark:text-sky-400',
   125→}
   126→
   127→function getDesignationColor(designation: string): string {
   128→  if (!designation) return 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400'
   129→  for (const [key, color] of Object.entries(DESIGNATION_COLORS)) {
   130→    if (designation.toLowerCase().includes(key.toLowerCase())) return color
   131→  }
   132→  return 'bg-gray-100 text-gray-600 dark:bg-gray-800 dark:text-gray-400'
   133→}

   559→                  onChange={(e) => {

   561→                    if (selectedDoctor) {

   565→                  }}
   566→                />
   567→              </div>
   568→              {/* Search results dropdown */}
   569→              <AnimatePresence>
   570→                {searchDebounce.length >= 2 && searchResults && !selectedDoctor && (

   578→                      <div className="p-3 text-center text-sm text-muted-foreground">Searching...</div>

   594→                          </AvatarFallback>
   595→                        </Avatar>
   596→                        <div className="min-w-0">
   597→                          <p className="truncate font-medium">{doc.name}</p>
   598→                          <span className="truncate text-xs text-muted-foreground">{doc.specialization}</span>
   599→                        </div>
   600→                      </button>
   601→                    ))}
   602→                  </motion.div>
   603→                )}

   606→                <div className="flex items-center gap-2 rounded-md border border-teal-200 bg-teal-50 px-3 py-2 dark:border-teal-800 dark:bg-teal-900/30">

   610→                      {selectedDoctor.name.charAt(0)}
   611→                    </AvatarFallback>
   612→                  </Avatar>
   613→                  <span className="flex-1 text-sm font-medium">{selectedDoctor.name}</span>

   615→                    type="button"
   616→                    variant="ghost"
   617→                    size="sm"
   618→                    className="h-6 w-6 p-0"
   619→                    onClick={() => {
   620→                      setSelectedDoctor(null)
   621→                      setSearchDoctor('')
   622→                      setAddForm((f) => ({ ...f, doctorId: '' }))
   623→                    }}
   624→                  >
   625→                    ×
   626→                  </Button>
   627→                </div>
   628→              )}
   629→            </div>
   630→
   631→            {/* Department select */}
   632→            <div className="space-y-2">
   633→              <Label>Department *</Label>
   634→              <Select
   635→                value={addForm.departmentId}
   636→                onValueChange={(v) => setAddForm((f) => ({ ...f, departmentId: v }))}
   637→              >
   638→                <SelectTrigger className="w-full">
   639→                  <SelectValue placeholder="Select department" />
   640→                </SelectTrigger>
   641→                <SelectContent>
   642→                  {departments.map((d) => (
   643→                    <SelectItem key={d.id} value={d.id}>
   644→                      {d.name} ({d.doctorCount} doctors)
   645→                    </SelectItem>
   646→                  ))}
   647→                </SelectContent>
   648→              </Select>
   649→            </div>
   650→
   651→            {/* Designation */}
   652→            <div className="space-y-2">
   653→              <Label htmlFor="add-designation">Designation</Label>
   654→              <Input
   655→                id="add-designation"
   656→                placeholder="e.g. Senior Consultant, HOD"
   657→                value={addForm.designation}
   658→                onChange={(e) => setAddForm((f) => ({ ...f, designation: e.target.value }))}
   659→              />
   660→            </div>
   661→

   663→            <div className="space-y-2">

   666→                id="add-fees"

   668→                min="0"

   671→                onChange={(e) => setAddForm((f) => ({ ...f, fees: e.target.value }))}
   672→              />
   673→            </div>
   674→
   675→            {/* OPD Timings */}
   676→            <div className="space-y-2">
   677→              <Label htmlFor="add-opd">OPD Timings</Label>

   679→                id="add-opd"

   687→              <Button type="button" variant="outline" onClick={closeAddDialog} disabled={isAdding}>

   694→                    Linking...
   695→                  </span>
   696→                ) : (
   697→                  'Link Doctor'
   698→                )}
   699→              </Button>
   700→            </DialogFooter>
   701→          </form>
   702→        </DialogContent>
   703→      </Dialog>
   704→
   705→      {/* Edit Dialog */}
   706→      <Dialog open={editDialogOpen} onOpenChange={(open) => { if (!open) closeEditDialog() }}>
   707→        <DialogContent className="sm:max-w-lg">
   708→          <DialogHeader>
   709→            <DialogTitle>Edit Doctor Link</DialogTitle>
   710→            <DialogDescription>
   711→              Update consultation details for {selectedLink?.doctor.name}.
   712→            </DialogDescription>
   713→          </DialogHeader>
   714→          <form onSubmit={handleEditSubmit} className="space-y-4">

   717→                <AvatarImage src={selectedLink?.doctor.profileImg || ''} />
   718→                <AvatarFallback className="bg-teal-100 text-sm font-semibold text-teal-700 dark:bg-teal-900 dark:text-teal-300">
   719→                  {selectedLink?.doctor.name.split(' ').map((w) => w[0]).join('').slice(0, 2).toUpperCase()}
   720→                </AvatarFallback>
   721→              </Avatar>
   722→              <div>
   723→                <p className="text-sm font-semibold">{selectedLink?.doctor.name}</p>
   724→                <p className="text-xs text-muted-foreground">{selectedLink?.department.name}</p>
   725→              </div>
   726→            </div>
   727→

   729→              <Label htmlFor="edit-designation">Designation</Label>
   730→              <Input
   731→                id="edit-designation"
   732→                placeholder="e.g. Senior Consultant, HOD"
   733→                value={editForm.designation}
   734→                onChange={(e) => setEditForm((f) => ({ ...f, designation: e.target.value }))}
   735→              />
   736→            </div>
   737→
   738→            <div className="space-y-2">
   739→              <Label htmlFor="edit-fees">Consultation Fees (₹)</Label>
   740→              <Input
   741→                id="edit-fees"
   742→                type="number"
   743→                min="0"
   744→                placeholder="e.g. 500"
   745→                value={editForm.fees}
   746→                onChange={(e) => setEditForm((f) => ({ ...f, fees: e.target.value }))}
   747→              />
   748→            </div>
   749→
   750→            <div className="space-y-2">
   751→              <Label htmlFor="edit-opd">OPD Timings</Label>

   777→                {isUpdating ? (
   778→                  <span className="flex items-center gap-2">
   779→                    <span className="h-4 w-4 animate-spin rounded-full border-2 border-white/30 border-t-white" />
   780→                    Saving...
   781→                  </span>
   782→                ) : (
   783→                  'Save Changes'
   784→                )}
   785→              </Button>

   788→        </DialogContent>

   790→
   791→      {/* Delete/Unlink Confirmation */}

   795→            <AlertDialogTitle>Unlink Doctor</AlertDialogTitle>

   799→          </AlertDialogHeader>
   800→          <AlertDialogFooter>
   801→            <AlertDialogCancel>Cancel</AlertDialogCancel>
   802→            <AlertDialogAction
   803→              onClick={() => selectedLink && deleteMutation.mutate(selectedLink.id)}
   804→              disabled={deleteMutation.isPending}

   809→          </AlertDialogFooter>
   810→        </AlertDialogContent>
   811→      </AlertDialog>
   812→    </div>
   813→  )
