    60→function FadeUpSection({
    61→  children,
    62→  className,
    63→  delay = 0,
    64→}: {
    65→  children: React.ReactNode
    66→  className?: string
    67→  delay?: number
    68→}) {
    69→  const ref = useRef<HTMLDivElement>(null)
    70→  const inView = useInView(ref, { once: true, margin: '-60px' })
    71→  return (
    72→    <motion.div
    73→      ref={ref}
    74→      initial={{ opacity: 0, y: 30 }}
    75→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 30 }}
    76→      transition={{ duration: 0.6, delay, ease: 'easeOut' }}
    77→      className={className}
    78→    >
    79→      {children}
    80→    </motion.div>
    81→  )
    82→}

   108→// ── Count-up hook ────────────────────────────────────────────────────────────
   109→
   110→function useCountUp(target: number, inView: boolean) {
   111→  const [value, setValue] = useState(0)
   112→  const rafRef = useRef<number | null>(null)
   113→
   114→  const animate = useCallback(() => {
   115→    const duration = 1800
   116→    const start = performance.now()
   117→    const from = 0
   118→    const to = target
   119→
   120→    const step = (now: number) => {
   121→      const elapsed = now - start
   122→      const progress = Math.min(elapsed / duration, 1)
   123→      const eased = 1 - Math.pow(1 - progress, 3)
   124→      setValue(Math.floor(from + (to - from) * eased))
   125→      if (progress < 1) {
   126→        rafRef.current = requestAnimationFrame(step)
   127→      }
   128→    }
   129→    rafRef.current = requestAnimationFrame(step)
   130→  }, [target])
   131→
   132→  useEffect(() => {
   133→    if (inView) animate()
   134→    return () => {
   135→      if (rafRef.current) cancelAnimationFrame(rafRef.current)
   136→    }
   137→  }, [inView, animate])
   138→
   139→  return value
   140→}

   741→function StatsGrid({ stats }: { stats: Stats }) {
   742→  const ref = useRef<HTMLDivElement>(null)
   743→  const inView = useInView(ref, { once: true, margin: '100px' })
   744→  const [delayPassed, setDelayPassed] = useState(false)
   745→
   746→  useEffect(() => {
   747→    const id = setTimeout(() => setDelayPassed(true), 800)
   748→    return () => clearTimeout(id)
   749→  }, [])
   750→
   751→  const shouldAnimate = inView || delayPassed
   752→
   753→  const doctorCount = useCountUp(stats.doctorCount, shouldAnimate)
   754→  const hospitalCount = useCountUp(stats.hospitalCount, shouldAnimate)
   755→  const patientCount = useCountUp(stats.patientCount, shouldAnimate)
   756→  const bookingCount = useCountUp(stats.bookingCount, shouldAnimate)
   757→
   758→  const items = [
   759→    { icon: Stethoscope, value: doctorCount, label: 'Verified Doctors', suffix: '+' },

   762→    { icon: CalendarCheck, value: bookingCount, label: 'Appointments Booked', suffix: '+' },
   763→  ]
   764→
   765→  return (
   766→    <div ref={ref} className="grid grid-cols-2 gap-6 lg:grid-cols-4">
   767→      {items.map((item, i) => (
   768→        <FadeUpItem key={item.label} index={i} className="flex flex-col items-center text-center">
