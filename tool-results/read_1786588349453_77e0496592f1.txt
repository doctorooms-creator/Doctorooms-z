   437→function OPDQueueSection({ queueData }: { queueData: QueueResponse }) {
   438→  const { queue, stats, currentServing } = queueData
   439→  const [refreshKey, setRefreshKey] = useState(0)
   440→  const [lastRefreshed, setLastRefreshed] = useState(new Date())
   441→
   442→  const handleManualRefresh = useCallback(() => {
   443→    setRefreshKey((k) => k + 1)
   444→    setLastRefreshed(new Date())
   445→  }, [])
   446→
   447→  // Update last refreshed time on each refetch
   448→  useEffect(() => {
   449→    setLastRefreshed(new Date())
   450→  }, [queueData])
   451→
   452→  const waitingItems = queue.filter((q) => q.status === 'Approve')
   453→  const inConsultationItems = queue.filter((q) => q.status === 'Visited')
   454→  const completedItems = queue.filter((q) => q.status === 'Finish')
   455→
   456→  return (
   457→    <motion.div
   458→      initial={{ opacity: 0, y: 10 }}
   459→      animate={{ opacity: 1, y: 0 }}
   460→      exit={{ opacity: 0, y: 10 }}
   461→      transition={{ duration: 0.4, delay: 0.1 }}
   462→      className="space-y-4"
   463→    >
   464→      {/* Section header */}
   465→      <div className="flex items-center justify-between">
   466→        <div className="flex items-center gap-2">
   467→          <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-violet-100 dark:bg-violet-900/50">
   468→            <UserPlus className="h-4 w-4 text-violet-600 dark:text-violet-400" />
   469→          </div>
   470→          <div>
   471→            <h3 className="text-sm font-semibold">Today's OPD Queue</h3>
   472→            <p className="text-[11px] text-muted-foreground">Auto-refreshes every 15s</p>
   473→          </div>
   474→        </div>
   475→        <Button
   476→          variant="ghost"
   477→          size="sm"
   478→          onClick={handleManualRefresh}
   479→          className="text-muted-foreground hover:text-foreground"
   480→        >
   481→          <RefreshCw className="h-3.5 w-3.5 mr-1.5" />
   482→          Refresh
   483→        </Button>
   484→      </div>
   485→
   486→      {/* Current Serving Banner */}
   487→      {currentServing && (
   488→        <motion.div
   489→          key={currentServing.tokenNumber || 'serving'}
   490→          initial={{ opacity: 0, scale: 0.97 }}
   491→          animate={{ opacity: 1, scale: 1 }}
   492→          className="rounded-xl border-2 border-teal-300 dark:border-teal-700 bg-gradient-to-r from-teal-50 to-emerald-50 dark:from-teal-950/30 dark:to-emerald-950/20 p-4"
   493→        >
   494→          <div className="flex items-center gap-4">
   495→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-full bg-teal-100 dark:bg-teal-900/60">
   496→              <UserCheck className="h-6 w-6 text-teal-600 dark:text-teal-400" />
   497→            </div>
   498→            <div className="flex-1 min-w-0">
   499→              <p className="text-xs font-medium text-teal-600 dark:text-teal-400 uppercase tracking-wider">
   500→                Currently Serving
   501→              </p>
   502→              <div className="flex items-center gap-2 mt-0.5">
   503→                {currentServing.tokenNumber && (
   504→                  <span className="text-lg font-bold font-mono text-teal-700 dark:text-teal-300">
   505→                    {currentServing.tokenNumber}
   506→                  </span>
   507→                )}
   508→                <span className="text-sm font-medium text-foreground truncate">
   509→                  {currentServing.patientName}
   510→                </span>
   511→              </div>
   512→            </div>
   513→          </div>
   514→        </motion.div>
   515→      )}
   516→
   517→      {/* Queue stats row */}

   633→      )}
   634→    </motion.div>
